#!/usr/bin/env zsh
set -euo pipefail

# 固定比较自然排序后的前 300 张图。
# 改成 nvidia-smi 中五张空闲的物理 GPU 编号。
GPU0=0
GPU1=1
GPU2=2
GPU3=3
GPU4=4
MAX_FRAMES=300

ROOT=/home/wlh50060092/vggtSam
IMAGE_DIR=/home/bod/182Nas/share/BaZhuaYu/DemoData/xiaoyi/blind_data/meeting_room_a02/frames
STREAM_CKPT=/home/bod/86Nas/95_data_bak/FoundationModels/StreamVGGT/checkpoints.pth
OUTPUT_ROOT=/home/wlh50060092/vggtSam/outputs/long_sequence_baselines/frames_300
STREAM_PYTHON=/home/huawei/miniconda3/envs/3am/bin/python

cd "$ROOT"

[[ -x "$STREAM_PYTHON" ]] || { echo "找不到 3am Python: $STREAM_PYTHON"; exit 1; }

PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
CUDA_VISIBLE_DEVICES="$GPU0,$GPU1,$GPU2,$GPU3,$GPU4" "$STREAM_PYTHON" \
  -m long_sequence_baselines.run_streamvggt_5gpu \
  --image-dir "$IMAGE_DIR" \
  --checkpoint "$STREAM_CKPT" \
  --output-root "$OUTPUT_ROOT/streamvggt" \
  --scene-name meeting_room_a02 \
  --confidence-percentile 50 \
  --depth-percentile-low 1 \
  --depth-percentile-high 99 \
  --voxel-size-ratio 0.01 \
  --min-voxel-observations 2 \
  --save-deployment-features \
  --max-frames "$MAX_FRAMES"
