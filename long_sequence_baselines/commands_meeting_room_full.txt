#!/usr/bin/env zsh
set -euo pipefail

# 固定比较自然排序后的前 300 张图；两个任务串行，不会同时占卡。
# 改成 nvidia-smi 中五张空闲的物理 GPU 编号。
GPU0=0
GPU1=1
GPU2=2
GPU3=3
GPU4=4
MAX_FRAMES=300

ROOT=/home/wlh50060092/vggtSam
IMAGE_DIR=/home/bod/182Nas/share/BaZhuaYu/DemoData/xiaoyi/blind_data/meeting_room_a02/frames
HORIZON_CKPT=/home/wlh50060092/vggtSam/models/HorizonStream.pt
STREAM_CKPT=/home/bod/86Nas/95_data_bak/FoundationModels/StreamVGGT/checkpoints.pth
OUTPUT_ROOT=/home/wlh50060092/vggtSam/outputs/long_sequence_baselines/frames_300
HORIZON_PYTHON=/home/huawei/miniconda3/envs/horizonstream/bin/python
STREAM_PYTHON=/home/huawei/miniconda3/envs/3am/bin/python

cd "$ROOT"

[[ -x "$HORIZON_PYTHON" ]] || { echo "找不到 HorizonStream Python: $HORIZON_PYTHON"; exit 1; }
[[ -x "$STREAM_PYTHON" ]] || { echo "找不到 3am Python: $STREAM_PYTHON"; exit 1; }

PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
CUDA_VISIBLE_DEVICES="$GPU0" "$HORIZON_PYTHON" \
  -m long_sequence_baselines.run_horizonstream \
  --image-dir "$IMAGE_DIR" \
  --checkpoint "$HORIZON_CKPT" \
  --output-root "$OUTPUT_ROOT/horizonstream" \
  --scene-name meeting_room_a02 \
  --sliding-size 10 \
  --confidence-percentile 50 \
  --depth-percentile-low 1 \
  --depth-percentile-high 99 \
  --voxel-size-ratio 0.01 \
  --min-voxel-observations 2 \
  --max-frames "$MAX_FRAMES"

PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True \
CUDA_VISIBLE_DEVICES="$GPU0,$GPU1,$GPU2,$GPU3,$GPU4" "$STREAM_PYTHON" \
  -m long_sequence_baselines.run_streamvggt_5gpu \
  --image-dir "$IMAGE_DIR" \
  --checkpoint "$STREAM_CKPT" \
  --output-root "$OUTPUT_ROOT/streamvggt" \
  --scene-name meeting_room_a02 \
  --confidence-percentile 50 \
  --depth-percentile-low 1 \
  --depth-percentile-high 99 \
  --voxel-size-ratio 0.01 \
  --min-voxel-observations 2 \
  --save-deployment-features \
  --max-frames "$MAX_FRAMES"

HORIZON_INPUTS="$OUTPUT_ROOT/horizonstream/input_images.txt"
STREAM_INPUTS="$OUTPUT_ROOT/streamvggt/meeting_room_a02/input_images.txt"
cmp "$HORIZON_INPUTS" "$STREAM_INPUTS"
[[ "$(wc -l < "$HORIZON_INPUTS")" -eq "$MAX_FRAMES" ]] || {
  echo "输入帧数不是 $MAX_FRAMES"; exit 1
}
for METHOD_ROOT in \
  "$OUTPUT_ROOT/horizonstream/meeting_room_a02" \
  "$OUTPUT_ROOT/streamvggt/meeting_room_a02"; do
  [[ -f "$METHOD_ROOT/plots/trajectory_compare.png" ]] || {
    echo "缺少统一位姿图: $METHOD_ROOT/plots/trajectory_compare.png"; exit 1
  }
  for PRODUCT in depthpose_raw depthpose_conf depthpose_conf_voxel; do
    [[ -f "$METHOD_ROOT/points/$PRODUCT.ply" ]] || {
      echo "缺少点云结果: $METHOD_ROOT/points/$PRODUCT.ply"; exit 1
    }
  done
done
for PRODUCT in pointhead_raw pointhead_conf pointhead_conf_voxel; do
  [[ -f "$OUTPUT_ROOT/streamvggt/meeting_room_a02/points/$PRODUCT.ply" ]] || {
    echo "缺少 StreamVGGT point-head 结果: $PRODUCT.ply"; exit 1
  }
done
echo "300-frame comparison complete; input lists are identical."
