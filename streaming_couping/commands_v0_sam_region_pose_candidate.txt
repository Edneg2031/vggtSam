#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="streaming_couping/configs/v0_baseline.yaml"
OUTPUT_DIR="outputs/streaming_couping_v0/sam_region_pose_candidate"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s outputs/streaming_couping_v0/baseline_summary.json ]] || {
  echo "missing V0 baseline; first run BASELINE_REBUILD_CACHE=1 zsh streaming_couping/commands_v0_baseline.txt" >&2
  exit 1
}

"$PYTHON_BIN" -c '
import json
from pathlib import Path
summary = json.loads(Path("outputs/streaming_couping_v0/baseline_summary.json").read_text())
expected = ["bed", "wardrobe", "picture", "mat", "chair"]
if summary.get("configured_instance_prompts") != expected:
    raise SystemExit(
        "stale V0 cache: run BASELINE_REBUILD_CACHE=1 "
        "zsh streaming_couping/commands_v0_baseline.txt"
    )
if summary.get("configured_permanent_slot_capacity") != 16:
    raise SystemExit(
        "stale V0 registry: run BASELINE_REBUILD_CACHE=1 "
        "zsh streaming_couping/commands_v0_baseline.txt"
    )
'

echo "V0 MESA-lite smoke: SAM identity filter, equal-count support, PnP and GT isolation"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_sam_region_pose_candidate

echo "V0 MESA-lite SAM-region pose candidate"
echo "No model is downloaded or trained; V0 selected pose remains raw StreamVGGT"
echo "GT is decoded only after all candidates are immutable"
PYTHONHASHSEED=0 PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_sam_region_pose_candidate \
  --config "$CONFIG"

REPORT="$OUTPUT_DIR/copyable_result.txt"
[[ -s "$REPORT" ]] || { echo "missing candidate report: $REPORT" >&2; exit 1; }
[[ -s "$OUTPUT_DIR/frame_diagnostics.csv" ]] || { echo "missing frame diagnostics" >&2; exit 1; }
[[ -s "$OUTPUT_DIR/fold_decision.csv" ]] || { echo "missing fold decision" >&2; exit 1; }

echo "Result to send back:"
sed -n '/COPYABLE_V0_SAM_REGION_POSE_RESULT_BEGIN/,/COPYABLE_V0_SAM_REGION_POSE_RESULT_END/p' "$REPORT"
