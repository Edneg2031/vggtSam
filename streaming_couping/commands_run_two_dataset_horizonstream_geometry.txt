#!/usr/bin/env zsh
set -euo pipefail

# Geometry-only HorizonStream diagnostic for the two requested RGB datasets.
# Defaults follow the proposed memory-safe comparison: every second source
# frame and a HorizonStream sliding step of five.  No SAM3, DINO, or semantic
# mapping inference is loaded.

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
HORIZON_PYTHON="${HORIZONSTREAM_PYTHON:-/home/huawei/miniconda3/envs/horizonstream/bin/python}"
HORIZON_REPO="$ROOT/externals/horizonstream"
HORIZON_CHECKPOINT="${HORIZONSTREAM_CHECKPOINT:-/home/bod/86Nas/95_data_bak/FoundationModels/HorizonStream.pt}"

WUHAN_FRAMES="/home/bod/182Nas/share/BaZhuaYu/DemoData/xiaoyi/wuhan_A3/images"
MEETING_ROOM_FRAMES="/home/bod/182Nas/share/BaZhuaYu/DemoData/xiaoyi/blind_data/meeting_room_a02/frames"

FRAME_START="${FRAME_START:-0}"
FRAME_STRIDE="${FRAME_STRIDE:-2}"
FRAME_COUNT="${FRAME_COUNT:-0}"
HORIZON_WINDOW_SIZE="${HORIZONSTREAM_WINDOW_SIZE:-10}"
HORIZON_SLIDING_SIZE="${HORIZONSTREAM_SLIDING_SIZE:-5}"
HORIZON_IMAGE_SIZE="${HORIZONSTREAM_IMAGE_SIZE:-518}"
HORIZON_PATCH_SIZE="${HORIZONSTREAM_PATCH_SIZE:-14}"
HORIZON_PRECISION="${HORIZONSTREAM_PRECISION:-float16}"
VOXEL_SIZE_M="${GEOMETRY_VOXEL_SIZE_M:-0.05}"
MIN_GEOMETRY_CONFIDENCE="${GEOMETRY_MIN_CONFIDENCE:-0.30}"
MAX_SCENE_VOXELS="${GEOMETRY_MAX_SCENE_VOXELS:-1000000}"
HORIZONSTREAM_DEVICE="${HORIZONSTREAM_DEVICE:-}"

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1
export PYTORCH_CUDA_ALLOC_CONF="${PYTORCH_CUDA_ALLOC_CONF:-expandable_segments:True}"
export PYTHONPATH="$HORIZON_REPO:$ROOT${PYTHONPATH:+:$PYTHONPATH}"

[[ -x "$HORIZON_PYTHON" ]] || {
  print -u2 "missing HorizonStream Python: $HORIZON_PYTHON"
  exit 1
}
[[ -d "$HORIZON_REPO/horizonstream" ]] || {
  print -u2 "missing HorizonStream source: $HORIZON_REPO"
  exit 1
}
[[ -f "$HORIZON_CHECKPOINT" ]] || {
  print -u2 "missing HorizonStream checkpoint: $HORIZON_CHECKPOINT"
  exit 1
}

if [[ -z "$HORIZONSTREAM_DEVICE" ]]; then
  (( $+commands[nvidia-smi] )) || {
    print -u2 "nvidia-smi is required when HORIZONSTREAM_DEVICE is not set"
    exit 1
  }
  GPU_MEMORY_TABLE="$(nvidia-smi --query-gpu=index,memory.free,memory.total \
    --format=csv,noheader,nounits)" || {
    print -u2 "unable to query GPU memory with nvidia-smi"
    exit 1
  }
  print "GPU memory at launch (index, free MiB, total MiB):"
  print -r -- "$GPU_MEMORY_TABLE"
  HORIZON_GPU_INDEX="$(
    print -r -- "$GPU_MEMORY_TABLE" | awk -F, '
      {
        gsub(/[[:space:]]/, "", $1)
        gsub(/[[:space:]]/, "", $2)
        if ($2 >= 10000 && $2 > best_free) {
          best_free = $2
          best_index = $1
        }
      }
      END { if (best_index != "") print best_index }
    '
  )"
  [[ -n "$HORIZON_GPU_INDEX" ]] || {
    print -u2 "no GPU has at least 10000 MiB free for HorizonStream"
    exit 1
  }
  HORIZONSTREAM_DEVICE="cuda:${HORIZON_GPU_INDEX}"
fi

print "geometry_only=true"
print "frame_start=${FRAME_START} frame_stride=${FRAME_STRIDE} frame_count=${FRAME_COUNT}"
print "window=${HORIZON_WINDOW_SIZE} sliding=${HORIZON_SLIDING_SIZE} precision=${HORIZON_PRECISION}"
print "horizon_device=${HORIZONSTREAM_DEVICE} voxel_size_m=${VOXEL_SIZE_M} min_confidence=${MIN_GEOMETRY_CONFIDENCE}"

print_gpu_memory() {
  local gpu_index="${HORIZONSTREAM_DEVICE#cuda:}"
  if (( $+commands[nvidia-smi] )); then
    nvidia-smi --id="$gpu_index" \
      --query-gpu=index,memory.used,memory.free,memory.total \
      --format=csv,noheader,nounits || print "nvidia-smi query failed"
  fi
}

run_geometry() {
  local dataset_name="$1"
  local frames_dir="$2"
  local output_dir="$STORAGE_ROOT/outputs/${dataset_name}_horizonstream_geometry_stride${FRAME_STRIDE}_slide${HORIZON_SLIDING_SIZE}"
  local geometry_cache="$output_dir/horizonstream_geometry.pt"

  [[ -d "$frames_dir" ]] || {
    print -u2 "missing RGB directory for ${dataset_name}: $frames_dir"
    return 1
  }
  mkdir -p "$output_dir"

  print "===== ${dataset_name}: HorizonStream geometry inference ====="
  print "HorizonStream GPU memory before loading (MiB):"
  print_gpu_memory
  "$HORIZON_PYTHON" \
    -m streaming_couping.scripts.generate_horizonstream_geometry_cache \
    --frames "$frames_dir" \
    --scene-id "$dataset_name" \
    --frame-start "$FRAME_START" \
    --frame-stride "$FRAME_STRIDE" \
    --frame-count "$FRAME_COUNT" \
    --repo-path "$HORIZON_REPO" \
    --checkpoint "$HORIZON_CHECKPOINT" \
    --output-cache "$geometry_cache" \
    --device "$HORIZONSTREAM_DEVICE" \
    --window-size "$HORIZON_WINDOW_SIZE" \
    --sliding-size "$HORIZON_SLIDING_SIZE" \
    --image-size "$HORIZON_IMAGE_SIZE" \
    --patch-size "$HORIZON_PATCH_SIZE" \
    --crop \
    --precision "$HORIZON_PRECISION" \
    --reuse-if-valid

  print "===== ${dataset_name}: pure geometry export ====="
  "$HORIZON_PYTHON" \
    -m streaming_couping.scripts.export_horizonstream_geometry \
    --cache "$geometry_cache" \
    --output-dir "$output_dir" \
    --voxel-size-m "$VOXEL_SIZE_M" \
    --min-geometry-confidence "$MIN_GEOMETRY_CONFIDENCE" \
    --max-scene-voxels "$MAX_SCENE_VOXELS"
  print "geometry_output=${output_dir}"
}

run_geometry "wuhan_A3" "$WUHAN_FRAMES"
run_geometry "meeting_room_a02" "$MEETING_ROOM_FRAMES"

print "===== TWO-DATASET HORIZONSTREAM GEOMETRY COMPLETE ====="
