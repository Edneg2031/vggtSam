#!/usr/bin/env zsh
set -euo pipefail

# One-command causal DINOv3 patch-to-patch geometry feasibility diagnostic.
# All paths and settings are fixed for the shared server environment.  The
# command refreshes only the frozen DINO dense-feature cache; it does not
# rerun StreamVGGT or SAM3 and it does not train or modify XYZ.
ROOT="/home/wlh50060092/vggtSam"
STORAGE_ROOT="/data184/open_source/vggtSam"
PYTHON_BIN="/home/huawei/miniconda3/envs/3am/bin/python"
DINO_ROOT="/home/bod/86Nas/95_data_bak/FoundationModels/dinov3"
PROTOCOL="${STORAGE_ROOT}/outputs/streaming_couping_multiclip/protocol.yaml"
FEATURE_DIR="${STORAGE_ROOT}/outputs/streaming_couping_dinov3_object_features"
OUTPUT_DIR="${STORAGE_ROOT}/outputs/streaming_couping_dinov3_patch_geometry_retrieval"

cd "${ROOT}"
[[ -x "${PYTHON_BIN}" ]] || {
  echo "missing Python: ${PYTHON_BIN}" >&2
  exit 1
}
[[ -s "${PROTOCOL}" ]] || {
  echo "missing protocol: ${PROTOCOL}" >&2
  echo "run the existing multi-scene setup command first" >&2
  exit 1
}
[[ -d "${STORAGE_ROOT}/outputs/streaming_couping_v0/cache" ]] || {
  echo "missing frozen StreamVGGT/SAM cache directory" >&2
  exit 1
}
[[ -d "${DINO_ROOT}/dinov3-vitl16" ]] || {
  echo "missing DINOv3 checkpoint directory: ${DINO_ROOT}/dinov3-vitl16" >&2
  exit 1
}

export VGGT_SAM_STORAGE_ROOT="${STORAGE_ROOT}"
export PYTHONUNBUFFERED=1
export PYTHONPATH="${ROOT}"

echo "[1/2] refresh frozen DINOv3 dense patch-feature cache"
CUDA_VISIBLE_DEVICES=3 \
  "${PYTHON_BIN}" -m streaming_couping.scripts.cache_dinov3_object_features \
  --config "${PROTOCOL}" \
  --output-dir "${FEATURE_DIR}" \
  --dinov3-root "${DINO_ROOT}" \
  --preferred-variant dinov3-vitl16 \
  --device cuda:0 \
  --dtype bfloat16 \
  --ema-beta 0.90 \
  --batch-size 1 \
  --raw-cache-variant sam31_online_forward \
  --overwrite

echo "[2/2] generate causal patch-retrieval diagnostics"
CUDA_VISIBLE_DEVICES=3 \
  "${PYTHON_BIN}" -m streaming_couping.scripts.analyze_dinov3_patch_geometry_retrieval \
  --protocol "${PROTOCOL}" \
  --feature-dir "${FEATURE_DIR}" \
  --output-dir "${OUTPUT_DIR}" \
  --device cuda:0 \
  --seed 2026 \
  --max-patches-per-object-frame 32 \
  --max-gt-surface-points 4096 \
  --mutual-nearest

echo "===== COPYABLE DIAGNOSTIC SUMMARY ====="
sed -n '1,240p' "${OUTPUT_DIR}/copyable_result.txt"
echo "DINOv3 patch geometry retrieval output=${OUTPUT_DIR}"
