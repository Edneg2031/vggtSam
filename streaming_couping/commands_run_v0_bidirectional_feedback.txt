#!/usr/bin/env zsh
set -euo pipefail

# This command is intentionally scoped to the retained one-scene V0 baseline.
# It reuses the completed V0 cache and does not rerun StreamVGGT or SAM3.1.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="streaming_couping/configs/v0_baseline.yaml"
OUTPUT_DIR="$STORAGE_ROOT/outputs/streaming_couping_v0_bidirectional_feedback"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V0 explicit bidirectional feedback diagnostic"
echo "models are not rerun; frozen V0 cache is reused"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_bidirectional_feedback
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v0_bidirectional_feedback \
  --config "$CONFIG" \
  --output-dir "$OUTPUT_DIR"

echo "V0 bidirectional feedback diagnostic output=$OUTPUT_DIR"
