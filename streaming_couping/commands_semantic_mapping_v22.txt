#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="${SEMANTIC_MAPPING_CONFIG:-streaming_couping/configs/v0_baseline.yaml}"
OUTPUT_DIR="${SEMANTIC_MAPPING_V22_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_v22/semantic_map}"
SAM_GPU="${V22_SAM_GPU:-3}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$CONFIG" ]] || { echo "missing config: $CONFIG" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

echo "Stage V2.2 failure-only world-space geometry-consistency semantic-map export"
echo "models are frozen; V0 cache and QK pose are reused"
echo "physical SAM GPU=$SAM_GPU logical SAM device=cuda:0"
PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" CUDA_VISIBLE_DEVICES="$SAM_GPU" \
  "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v22_geometry_validation

PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" CUDA_VISIBLE_DEVICES="$SAM_GPU" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_v22_semantic_map \
  --config "$CONFIG" \
  --output-dir "$OUTPUT_DIR" \
  --sam3-device cuda:0 \
  "$@"

SUMMARY="$OUTPUT_DIR/semantic_map_v22_summary.json"
[[ -s "$SUMMARY" ]] || { echo "missing V2.2 semantic_map_v22_summary.json" >&2; exit 1; }
[[ -s "$OUTPUT_DIR/semantic_map.pt" ]] || {
  echo "missing V2.2 semantic_map.pt" >&2
  exit 1
}
[[ -s "$OUTPUT_DIR/recovery_events.csv" ]] || {
  echo "missing V2.2 recovery_events.csv" >&2
  exit 1
}

export SUMMARY
PYTHONPATH="$ROOT" "$PYTHON_BIN" -c '
import json
import os
from pathlib import Path

summary = json.loads(Path(os.environ["SUMMARY"]).read_text())
assert summary["identity_mode"] == "v2_2_failure_only_geometry_consistency_recovery"
assert summary["baseline_status"] == "frozen"
assert summary["map_generation_gt_fields"] == 0
assert summary["streamvggt_frozen"] == 1
assert summary["sam3_frozen"] == 1
assert summary["pointmap_modified"] == 0
assert summary["selected_pose_branch"] == "retrieve_qk"
assert summary["semantic_map_pipeline_ready"] == 1
stats = summary["recovery_stats"]
allowed = {"centroid_far", "low_point_overlap", "shape_inconsistent"}
assert set(stats["geometry_validation_reject_reasons"]) <= allowed
assert float(stats["normal_frame_unchanged_ratio"]) >= 1.0 - 1e-12
assert float(stats["raw_fallback_unchanged_ratio"]) >= 1.0 - 1e-12
print(
    "V2.2 export accepted triggers={} geometry_accept={} geometry_reject={} "
    "reasons={} raw_fallback_unchanged={:.4f}".format(
        stats["recovery_trigger_count"],
        stats["geometry_validation_accept_count"],
        stats["geometry_validation_reject_count"],
        stats["geometry_validation_reject_reasons"],
        float(stats["raw_fallback_unchanged_ratio"]),
    )
)
'

echo "===== V2.2 COPYABLE RESULT ====="
sed -n '1,160p' "$OUTPUT_DIR/copyable_result.txt"
