#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
BASE_CONFIG="${SEMANTIC_MAPPING_BASE_CONFIG:-streaming_couping/configs/v0_baseline.yaml}"
OUTPUT_CONFIG="${SEMANTIC_MAPPING_MULTICLIP_CONFIG:-$STORAGE_ROOT/outputs/streaming_couping_multiclip/protocol.yaml}"
PER_CLIP_DIR="${SEMANTIC_MAPPING_MULTICLIP_RUN_ROOT:-$STORAGE_ROOT/outputs/streaming_couping_multiclip}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$BASE_CONFIG" ]] || { echo "missing base config: $BASE_CONFIG" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

SCENE_ARGS=()
[[ -n "${SEMANTIC_MAPPING_SCENES:-}" ]] && SCENE_ARGS+=(--scene-ids "$SEMANTIC_MAPPING_SCENES")

echo "Plan multi-scene semantic-map protocol"
PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
  "$PYTHON_BIN" -m streaming_couping.scripts.plan_multiclip_semantic_mapping \
  --base-config "$BASE_CONFIG" \
  --output-config "$OUTPUT_CONFIG" \
  --per-clip-dir "$PER_CLIP_DIR" \
  "${SCENE_ARGS[@]}" \
  "$@"
