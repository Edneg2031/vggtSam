#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V73_GPU:-1}"
STEPS="${V73_LONG_STEPS:-3000}"
OUTPUT=outputs/streaming_couping_v73_long_capacity
LONG_CACHE=outputs/streaming_couping_v7_fusion_ablation/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s outputs/streaming_couping_v71_instance_causality/frozen_l0.pt ]] || {
  echo "V7.3 long capacity requires the completed V7.1 frozen L0." >&2
  exit 1
}
[[ -s "$LONG_CACHE" ]] || {
  echo "V7.3 long capacity requires the augmented V7.2 long cache: $LONG_CACHE" >&2
  exit 1
}

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V7.3 long capacity: dependency-free tensor checks"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v73_runtime

RESUME_ARGS=(--resume)
if [[ "${V73_LONG_FRESH:-0}" == "1" ]]; then
  RESUME_ARGS=()
fi

echo "V7.3 long capacity GPU=$GPU K=8 steps=$STEPS"
echo "Training all active frames from 105 to 525; this is capacity, not generalization"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v73_long_capacity \
  --config streaming_couping/configs/v73_correspondence_ablation.yaml \
  --output-dir "$OUTPUT" \
  --device cuda:0 \
  --token-count 8 \
  --steps "$STEPS" \
  "${RESUME_ARGS[@]}"

[[ -s "$OUTPUT/v73_long_capacity.csv" ]] || {
  echo "missing V7.3 long capacity CSV" >&2
  exit 1
}
echo "V7.3 LONG CAPACITY COPY THIS CSV"
cat "$OUTPUT/v73_long_capacity.csv"
