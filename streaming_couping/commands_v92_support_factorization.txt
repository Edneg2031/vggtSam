#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${V92_STREAM_GPU0:-1}"
STREAM_GPU1="${V92_STREAM_GPU1:-2}"
SAM_GPU="${V92_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v92_support_factorization.yaml
DATA_CONFIG=streaming_couping/configs/v92_support_data.yaml
OUTPUT="${V92_OUTPUT:-outputs/streaming_couping_v92_support_factorization}"
CACHE=outputs/streaming_couping_v92_support_factorization/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V9.2 smoke: nested support, collision policies, schemas and fallback inputs"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.smoke_v92_support_factorization

NEED_REBUILD=0
if [[ "${V92_REBUILD_CACHE:-0}" == "1" || ! -s "$CACHE" ]]; then
  NEED_REBUILD=1
else
  echo "V9.2 validating its standalone 256-token dynamic-instance cache"
  if ! PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.audit_v7_cache \
    --config "$DATA_CONFIG" \
    --output-dir "$OUTPUT/cache_audit_preflight" \
    --require-local \
    --strict; then
    NEED_REBUILD=1
  fi
fi

if (( NEED_REBUILD )); then
  echo "V9.2 building a clean 30-frame/256-token cache"
  echo "physical GPUs=$VISIBLE_GPUS; StreamVGGT=cuda:0,cuda:1; SAM3.1=cuda:2"
  echo "dynamic slots=8; prompts=bed,wardrobe; later instance birth remains enabled"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
  PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.run_instance_token_pose \
    --config "$DATA_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0 \
    --training-device cuda:0 \
    --rebuild-cache
else
  echo "V9.2 reuses only its own validated cache; both backbones stay unloaded"
fi

[[ -s "$CACHE" ]] || { echo "missing V9.2 cache: $CACHE" >&2; exit 1; }
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

echo "V9.2 CPU oracle: current query32 x history key32/64/128/256"
echo "strategies=nearest,mutual,greedy_unique; no model is trained"
CUDA_VISIBLE_DEVICES="" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v92_support_factorization \
  --config "$CONFIG" \
  --output-dir "$OUTPUT"

SUMMARY="$OUTPUT/v92_support_summary.csv"
FRAMES="$OUTPUT/v92_selected_frames.csv"
DECISION="$OUTPUT/v92_support_decision.md"
METADATA="$OUTPUT/v92_support_metadata.json"
for file in "$SUMMARY" "$FRAMES" "$DECISION" "$METADATA"; do
  [[ -s "$file" ]] || { echo "missing V9.2 output: $file" >&2; exit 1; }
done

echo "V9.2 COPY THIS DECISION"
cat "$DECISION"
echo "V9.2 COPY THIS COMPACT SUMMARY CSV (36 rows)"
cat "$SUMMARY"
echo "V9.2 selected-frame diagnostics: $FRAMES"
echo "V9.2 provenance: $METADATA"
