#!/usr/bin/env zsh
set -euo pipefail

# Real SAM A/B on the retained, scene-disjoint V0 baseline scene.  The A--E
# candidate matrix must already exist.  StreamVGGT is not loaded or rerun.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="streaming_couping/configs/v0_baseline.yaml"
CANDIDATE_DIR="${V0_TEMPORAL_PROMPT_MATRIX_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_v0_temporal_prompt_matrix}"
OUTPUT_DIR="${V0_TEMPORAL_PROMPT_SAM_AB_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_v0_temporal_prompt_sam_ab}"

# The default is intentionally a small deterministic subset.  Set
# V0_TEMPORAL_PROMPT_MAX_QUERIES=0 to run all frozen causal frame-slot queries.
MAX_QUERIES="${V0_TEMPORAL_PROMPT_MAX_QUERIES:-96}"
SAM_GPU="${V0_TEMPORAL_PROMPT_SAM_GPU:-3}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}
[[ -s "$CANDIDATE_DIR/candidate_metadata.json" ]] || {
  echo "missing frozen A--E candidate artifacts: $CANDIDATE_DIR" >&2
  echo "run zsh streaming_couping/commands_run_v0_temporal_prompt_matrix.txt first" >&2
  exit 1
}

export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1
export PYTHONPATH="$ROOT"

"$PYTHON_BIN" -m streaming_couping.scripts.smoke_temporal_prompt_sam_ab

echo "real SAM A/B: frozen candidates, single-frame points-only calls"
echo "physical SAM GPU=$SAM_GPU common_query_limit=$MAX_QUERIES"
CUDA_VISIBLE_DEVICES="$SAM_GPU" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_v0_temporal_prompt_sam_ab \
    --config "$CONFIG" \
    --candidate-dir "$CANDIDATE_DIR" \
    --output-dir "$OUTPUT_DIR" \
    --sam3-device cuda:0 \
    --max-query-events "$MAX_QUERIES" \
    --max-positive-points 5 \
    --score-fallback-margin 0.10 \
    --score-fallback-ratio 0.80 \
    --worsened-iou-margin 0.05

echo "===== COPYABLE REAL SAM A/B RESULT ====="
sed -n '1,220p' "$OUTPUT_DIR/copyable_result.txt"
echo "V0 real SAM A/B output=$OUTPUT_DIR"
