#!/usr/bin/env zsh
set -euo pipefail

# One-command V2.4 evaluation-only map-safe gate.
# The residual models are reused from the previous DINOv3 experiment; this
# command does not retrain them and only evaluates a causal world-space gate.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
PROTOCOL="$STORAGE_ROOT/outputs/streaming_couping_multiclip/protocol.yaml"
FEATURE_DIR="$STORAGE_ROOT/outputs/streaming_couping_dinov3_object_features"
BASE_OUTPUT_DIR="$STORAGE_ROOT/outputs/streaming_couping_dinov3_object_geometry"
MODEL_PATH="$BASE_OUTPUT_DIR/models.pt"
OUTPUT_DIR="$STORAGE_ROOT/outputs/streaming_couping_dinov3_object_geometry_v24"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

if [[ ! -s "$PROTOCOL" ]]; then
  zsh streaming_couping/commands_plan_semantic_mapping_multiclip.txt
fi

feature_files=("$FEATURE_DIR"/*.pt(N))
if (( ${#feature_files} < 4 )); then
  zsh streaming_couping/commands_cache_dinov3_object_features.txt
fi

if [[ ! -s "$MODEL_PATH" ]]; then
  zsh streaming_couping/commands_run_dinov3_object_geometry.txt
fi

CUDA_VISIBLE_DEVICES=3 \
  PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
  "$PYTHON_BIN" -m streaming_couping.scripts.smoke_dinov3_object_geometry

CUDA_VISIBLE_DEVICES=3 \
  PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_dinov3_object_geometry_v24 \
  --protocol "$PROTOCOL" \
  --feature-dir "$FEATURE_DIR" \
  --model-path "$MODEL_PATH" \
  --output-dir "$OUTPUT_DIR" \
  --device cuda:0

echo "DINOv3 V2.4 map-safe gate output=$OUTPUT_DIR"
