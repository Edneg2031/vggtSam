#!/usr/bin/env zsh
set -euo pipefail

# Frozen, geometry-only A--E diagnostic for the retained V0 baseline scene.
# The command never reruns SAM/StreamVGGT.  The Python runner writes causal
# candidates first and opens GT only in its sealed evaluation phase.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="streaming_couping/configs/v0_baseline.yaml"
OUTPUT_DIR="$STORAGE_ROOT/outputs/streaming_couping_v0_temporal_prompt_matrix"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V0 temporal prompt A-E matrix diagnostic"
echo "models are not rerun; frozen V0 cache is reused"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_temporal_prompt_matrix
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v0_temporal_prompt_matrix \
  --config "$CONFIG" \
  --output-dir "$OUTPUT_DIR" \
  --depth-relative-tolerances "0.02,0.05,0.10,0.15,0.20,0.30,0.50"

echo "V0 temporal prompt matrix output=$OUTPUT_DIR"
