#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="${SEMANTIC_MAPPING_CONFIG:-${SEMANTIC_MAPPING_MULTICLIP_CONFIG:-streaming_couping/configs/v0_baseline.yaml}}"
OUTPUT_DIR="${SEMANTIC_MAPPING_ABLATION_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_semantic_map_multiclip_ablation}"
VARIANTS="${SEMANTIC_MAPPING_ABLATION_VARIANTS:-raw,v21,v22,v23_all_visible,v23}"

MULTICLIP_ROOT="${SEMANTIC_MAPPING_MULTICLIP_RUN_ROOT:-}"
if [[ -n "$MULTICLIP_ROOT" ]]; then
  V21_DEFAULT="$MULTICLIP_ROOT/v21"
  V22_DEFAULT="$MULTICLIP_ROOT/v22"
  V23_DEFAULT="$MULTICLIP_ROOT/v23"
else
  V21_DEFAULT="$STORAGE_ROOT/outputs/streaming_couping_v21/semantic_map"
  V22_DEFAULT="$STORAGE_ROOT/outputs/streaming_couping_v22/semantic_map"
  V23_DEFAULT="$STORAGE_ROOT/outputs/streaming_couping_v23/semantic_map"
fi
V21_ROOT="${SEMANTIC_MAPPING_V21_ROOT:-$V21_DEFAULT}"
V22_ROOT="${SEMANTIC_MAPPING_V22_ROOT:-$V22_DEFAULT}"
V23_ROOT="${SEMANTIC_MAPPING_V23_ROOT:-$V23_DEFAULT}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$CONFIG" ]] || { echo "missing config: $CONFIG" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

CLIP_ARGS=()
if [[ -n "${SEMANTIC_MAPPING_CLIPS:-}" ]]; then
  for clip in ${(s:,:)SEMANTIC_MAPPING_CLIPS}; do
    [[ -n "$clip" ]] && CLIP_ARGS+=(--clip "$clip")
  done
fi

echo "Multi-clip semantic-map offline ablation"
echo "models are not rerun; frozen caches and map artifacts are reused"
echo "variants=$VARIANTS"
echo "v21_root=$V21_ROOT"
echo "v22_root=$V22_ROOT"
echo "v23_root=$V23_ROOT"

PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
  "$PYTHON_BIN" -m streaming_couping.scripts.smoke_multiclip_semantic_ablation

PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_multiclip_semantic_ablation \
  --config "$CONFIG" \
  --output-dir "$OUTPUT_DIR" \
  --variants "$VARIANTS" \
  --v21-root "$V21_ROOT" \
  --v22-root "$V22_ROOT" \
  --v23-root "$V23_ROOT" \
  "${CLIP_ARGS[@]}" \
  "$@"

SUMMARY="$OUTPUT_DIR/semantic_map_multiclip_ablation_summary.json"
[[ -s "$SUMMARY" ]] || {
  echo "missing multi-clip ablation summary: $SUMMARY" >&2
  exit 1
}
[[ -s "$OUTPUT_DIR/clip_variant_summary.csv" ]] || {
  echo "missing clip_variant_summary.csv" >&2
  exit 1
}
echo "===== MULTI-CLIP ABLATION COPYABLE RESULT ====="
sed -n '1,240p' "$OUTPUT_DIR/copyable_result.txt"
