#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V73_GPU:-1}"
OUTPUT=outputs/streaming_couping_v73_smoke

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

echo "V7.3 smoke 1/2: unit tests"
PYTHONPATH=. "$PYTHON_BIN" -m pytest -q \
  streaming_couping/tests/test_v73_correspondence_fusion.py \
  streaming_couping/tests/test_v73_reporting.py \
  streaming_couping/tests/test_v72_local_fusion.py

echo "V7.3 smoke 2/2: two steps, complete K=8 mechanism ladder"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v73_correspondence_ablation \
  --config streaming_couping/configs/v73_correspondence_ablation.yaml \
  --output-dir "$OUTPUT" \
  --device cuda:0 \
  --token-counts 8 \
  --residual-steps 2

[[ -s "$OUTPUT/v73_correspondence_ablation.csv" ]] || {
  echo "V7.3 smoke did not produce its CSV" >&2
  exit 1
}
echo "V7.3 smoke passed"
