#!/usr/bin/env zsh

# What does this machine actually have?  (CPU only, seconds, read-only)
#
#   zsh streaming_couping/commands_report_environment.txt
#
# Interpreter names, checkpoint paths and dataset locations are defaults inside
# the scripts.  Whether they exist, which versions are installed and which GPU
# is which are facts about the machine, and they belong in the handover
# document, so this prints them in a form that can be pasted into it.
#
# Every probe is independent: a missing checkpoint does not stop the rest from
# being reported, because the point is to learn what is missing.
set -uo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON="${STREAMING_COUPING_PYTHON:-$(command -v python)}"

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1

PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.report_environment
