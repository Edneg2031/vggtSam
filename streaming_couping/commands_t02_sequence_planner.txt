#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
MANIFEST=data/processed/scannetpp_pinhole_2d/manifest.json
OUTPUT=outputs/streaming_couping_t02_sequence_planner

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$MANIFEST" ]] || { echo "missing manifest: $MANIFEST" >&2; exit 1; }

rm -rf "$OUTPUT"
echo "T0.2 same-scene unseen-time holdout planning"
echo "Reads manifest metadata/file existence only; no mask, pointmap, model or GPU"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.plan_t02_confirmation_sequence \
  --manifest "$MANIFEST" \
  --discovery-scene-id 00a231a370 \
  --discovery-last-frame 525 \
  --frame-count 30 \
  --frame-stride 2 \
  --maximum-candidates 1 \
  --output-dir "$OUTPUT"

[[ -s "$OUTPUT/summary.json" ]] || { echo "missing T0.2 planner summary" >&2; exit 1; }
[[ -s "$OUTPUT/copyable_result.txt" ]] || { echo "missing T0.2 planner report" >&2; exit 1; }
PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path

summary = json.loads(Path(
    "outputs/streaming_couping_t02_sequence_planner/summary.json"
).read_text())
assert summary["schema"] == 1
assert summary["revision"] == "t02_manifest_only_temporal_holdout_planner_r3"
assert summary["role"] == "t02_same_scene_temporal_holdout_planning_only"
assert summary["validation_scope"] == "same_scene_unseen_temporal_holdout"
assert summary["same_scene_required"] == 1
assert summary["discovery_last_frame"] == 525
assert summary["gt_geometry_values_read"] == 0
assert summary["pointmap_file_content_read"] == 0
assert summary["instance_mask_file_content_read"] == 0
assert summary["holdout_selection_gt_fields"] == 0
assert summary["model_loaded_or_run"] == 0
assert summary["planner_ready"] in {0, 1}
if summary["planner_ready"]:
    assert summary["reported_candidate_count"] == 1
    assert summary["recommended_scene_id"] == "00a231a370"
    assert summary["recommended_clip_name"] == "00a231a370_533_591_step2_t02_holdout"
    assert summary["recommended_frame_indices"] == list(range(533, 592, 2))
eligible = summary["eligible_holdout_count"]
recommended = summary["recommended_scene_id"]
ready = summary["planner_ready"]
print(
    "T0.2 sequence planner accepted "
    f"ready={ready} "
    f"eligible={eligible} "
    f"recommended={recommended}"
)
'

echo ""
sed -n '1,180p' "$OUTPUT/copyable_result.txt"
