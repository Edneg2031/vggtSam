#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG=streaming_couping/configs/v0_baseline.yaml
OUTPUT=outputs/streaming_couping_v0/semantic_map_ab

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s outputs/streaming_couping_v0/baseline_summary.json ]] || {
  echo "missing V0 baseline summary; run commands_v0_baseline.txt first" >&2
  exit 1
}
[[ -s outputs/streaming_couping_v0/poses.pt ]] || {
  echo "missing V0 selected poses; run commands_v0_baseline.txt first" >&2
  exit 1
}

export PYTHONHASHSEED=0
export OMP_NUM_THREADS="${SEMANTIC_MAP_CPU_THREADS:-8}"
export MKL_NUM_THREADS="$OMP_NUM_THREADS"

echo "V0 semantic map A/B: raw pose versus selected QK pose"
echo "Shared inputs: raw depth/K, SAM masks/IDs, RGB and confidence support"
echo "No model is loaded or trained; candidate pointmap is not used"
echo "GT enters only after both native-coordinate maps are generated"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v0_semantic_map_ab

rm -rf "$OUTPUT"
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v0_semantic_map_ab \
  --config "$CONFIG"

[[ -s "$OUTPUT/semantic_map_summary.json" ]] || {
  echo "missing semantic_map_summary.json" >&2
  exit 1
}
[[ -s "$OUTPUT/semantic_maps.pt" ]] || {
  echo "missing semantic_maps.pt" >&2
  exit 1
}
[[ -s "$OUTPUT/raw_semantic_map.ply" ]] || {
  echo "missing raw_semantic_map.ply" >&2
  exit 1
}
[[ -s "$OUTPUT/selected_semantic_map.ply" ]] || {
  echo "missing selected_semantic_map.ply" >&2
  exit 1
}

PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path
summary = json.loads(Path("outputs/streaming_couping_v0/semantic_map_ab/semantic_map_summary.json").read_text())
assert summary["schema"] == 1
assert summary["revision"] == "v0_semantic_map_shared_raw_geometry_ab_r2"
assert summary["selected_pose_branch"] == "retrieve_qk"
assert summary["map_generation_gt_fields"] == 0
assert summary["shared_support_exact"] == 1
assert summary["raw_depth_shared"] == 1
assert summary["raw_intrinsics_shared"] == 1
assert summary["sam_masks_shared"] == 1
assert summary["candidate_pointmap_used"] == 0
assert summary["saved_map_points"] > 0
geometry_pass = summary["geometry_map_pass"]
print(
    "V0 semantic map schema passed "
    f"geometry_pass={geometry_pass}"
)
'

echo ""
echo "===== V0 SEMANTIC MAP A/B COPYABLE RESULT ====="
sed -n '1,160p' "$OUTPUT/copyable_result.txt"
