#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${MULTISCENE_TRAIN_GPU:-1}"
CONFIG="${MULTISCENE_TRAIN_CONFIG:-streaming_couping/configs/multiscene_geometry_train.yaml}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

echo "Multi-scene geometry-only residual training"
echo "physical GPU=$GPU logical device=cuda:0"
echo "StreamVGGT/backbone/PointHead frozen; SAM=0"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT:$ROOT/src" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_multiscene_geometry_training \
  --config "$CONFIG" \
  "$@"
