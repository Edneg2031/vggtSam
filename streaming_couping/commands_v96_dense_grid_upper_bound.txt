#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG=streaming_couping/configs/v96_dense_grid_upper_bound.yaml
DATA_CONFIG=streaming_couping/configs/v92_support_data.yaml
OUTPUT="${V96_OUTPUT:-outputs/streaming_couping_v96_dense_grid_upper_bound}"
CACHE=outputs/streaming_couping_v92_support_factorization/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8

echo "V9.6 smoke: actual grid, bilinear K4, region controls and schemas"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.smoke_v96_dense_grid_upper_bound

if [[ ! -s "$CACHE" ]]; then
  echo "V9.6 requires the completed V9.2 256-token cache: $CACHE" >&2
  echo "Run: zsh streaming_couping/commands_v92_support_factorization.txt" >&2
  exit 1
fi

echo "V9.6 validating the retained dynamic-instance cache"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

echo "V9.6 CPU-only actual-grid coordinate upper bound"
echo "No dense descriptor is cached, evaluated or trained"
CUDA_VISIBLE_DEVICES="" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v96_dense_grid_upper_bound \
  --config "$CONFIG" \
  --output-dir "$OUTPUT"

SUMMARY="$OUTPUT/v96_dense_grid_summary.csv"
FRAMES="$OUTPUT/v96_dense_grid_frames.csv"
DECISION="$OUTPUT/v96_dense_grid_decision.md"
METADATA="$OUTPUT/v96_dense_grid_metadata.json"
for file in "$SUMMARY" "$FRAMES" "$DECISION" "$METADATA"; do
  [[ -s "$file" ]] || { echo "missing V9.6 output: $file" >&2; exit 1; }
done

echo "V9.6 COPY THIS DECISION"
cat "$DECISION"
echo "V9.6 compact 36-row CSV: $SUMMARY"
echo "V9.6 detailed frame rows: $FRAMES"
echo "V9.6 provenance: $METADATA"
