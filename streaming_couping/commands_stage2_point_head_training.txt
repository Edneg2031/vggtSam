#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPUS="${STAGE2_GPUS:-1,2}"
CONFIG="${STAGE2_CONFIG:-streaming_couping/configs/stage2_point_head_training.yaml}"
OUTPUT="${STAGE2_OUTPUT:-/data184/open_source/vggtSam/outputs/streaming_couping_stage2_point_head}"
LOG="$OUTPUT/run.log"
if [[ -n "${STAGE2_R3_SUMMARY:-}" ]]; then
  R3_SUMMARY="$STAGE2_R3_SUMMARY"
else
  R3_SUMMARY="$STORAGE_ROOT/outputs/streaming_couping_phase1_cross_scene/summary.json"
fi

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
if [[ "${STAGE2_REQUIRE_R3_GO:-1}" == "1" ]]; then
  [[ -s "$R3_SUMMARY" ]] || {
    echo "missing frozen multi-scene R3 summary: $R3_SUMMARY" >&2
    echo "Run commands_r3_cross_scene_residual_two_scenes.txt first." >&2
    exit 1
  }
  R3_SUMMARY="$R3_SUMMARY" "$PYTHON_BIN" - <<'PY'
import json
import os
from pathlib import Path

summary = json.loads(Path(os.environ["R3_SUMMARY"]).read_text(encoding="utf8"))
if summary.get("all_scenes_go") != 1:
    raise SystemExit(
        "Stage 2 requires all frozen R3 held-out scenes to pass before training"
    )
if summary.get("all_scenes_strict_majority_frames_improved") != 1:
    raise SystemExit(
        "Stage 2 requires a strict majority of improved frames in every R3 scene"
    )
print("Frozen multi-scene R3 prerequisite accepted")
PY
fi
mkdir -p "$OUTPUT"

echo "Stage 2 trainable original PointHead baseline"
echo "physical GPUs=$GPUS logical devices=cuda:0,cuda:1"
echo "backbone/aggregator/camera frozen; SAM=0; QK unchanged"
echo "output=$OUTPUT"

set +e
CUDA_VISIBLE_DEVICES="$GPUS" PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT:$ROOT/src" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_stage2_point_head_training \
  --config "$CONFIG" \
  > "$LOG" 2>&1
STATUS=$?
set -e

if (( STATUS != 0 )); then
  sed -n '1,240p' "$LOG"
  exit "$STATUS"
fi

for artifact in summary.json point_head_only.pt training_curve.csv heldout_frame_metrics.csv copyable_result.txt; do
  [[ -s "$OUTPUT/$artifact" ]] || {
    echo "missing Stage 2 artifact: $OUTPUT/$artifact" >&2
    exit 1
  }
done

export OUTPUT
PYTHONPATH="$ROOT:$ROOT/src" "$PYTHON_BIN" - <<'PY' >> "$LOG" 2>&1
import json
import os
from pathlib import Path

output = Path(os.environ["OUTPUT"])
summary = json.loads((output / "summary.json").read_text(encoding="utf8"))
assert summary["revision"] == "stage2_trainable_point_head_geometry_only_r2"
assert summary["sam_inputs"] == 0
assert summary["qk_pose_modified"] == 0
assert summary["backbone_parameters_updated"] == 0
assert summary["camera_head_parameters_updated"] == 0
assert summary["point_head_parameters_updated"] == 1
assert summary["formal_v0_modified"] == 0
assert len(summary["heldout_scenes"]) >= 2
assert summary["decision"]["stage2_point_head_decision"] in {"GO", "NO_GO"}
assert summary["decision"]["minimum_improved_frame_ratio"] == 0.5
print("Stage 2 PointHead schema accepted")
PY

sed -n '1,260p' "$LOG"
echo ""
sed -n '1,180p' "$OUTPUT/copyable_result.txt"
