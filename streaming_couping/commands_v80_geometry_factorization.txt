#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${V80_O26_STREAM_GPU0:-1}"
STREAM_GPU1="${V80_O26_STREAM_GPU1:-2}"
SAM_GPU="${V80_O26_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v80_geometry_factorization.yaml
DATA_CONFIG=streaming_couping/configs/v74_temporal_data.yaml
OUTPUT="${V80_O26_OUTPUT:-outputs/streaming_couping_v80_geometry_factorization}"
CACHE=outputs/streaming_couping_v74_temporal_scaling/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

NEED_REBUILD=0
if [[ "${V80_O26_REBUILD_CACHE:-0}" == "1" || ! -s "$CACHE" ]]; then
  NEED_REBUILD=1
else
  echo "V8 O2.6 validating existing cache: $CACHE"
  if ! PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.audit_v7_cache \
    --config "$DATA_CONFIG" \
    --output-dir "$OUTPUT/cache_audit_preflight" \
    --require-local \
    --strict; then
    echo "V8 O2.6 existing cache is stale/incomplete; rebuilding it"
    NEED_REBUILD=1
  fi
fi

if (( NEED_REBUILD )); then
  echo "V8 O2.6 rebuilding only the 30-frame observation cache"
  echo "physical GPUs=$VISIBLE_GPUS"
  echo "logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
  PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.run_instance_token_pose \
    --config "$DATA_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0 \
    --training-device cuda:0 \
    --rebuild-cache
else
  echo "V8 O2.6 reusing cache; both backbones stay unloaded"
fi

[[ -s "$CACHE" ]] || {
  echo "V8 O2.6 cache was not created: $CACHE" >&2
  exit 1
}

PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

echo "V8 O2.6 smoke: Umeyama, depth calibration and backprojection"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.smoke_v80_geometry_factorization

echo "V8 O2.6: fixed-support depth/intrinsics/scale factorization"
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v80_geometry_factorization \
  --config "$CONFIG" \
  --output-dir "$OUTPUT"

PRIMARY="$OUTPUT/v80_o26_decision.csv"
SUMMARY="$OUTPUT/v80_o26_geometry_factorization.csv"
MEDIUM="$OUTPUT/v80_o26_medium_diagnosis.csv"
FRAMES="$OUTPUT/v80_o26_frame_diagnostics.csv"
CALIBRATION="$OUTPUT/v80_o26_depth_intrinsics_diagnostics.csv"
DECISION="$OUTPUT/v80_o26_decision.md"
[[ -s "$PRIMARY" ]] || { echo "missing primary CSV: $PRIMARY" >&2; exit 1; }
[[ -s "$SUMMARY" ]] || { echo "missing fold summary: $SUMMARY" >&2; exit 1; }
[[ -s "$MEDIUM" ]] || { echo "missing medium diagnosis: $MEDIUM" >&2; exit 1; }
[[ -s "$FRAMES" ]] || { echo "missing frame CSV: $FRAMES" >&2; exit 1; }
[[ -s "$CALIBRATION" ]] || { echo "missing calibration CSV: $CALIBRATION" >&2; exit 1; }
[[ -s "$DECISION" ]] || { echo "missing decision: $DECISION" >&2; exit 1; }

echo "V8 O2.6 compact decision CSV validated: $PRIMARY"
echo "V8 O2.6 full fold summary: $SUMMARY"
echo "V8 O2.6 medium four-frame diagnosis: $MEDIUM"
echo "V8 O2.6 frame diagnostics: $FRAMES"
echo "V8 O2.6 depth/intrinsics diagnostics: $CALIBRATION"
echo "V8 O2.6 decision: $DECISION"
