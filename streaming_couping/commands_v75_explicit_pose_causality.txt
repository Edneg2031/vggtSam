#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V75_GPU:-1}"
STREAM_GPU0="${V75_STREAM_GPU0:-1}"
STREAM_GPU1="${V75_STREAM_GPU1:-2}"
SAM_GPU="${V75_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
ICP_DEVICE="${V75_ICP_DEVICE:-cuda:0}"
OUTPUT="${V75_OUTPUT:-outputs/streaming_couping_v75_explicit_pose_causality}"
DATA_CONFIG=streaming_couping/configs/v74_temporal_data.yaml
EXPERIMENT_CONFIG=streaming_couping/configs/v75_explicit_pose_causality.yaml
CACHE=outputs/streaming_couping_v74_temporal_scaling/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V7.5 1/4: dependency-free protocol smoke"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v75_protocol

NEED_REBUILD=0
if [[ "${V75_REBUILD_CACHE:-0}" == "1" || ! -s "$CACHE" ]]; then
  NEED_REBUILD=1
else
  echo "V7.5 2/4: validate existing dynamic-instance cache=$CACHE"
  if ! PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.audit_v7_cache \
    --config "$DATA_CONFIG" \
    --output-dir "$OUTPUT/cache_audit_preflight" \
    --strict; then
    echo "V7.5 existing cache is stale/incomplete; rebuilding it"
    NEED_REBUILD=1
  fi
fi

if (( NEED_REBUILD )); then
  echo "V7.5 2/4: build V7.4 dynamic-instance cache on physical GPUs=$VISIBLE_GPUS"
  echo "logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
  PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.run_instance_token_pose \
    --config "$DATA_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0 \
    --training-device cuda:0 \
    --rebuild-cache
else
  echo "V7.5 2/4: reuse existing dynamic-instance cache=$CACHE"
fi

[[ -s "$CACHE" ]] || {
  echo "V7.5 cache is missing after preflight: $CACHE" >&2
  exit 1
}

echo "V7.5 3/4: tensor smoke and strict cache audit"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v75_geometry
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --strict

echo "V7.5 4/4: no-training explicit pose causality on physical GPU=$GPU"
echo "backbones remain unloaded; GPU is used only for bounded ICP cdist"
CUDA_VISIBLE_DEVICES="$GPU" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v75_explicit_pose_causality \
  --config "$EXPERIMENT_CONFIG" \
  --output-dir "$OUTPUT" \
  --compute-device "$ICP_DEVICE"

SUMMARY="$OUTPUT/v75_explicit_pose_causality.csv"
FRAMES="$OUTPUT/v75_explicit_pose_frames.csv"
DECISION="$OUTPUT/v75_decision_summary.md"
[[ -s "$SUMMARY" ]] || { echo "missing V7.5 summary CSV" >&2; exit 1; }
[[ -s "$FRAMES" ]] || { echo "missing V7.5 frame CSV" >&2; exit 1; }
[[ -s "$DECISION" ]] || { echo "missing V7.5 decision summary" >&2; exit 1; }

echo "V7.5 COPY THIS CSV"
cat "$SUMMARY"
echo "V7.5 COPY THIS DECISION SUMMARY"
cat "$DECISION"
echo "V7.5 per-frame diagnostics=$FRAMES"
