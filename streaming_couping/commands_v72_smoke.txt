#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V72_TRAIN_GPU:-1}"
OUTPUT=outputs/streaming_couping_v72_smoke

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

echo "V7.2 smoke: unit tests"
PYTHONPATH=. "$PYTHON_BIN" -m pytest -q \
  streaming_couping/tests/test_v72_local_fusion.py \
  streaming_couping/tests/test_v71_causal_fusion.py

echo "V7.2 smoke: exact V7.1 L0 plus two residual steps, K=8/32"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v72_local_token_ablation \
  --config streaming_couping/configs/v72_local_token_ablation.yaml \
  --output-dir "$OUTPUT" \
  --device cuda:0 \
  --residual-steps 2 \
  --architectures sam_local_match,dual_key_global_geometry \
  --token-counts 8,32

[[ -s "$OUTPUT/v72_local_token_ablation.csv" ]] || {
  echo "V7.2 smoke did not produce its CSV" >&2
  exit 1
}
echo "V7.2 smoke passed"
