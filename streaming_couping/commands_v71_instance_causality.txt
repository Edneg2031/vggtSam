#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${V71_STREAM_GPU0:-${V7_STREAM_GPU0:-1}}"
STREAM_GPU1="${V71_STREAM_GPU1:-${V7_STREAM_GPU1:-2}}"
SAM_GPU="${V71_SAM_GPU:-${V7_SAM_GPU:-3}}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
DATA_CONFIG=streaming_couping/configs/v7_fusion_data.yaml
CONFIG=streaming_couping/configs/v71_instance_causality.yaml
OUTPUT="${V71_OUTPUT_DIR:-outputs/streaming_couping_v71_instance_causality}"
SEED="${V71_SEED:-0}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}

echo "V7.1 physical GPUs=$VISIBLE_GPUS"
echo "V7.1 logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
"$PYTHON_BIN" -c \
  'import sys, torch; print(f"python={sys.executable} torch={torch.__version__}")'

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

REBUILD_ARGS=()
if [[ "${V71_REBUILD_CACHE:-0}" == "1" ]]; then
  REBUILD_ARGS=(--rebuild-cache)
fi
RESUME_ARGS=(--resume)
if [[ "${V71_FRESH:-0}" == "1" ]]; then
  RESUME_ARGS=()
  echo "V7.1 fresh training requested; checkpoints will be overwritten"
fi

# Usually a cheap validation no-op after V7. It also makes V7.1 reproducible
# when invoked directly on a clean checkout.
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_instance_token_pose \
  --config "$DATA_CONFIG" \
  --stage cache \
  --streamvggt-devices cuda:0,cuda:1 \
  --sam3-device cuda:2 \
  --geometry-device cuda:0 \
  --training-device cuda:0 \
  "${REBUILD_ARGS[@]}"

CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v71_instance_causality \
  --config "$CONFIG" \
  --device cuda:0 \
  --seed "$SEED" \
  --output-dir "$OUTPUT" \
  "${RESUME_ARGS[@]}"

PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.analyze_v71_results \
  --csv "$OUTPUT/v71_instance_causality.csv" \
  --output "$OUTPUT/v71_result_summary.md"

for required in \
  "$OUTPUT/v71_instance_causality.csv" \
  "$OUTPUT/v71_frame_diagnostics.csv" \
  "$OUTPUT/run_metadata.json" \
  "$OUTPUT/v71_result_summary.md"
do
  [[ -s "$required" ]] || {
    echo "missing V7.1 artifact: $required" >&2
    exit 1
  }
done

echo "V7.1 COPY THIS ONE CSV"
cat "$OUTPUT/v71_instance_causality.csv"
echo "V7.1 automatic interpretation=$OUTPUT/v71_result_summary.md"
