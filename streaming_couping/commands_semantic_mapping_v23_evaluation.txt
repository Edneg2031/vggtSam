#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="${SEMANTIC_MAPPING_CONFIG:-streaming_couping/configs/v0_baseline.yaml}"
OUTPUT_DIR="${SEMANTIC_MAPPING_V23_EVAL_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_semantic_map_v23}"
MAP_DIR="${SEMANTIC_MAPPING_V23_MAP_DIR:-$STORAGE_ROOT/outputs/streaming_couping_v23/semantic_map}"
CLIP_ARGS=()
[[ -n "${SEMANTIC_MAPPING_CLIP:-}" ]] && CLIP_ARGS+=(--clip "$SEMANTIC_MAPPING_CLIP")

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$CONFIG" ]] || { echo "missing config: $CONFIG" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

echo "Stage V2.3 confidence-aware voxel-memory offline evaluation"
echo "models are not rerun; GT is opened only inside evaluation"
echo "map_dir=$MAP_DIR"
PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_v23_semantic_map_evaluation \
  --config "$CONFIG" \
  --output-dir "$OUTPUT_DIR" \
  --map-dir "$MAP_DIR" \
  "${CLIP_ARGS[@]}" \
  "$@"

[[ -s "$OUTPUT_DIR/summary.json" ]] || {
  echo "missing V2.3 evaluation summary" >&2
  exit 1
}
[[ -s "$OUTPUT_DIR/semantic_map_v23_summary.json" ]] || {
  echo "missing named V2.3 evaluation summary" >&2
  exit 1
}
echo "===== V2.3 EVALUATION COPYABLE RESULT ====="
sed -n '1,240p' "$OUTPUT_DIR/copyable_result.txt"
