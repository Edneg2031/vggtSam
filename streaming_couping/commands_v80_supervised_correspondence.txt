#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V80_GPU:-1}"
STREAM_GPU0="${V80_STREAM_GPU0:-$GPU}"
STREAM_GPU1="${V80_STREAM_GPU1:-2}"
SAM_GPU="${V80_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v80_supervised_correspondence.yaml
DATA_CONFIG=streaming_couping/configs/v74_temporal_data.yaml
OUTPUT="${V80_OUTPUT:-outputs/streaming_couping_v80_supervised_correspondence}"
LONG_CACHE=outputs/streaming_couping_v74_temporal_scaling/cache/00a231a370_90_525_step15_37_68_54.pt
MATCH_STEPS="${V80_MATCH_STEPS:-1200}"
POSE_STEPS="${V80_POSE_STEPS:-1200}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V8 smoke: shape, GT-label, gradient and fallback checks"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v80_supervised_correspondence

REBUILD_ARGS=()
if [[ "${V80_REBUILD_CACHE:-0}" == "1" ]]; then
  REBUILD_ARGS=(--rebuild-cache)
fi
if [[ ! -s "$LONG_CACHE" || "${V80_REBUILD_CACHE:-0}" == "1" ]]; then
  echo "V8 building dynamic SAM3.1/StreamVGGT cache on physical GPUs=$VISIBLE_GPUS"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
  PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.run_instance_token_pose \
    --config "$DATA_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0 \
    --training-device cuda:0 \
    "${REBUILD_ARGS[@]}"
else
  echo "V8 reusing frozen observation cache: $LONG_CACHE"
fi

[[ -s "$LONG_CACHE" ]] || { echo "missing V8 cache: $LONG_CACHE" >&2; exit 1; }
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

RESUME_ARGS=(--resume)
if [[ "${V80_FRESH:-0}" == "1" ]]; then
  RESUME_ARGS=()
fi

echo "V8 matcher-first training GPU=$GPU match_steps=$MATCH_STEPS pose_steps=$POSE_STEPS"
echo "V8 uses GT only for match supervision; inference remains frozen SAM3.1 + StreamVGGT"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v80_supervised_correspondence \
  --config "$CONFIG" \
  --output-dir "$OUTPUT" \
  --device cuda:0 \
  --match-steps "$MATCH_STEPS" \
  --pose-steps "$POSE_STEPS" \
  "${RESUME_ARGS[@]}"

RESULT="$OUTPUT/v80_supervised_correspondence.csv"
[[ -s "$RESULT" ]] || { echo "missing V8 CSV: $RESULT" >&2; exit 1; }
echo "V8 COPY THIS CSV"
cat "$RESULT"
