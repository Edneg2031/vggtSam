#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"

# Override these with three currently free physical cards, for example:
# V6_STREAM_GPU0=4 V6_STREAM_GPU1=5 V6_SAM_GPU=7 zsh ...
STREAM_GPU0="${V6_STREAM_GPU0:-1}"
STREAM_GPU1="${V6_STREAM_GPU1:-2}"
SAM_GPU="${V6_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"

BASE_CONFIG=streaming_couping/configs/v6_sam31_long30_base.yaml
V6_CONFIG=streaming_couping/configs/v6_sam31_long30_camera.yaml
OUTPUT=outputs/streaming_couping_v6_sam31_long30/camera

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}

echo "physical GPUs=$VISIBLE_GPUS"
echo "logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
"$PYTHON_BIN" -c \
  'import sys, torch; print(f"python={sys.executable} torch={torch.__version__}")'

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

# Preflight validates frame 525 and every RGB path before loading either model.
# The cache is independent from the retained short-sequence V6 cache.
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_instance_token_pose \
  --config "$BASE_CONFIG" \
  --stage cache \
  --streamvggt-devices cuda:0,cuda:1 \
  --sam3-device cuda:2 \
  --geometry-device cuda:0 \
  --training-device cuda:0

CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v6_camera_overfit \
  --config "$V6_CONFIG" \
  --device cuda:0

for required in \
  "$OUTPUT/v6_cross_clip_summary.csv" \
  "$OUTPUT/v6_frame_diagnostics.csv" \
  "$OUTPUT/v6_validation_summary.csv"
do
  [[ -s "$required" ]] || {
    echo "missing result: $required" >&2
    exit 1
  }
done

echo "V6 LONG30 SHORT SUMMARY"
cat "$OUTPUT/v6_cross_clip_summary.csv"
echo "per-frame diagnostics=$OUTPUT/v6_frame_diagnostics.csv"
