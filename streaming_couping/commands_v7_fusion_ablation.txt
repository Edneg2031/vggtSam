#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${V7_STREAM_GPU0:-${V7_GPU:-1}}"
STREAM_GPU1="${V7_STREAM_GPU1:-2}"
SAM_GPU="${V7_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
DATA_CONFIG=streaming_couping/configs/v7_fusion_data.yaml
ABLATION_CONFIG=streaming_couping/configs/v7_fusion_ablation.yaml
OUTPUT=outputs/streaming_couping_v7_fusion_ablation

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}

echo "V7 physical GPUs=$VISIBLE_GPUS"
echo "V7 logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
"$PYTHON_BIN" -c \
  'import sys, torch; print(f"python={sys.executable} torch={torch.__version__}")'

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

REBUILD_ARGS=()
if [[ "${V7_REBUILD_CACHE:-0}" == "1" ]]; then
  REBUILD_ARGS=(--rebuild-cache)
  echo "V7 cache rebuild explicitly enabled"
fi

# V7 owns all four SAM3.1/StreamVGGT caches. This is a no-op when every cache
# is complete; it never reads a V6 output directory.
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_instance_token_pose \
  --config "$DATA_CONFIG" \
  --stage cache \
  --streamvggt-devices cuda:0,cuda:1 \
  --sam3-device cuda:2 \
  --geometry-device cuda:0 \
  --training-device cuda:0 \
  "${REBUILD_ARGS[@]}"

echo "V7 cache stage complete; training five fusion architectures"
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v7_fusion_ablation \
  --config "$ABLATION_CONFIG" \
  --device cuda:0

[[ -s "$OUTPUT/v7_ablation.csv" ]] || {
  echo "missing V7 result: $OUTPUT/v7_ablation.csv" >&2
  exit 1
}

echo "V7 COPY THIS ONE CSV"
cat "$OUTPUT/v7_ablation.csv"
