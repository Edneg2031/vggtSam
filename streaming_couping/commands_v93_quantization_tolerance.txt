#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG=streaming_couping/configs/v93_quantization_tolerance.yaml
DATA_CONFIG=streaming_couping/configs/v92_support_data.yaml
OUTPUT="${V93_OUTPUT:-outputs/streaming_couping_v93_quantization_tolerance}"
CACHE=outputs/streaming_couping_v92_support_factorization/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8

echo "V9.3 smoke: continuous/hard/soft-convex/filter/noise checks"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.smoke_v93_quantization_tolerance

if [[ ! -s "$CACHE" ]]; then
  echo "V9.3 requires the completed standalone V9.2 256-token cache: $CACHE" >&2
  echo "Run: zsh streaming_couping/commands_v92_support_factorization.txt" >&2
  exit 1
fi

echo "V9.3 validating the V9.2 dynamic-instance/256-token cache"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

echo "V9.3 CPU-only fixed-edge oracle diagnosis; SAM3.1/StreamVGGT stay unloaded"
echo "No matcher or pose model is trained"
CUDA_VISIBLE_DEVICES="" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v93_quantization_tolerance \
  --config "$CONFIG" \
  --output-dir "$OUTPUT"

SUMMARY="$OUTPUT/v93_quantization_summary.csv"
FRAMES="$OUTPUT/v93_quantization_frames.csv"
DECISION="$OUTPUT/v93_quantization_decision.md"
METADATA="$OUTPUT/v93_quantization_metadata.json"
for file in "$SUMMARY" "$FRAMES" "$DECISION" "$METADATA"; do
  [[ -s "$file" ]] || { echo "missing V9.3 output: $file" >&2; exit 1; }
done

echo "V9.3 COPY THIS DECISION"
cat "$DECISION"
echo "V9.3 COPY THIS COMPACT SUMMARY CSV (42 rows)"
cat "$SUMMARY"
echo "V9.3 detailed fixed-edge/noise rows: $FRAMES"
echo "V9.3 provenance: $METADATA"
