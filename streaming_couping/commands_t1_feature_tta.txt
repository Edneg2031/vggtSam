#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU0="${T1_FEATURE_TTA_GPU0:-1}"
GPU1="${T1_FEATURE_TTA_GPU1:-2}"
SAM_GPU="${T1_FEATURE_TTA_SAM_GPU:-3}"
KEEP_CLEAN="${T1_KEEP_CLEAN_ARTIFACT:-0}"
KEEP_REBUILT_V0_CACHE="${T1_KEEP_REBUILT_V0_CACHE:-0}"
CONFIG=streaming_couping/configs/t1_feature_tta.yaml
V0_CONFIG=streaming_couping/configs/v0_baseline.yaml
V0_CACHE=outputs/streaming_couping_v0/cache/00a231a370_90_525_step15_37_68_54.pt
QK_POSE=outputs/streaming_couping_v0/qk_pose_retrieval/qk_pose_output.pt
QK_SUMMARY=outputs/streaming_couping_v0/qk_pose_retrieval/qk_pose_summary.json
OUTPUT=outputs/streaming_couping_t1_feature_tta

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "T1 smoke: zero-output LoRA, deterministic history dropout and shuffled teacher"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_t1_feature_tta

REBUILT_V0_CACHE=0
if [[ ! -s "$V0_CACHE" ]]; then
  REBUILT_V0_CACHE=1
  echo "Frozen V0 cache is absent; rebuilding only the required cache stage"
  echo "physical GPUs=$GPU0,$GPU1,$SAM_GPU logical StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
  CUDA_VISIBLE_DEVICES="$GPU0,$GPU1,$SAM_GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
    --config "$V0_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0 \
    --rebuild-cache
fi
[[ -s "$V0_CACHE" ]] || { echo "failed to rebuild frozen V0 cache" >&2; exit 1; }

if [[ ! -s "$QK_POSE" || ! -s "$QK_SUMMARY" ]]; then
  rm -rf outputs/streaming_couping_v0/qk_pose_retrieval
  echo "Frozen QK pose artifact is absent; rebuilding the RGB/QK pose branch only"
  CUDA_VISIBLE_DEVICES="$GPU0,$GPU1" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_v0_qk_pose_retrieval \
    --config "$V0_CONFIG" \
    --streamvggt-devices cuda:0,cuda:1
fi
[[ -s "$QK_POSE" ]] || { echo "failed to rebuild frozen V0 QK pose artifact" >&2; exit 1; }

rm -rf "$OUTPUT"
mkdir -p "$OUTPUT"
echo "T1 stage 1/3: export strict clean RGB/full-history-teacher artifact"
echo "V0 cache is read here; raw pointmap and GT fields are not exported"
PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_t1_feature_tta \
  --config "$CONFIG" --stage prepare

[[ -s "$OUTPUT/clean_tta_input.pt" ]] || { echo "missing T1 clean artifact" >&2; exit 1; }
echo "T1 stage 2/3: masked-history LoRA adaptation and full-history candidate replay"
echo "physical GPUs=$GPU0,$GPU1 logical StreamVGGT=cuda:0,cuda:1"
echo "This process can open only the clean artifact; GT/raw pointmap/SAM are absent"
CUDA_VISIBLE_DEVICES="$GPU0,$GPU1" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_t1_feature_tta \
  --config "$CONFIG" --stage adapt \
  --streamvggt-devices cuda:0,cuda:1

[[ -s "$OUTPUT/candidate_pointmaps.pt" ]] || { echo "missing frozen T1 candidates" >&2; exit 1; }
[[ -s "$OUTPUT/lora_states.pt" ]] || { echo "missing frozen T1 LoRA states" >&2; exit 1; }
[[ -s "$OUTPUT/adaptation_summary.json" ]] || { echo "missing T1 adaptation summary" >&2; exit 1; }

echo "T1 stage 3/3: hash frozen candidates, then introduce V0 raw/GT for scoring"
PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_t1_feature_tta \
  --config "$CONFIG" --stage score

[[ -s "$OUTPUT/summary.json" ]] || { echo "missing T1 summary" >&2; exit 1; }
[[ -s "$OUTPUT/copyable_result.txt" ]] || { echo "missing T1 copyable report" >&2; exit 1; }
PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path

path = Path("outputs/streaming_couping_t1_feature_tta/summary.json")
summary = json.loads(path.read_text())
assert summary["schema"] == 1
assert summary["revision"] == "t1_full_history_teacher_masked_history_lora_tta_r1"
assert summary["baseline_version"] == "v0"
assert summary["baseline_status"] == "frozen_unchanged"
assert len(summary["adaptation_frames"]) == 18
assert len(summary["evaluation_frames"]) == 12
assert [row["branch"] for row in summary["branches"]] == [
    "raw_full_history",
    "correct_teacher_lora",
    "shuffled_teacher_lora",
    "zero_update_lora",
]
assert summary["candidate_generation_gt_fields"] == 0
assert summary["candidate_generation_raw_pointmap_fields"] == 0
assert summary["sam_candidate_inputs"] == 0
assert summary["base_model_parameters_updated"] == 0
assert summary["lora_parameters_updated"] == 1
assert summary["camera_head_run"] == 0
assert summary["depth_head_run"] == 0
assert summary["point_head_run"] == 1
decision = summary["decision"]
assert decision["t1_decision"] in {"GO", "NO_GO"}
assert decision["formal_qk_artifact_sha256_unchanged"] == 1
assert decision["formal_v0_pose_modified"] == 0
assert decision["formal_v0_pointmap_modified"] == 0
assert decision["formal_v0_semantic_map_modified"] == 0
correct = summary["branch_lookup"]["correct_teacher_lora"]
zero_pass = decision["zero_update_equivalence_pass"]
rmse_gain = correct["rmse_gain_vs_raw_percent"]
p90_gain = correct["p90_gain_vs_raw_percent"]
improved = correct["improved_frames_vs_raw"]
result = decision["t1_decision"]
print(
    "T1 output schema accepted "
    f"zero={zero_pass} "
    f"rmse_gain={rmse_gain:.4f}% "
    f"p90_gain={p90_gain:.4f}% "
    f"frames={improved}/12 "
    f"decision={result}"
)
'

if [[ "$KEEP_CLEAN" != "1" ]]; then
  rm -f "$OUTPUT/clean_tta_input.pt"
  echo "Removed temporary clean TTA artifact to save storage (set T1_KEEP_CLEAN_ARTIFACT=1 to retain it)"
fi
if [[ "$REBUILT_V0_CACHE" == "1" && "$KEEP_REBUILT_V0_CACHE" != "1" ]]; then
  rm -f "$V0_CACHE"
  echo "Removed the V0 cache rebuilt by this command (set T1_KEEP_REBUILT_V0_CACHE=1 to retain it)"
fi

echo ""
sed -n '1,220p' "$OUTPUT/copyable_result.txt"
