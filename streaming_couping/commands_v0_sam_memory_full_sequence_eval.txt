#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
INPUT=outputs/streaming_couping_v0/sam_memory_retrieval
OUTPUT="$INPUT/full_sequence_eval"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$INPUT/retrieval_summary.json" ]] || {
  echo "missing completed r2 retrieval_summary.json" >&2
  echo "run streaming_couping/commands_v0_sam_memory_retrieval.txt first" >&2
  exit 1
}
[[ -s "$INPUT/frame_metrics.csv" ]] || {
  echo "missing completed r2 frame_metrics.csv" >&2
  exit 1
}

echo "V0 training-free full 30-frame sequence pose evaluation"
echo "No model is loaded or rerun; existing causal r2 candidates are aggregated"
echo "Frame 90 is the gauge reference; pose metrics use the other 29 frames"
rm -rf "$OUTPUT"
PYTHONPATH=. "$PYTHON_BIN" -m \
  streaming_couping.scripts.summarize_v0_sam_memory_full_sequence \
  --input-dir "$INPUT" \
  --output-dir "$OUTPUT" \
  --reference-sequence-index 0

[[ -s "$OUTPUT/full_sequence_summary.json" ]] || {
  echo "missing full_sequence_summary.json" >&2
  exit 1
}
[[ -s "$OUTPUT/copyable_result.txt" ]] || {
  echo "missing full-sequence copyable_result.txt" >&2
  exit 1
}

"$PYTHON_BIN" -c '
import json
from pathlib import Path
path = Path("outputs/streaming_couping_v0/sam_memory_retrieval/full_sequence_eval/full_sequence_summary.json")
summary = json.loads(path.read_text())
assert summary["revision"] == "v0_sam_memory_full_sequence_pose_eval_r1"
assert summary["stream_frames"] == 30
assert summary["evaluated_pose_frames"] == 29
assert summary["model_trained"] == 0
assert summary["candidate_generation_gt_fields"] == 0
print("V0 full-sequence evaluation schema passed")
'

echo ""
echo "===== V0 FULL-SEQUENCE POSE COPYABLE RESULT ====="
sed -n '1,200p' "$OUTPUT/copyable_result.txt"
