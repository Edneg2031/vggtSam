#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PINHOLE_ROOT="${SCANNETPP_PINHOLE_ROOT:-/data184/open_source/scannet_pp_pinhole}"
ANNOTATION_ROOT="${SCANNETPP_ANNOTATION_ROOT:-/data184/open_source/scannet_pp}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
MODE="${SCANNETPP_PREPROCESS_MODE:-preflight}"
CONFIG=configs/scannetpp_pinhole_2d.yaml
PROCESSED_ROOT="$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d"
MANIFEST="$PROCESSED_ROOT/manifest.json"
SCENE_ROOT="$PINHOLE_ROOT/data"
SCENE_LIST="$PINHOLE_ROOT/scannetpp_part_valid_scenes.txt"
ANNOTATION_DATA_ROOT="$ANNOTATION_ROOT/data"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -f "$ROOT/src/vggtsam/data/scannetpp/preprocess.py" ]] || {
  echo "missing ScanNet++ preprocessing package: $ROOT/src/vggtsam/data/scannetpp" >&2
  echo "sync/pull the restored src/vggtsam/data source files before running" >&2
  exit 1
}
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

mkdir -p \
  "$STORAGE_ROOT/data/processed" \
  "$STORAGE_ROOT/outputs"

echo "ScanNet++ NAS preprocessing mode=$MODE"
echo "storage=$STORAGE_ROOT"
echo "pinhole scenes=$SCENE_ROOT/<scene_id> (read-only source)"
echo "annotations=$ANNOTATION_DATA_ROOT/<scene_id>/scans (read-only source)"
echo "processed=$PROCESSED_ROOT"
df -h "$STORAGE_ROOT" | tail -n 1

[[ -d "$SCENE_ROOT" ]] || {
  echo "missing historical pinhole source: $SCENE_ROOT" >&2
  exit 1
}
[[ -d "$ANNOTATION_DATA_ROOT" ]] || {
  echo "missing historical annotation source: $ANNOTATION_DATA_ROOT" >&2
  exit 1
}

if [[ "$MODE" == "preflight" ]]; then
  PYTHONPATH="$ROOT/src" "$PYTHON_BIN" -c '
import numpy
import yaml
print("preflight dependencies passed")
'
else
  PYTHONPATH="$ROOT/src" "$PYTHON_BIN" -c '
import cv2
import numpy
import plyfile
import yaml
print("preprocess base dependencies passed")
'
  if PYTHONPATH="$ROOT/src" "$PYTHON_BIN" -c 'import numba' 2>/dev/null; then
    echo "numba rasterizer available"
  elif [[ "$MODE" == "full" && "${SCANNETPP_ALLOW_SLOW_RASTERIZER:-0}" != "1" ]]; then
    echo "numba is required for full-scene rasterization (795 frames in 00a231a370)" >&2
    echo "install with: $PYTHON_BIN -m pip install numba" >&2
    echo "or explicitly set SCANNETPP_ALLOW_SLOW_RASTERIZER=1 to use the very slow fallback" >&2
    exit 1
  else
    echo "numba=0; using the slow Python rasterizer"
  fi
fi

SCENE_ARGS=()
if [[ -n "${SCANNETPP_SCENES:-}" ]]; then
  SCENE_ARGS+=(--scene-ids ${=SCANNETPP_SCENES})
elif [[ "$MODE" == "preflight" || "$MODE" == "debug" ]]; then
  SCENE_ARGS=(--scene-ids "${SCANNETPP_DEFAULT_SCENE:-00a231a370}")
fi

SOURCE_ARGS=(
  --data-root "$SCENE_ROOT"
  --annotation-root "$ANNOTATION_DATA_ROOT"
  --metadata-root "$PINHOLE_ROOT/metadata"
  --scene-list "$SCENE_LIST"
)

case "$MODE" in
  preflight)
    echo "Read-only asset preflight; no masks or pointmaps are generated"
    PYTHONPATH="$ROOT/src" "$PYTHON_BIN" scripts/prepare_scannetpp_2d.py \
      --config "$CONFIG" \
      "${SOURCE_ARGS[@]}" \
      "${SCENE_ARGS[@]}" \
      --max-frames "${SCANNETPP_DEBUG_FRAMES:-5}" \
      --dry-run
    ;;
  debug)
    if (( ${#SCENE_ARGS[@]} == 0 )); then
      SCENE_ARGS=(--scene-ids 00a231a370)
    fi
    echo "Generate a five-frame visual/debug slice on NAS"
    PYTHONPATH="$ROOT/src" "$PYTHON_BIN" scripts/prepare_scannetpp_2d.py \
      --config "$CONFIG" \
      "${SOURCE_ARGS[@]}" \
      "${SCENE_ARGS[@]}" \
      --max-frames "${SCANNETPP_DEBUG_FRAMES:-5}" \
      --overwrite \
      --save-visualizations
    ;;
  full)
    if (( ${#SCENE_ARGS[@]} == 0 )) && [[ "${SCANNETPP_USE_SCENE_LIST:-0}" != "1" ]]; then
      echo "full mode requires SCANNETPP_SCENES to avoid processing the entire release" >&2
      echo "Set SCANNETPP_USE_SCENE_LIST=1 only when the full scene list is intended" >&2
      exit 2
    fi
    echo "Generate/resume full selected scenes; existing complete frames are kept"
    PYTHONPATH="$ROOT/src" "$PYTHON_BIN" scripts/prepare_scannetpp_2d.py \
      --config "$CONFIG" \
      "${SOURCE_ARGS[@]}" \
      "${SCENE_ARGS[@]}" \
      --skip-existing \
      --no-visualizations
    ;;
  audit)
    [[ -s "$MANIFEST" ]] || { echo "missing manifest: $MANIFEST" >&2; exit 1; }
    ;;
  *)
    echo "SCANNETPP_PREPROCESS_MODE must be preflight, debug, full or audit" >&2
    exit 2
    ;;
esac

if [[ "$MODE" != "preflight" ]]; then
  [[ -s "$MANIFEST" ]] || { echo "missing manifest after preprocessing" >&2; exit 1; }
  PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.audit_processed_dataset \
    --manifest "$MANIFEST" \
    --output-dir "$STORAGE_ROOT/outputs/processed_dataset_audit" \
    --minimum-frames "${SCANNETPP_MINIMUM_FRAMES:-30}"
fi
