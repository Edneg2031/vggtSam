#!/usr/bin/env zsh
set -euo pipefail

# 100-frame V2.1 continuity-restored external SAM/HorizonStream pose-correction loop.
# HorizonStream is run once to produce the geometry cache.  The semantic-map
# stage then processes frames causally with a corrected external pose state and
# a short sliding-window object constraint optimizer with a deterministic
# correspondence holdout.  HorizonStream's latent/cache state is deliberately
# not mutated in this experiment.  The previous V2 output is preserved.

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
HORIZON_PYTHON="${HORIZONSTREAM_PYTHON:-/home/huawei/miniconda3/envs/horizonstream/bin/python}"
SAM_PYTHON="${STREAMING_COUPING_PYTHON:-$HORIZON_PYTHON}"
HORIZON_REPO="$ROOT/externals/horizonstream"
HORIZON_CHECKPOINT="${HORIZONSTREAM_CHECKPOINT:-/home/bod/86Nas/95_data_bak/FoundationModels/HorizonStream.pt}"
HORIZON_DEVICE="${HORIZONSTREAM_DEVICE:-cuda:0}"
SAM_DEVICE="${SAM_DEVICE:-cuda:2}"

MANIFEST_PATH="$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d/manifest.json"
SCENE_ID="00a231a370"
FRAME_START=90
FRAME_STRIDE=1
FRAME_COUNT=100
PROMPTS=("bed" "wardrobe" "chair" "rug" "dustbin")

BASELINE_OUTPUT_DIR="$STORAGE_ROOT/outputs/semantic_map_100frames_horizonstream_consecutive"
GEOMETRY_CACHE="${HORIZONSTREAM_GEOMETRY_CACHE:-$BASELINE_OUTPUT_DIR/horizonstream_geometry.pt}"
OUTPUT_DIR="${OBJECT_POSE_ONLINE_OUTPUT_DIR:-$STORAGE_ROOT/outputs/semantic_map_100frames_horizonstream_object_pose_online_loop_v2_1}"
EVAL_DIR="$OUTPUT_DIR/gt_evaluation"
RECOVERY_CONFIG="${RECOVERY_CONFIG_PATH:-$ROOT/streaming_couping/configs/recovery_dynamic_instance.yaml}"
GEOMETRY_LOG="$OUTPUT_DIR/horizonstream_geometry.log"
RUN_LOG="$OUTPUT_DIR/run.log"
EVAL_LOG="$EVAL_DIR/evaluator.log"

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1
export PYTORCH_CUDA_ALLOC_CONF="${PYTORCH_CUDA_ALLOC_CONF:-expandable_segments:True}"

[[ -x "$HORIZON_PYTHON" ]] || { print -u2 "missing Python: $HORIZON_PYTHON"; exit 1; }
[[ -x "$SAM_PYTHON" ]] || { print -u2 "missing Python: $SAM_PYTHON"; exit 1; }
[[ -d "$HORIZON_REPO/horizonstream" ]] || { print -u2 "missing HorizonStream source: $HORIZON_REPO"; exit 1; }
[[ -f "$HORIZON_CHECKPOINT" ]] || { print -u2 "missing HorizonStream checkpoint: $HORIZON_CHECKPOINT"; exit 1; }
[[ -f "$MANIFEST_PATH" ]] || { print -u2 "missing ScanNet++ manifest: $MANIFEST_PATH"; exit 1; }
[[ -f "$RECOVERY_CONFIG" ]] || { print -u2 "missing recovery config: $RECOVERY_CONFIG"; exit 1; }

if ! "$SAM_PYTHON" -c 'import iopath, ftfy, regex, huggingface_hub, timm, einops, pycocotools' >/dev/null 2>&1; then
  print -u2 "SAM3 runtime dependencies are missing"
  exit 1
fi

mkdir -p "$OUTPUT_DIR" "$EVAL_DIR"

HORIZON_ARGS=(
  --manifest "$MANIFEST_PATH"
  --scene-id "$SCENE_ID"
  --frame-start "$FRAME_START"
  --frame-stride "$FRAME_STRIDE"
  --frame-count "$FRAME_COUNT"
  --repo-path "$HORIZON_REPO"
  --checkpoint "$HORIZON_CHECKPOINT"
  --output-cache "$GEOMETRY_CACHE"
  --device "$HORIZON_DEVICE"
  --window-size 10
  --sliding-size 1
  --image-size 518
  --patch-size 14
  --crop
  --precision float16
  --reuse-if-valid
)

ARGS=(
  --manifest "$MANIFEST_PATH"
  --scene-id "$SCENE_ID"
  --frame-start "$FRAME_START"
  --frame-stride "$FRAME_STRIDE"
  --frame-count "$FRAME_COUNT"
  --prompts "${PROMPTS[@]}"
  --geometry-cache "$GEOMETRY_CACHE"
  --output-dir "$OUTPUT_DIR"
  --fusion-policy raw
  --recovery-config "$RECOVERY_CONFIG"
  --sam-device "$SAM_DEVICE"
  --sam-grounding-batch-size 1
  --sam-offload-video-to-cpu
  --object-only
  --object-pose-online-loop
  --object-pose-loss-anchor-frames 3
  --object-pose-loss-max-anchor-observations 3
  --object-pose-loss-max-history-observations 3
  --object-pose-loss-max-points-per-observation 256
  --object-pose-loss-min-points-per-observation 24
  --object-pose-loss-min-track-score 0.50
  --object-pose-loss-static-score-threshold 0.20
  --object-pose-loss-min-geometry-confidence 0.30
  --object-pose-loss-min-mask-pixels 32
  --object-pose-loss-max-match-distance-m 0.25
  --object-pose-loss-trim-ratio 0.70
  --object-pose-loss-min-matches-per-pair 8
  --object-pose-loss-device cpu
  --object-pose-online-window-size 5
  --object-pose-online-max-reference-frames 3
  --object-pose-online-max-reference-gap 10
  --object-pose-online-anchor-frames 3
  --object-pose-online-always-include-anchor-reference
  --object-pose-online-carry-rejected-correction
  --object-pose-online-min-independent-instances 2
  --object-pose-online-min-matches-per-instance 15
  --object-pose-online-min-total-matches 40
  --object-pose-online-rematch-iterations 2
  --object-pose-online-optimizer-steps 50
  --object-pose-online-learning-rate 0.01
  --object-pose-online-huber-delta-m 0.05
  --object-pose-online-pose-prior-weight 1.0
  --object-pose-online-temporal-edge-weight 1.0
  --object-pose-online-correction-smoothness-weight 0.25
  --object-pose-online-max-local-correction-rotation-deg 0.3
  --object-pose-online-max-local-correction-translation-m 0.01
  --object-pose-online-min-relative-improvement 0.10
  --object-pose-online-max-validation-residual-m 0.10
  --object-pose-online-validation-fraction 0.25
  --object-pose-online-min-validation-matches-per-instance 2
  --object-pose-online-min-validation-relative-improvement 0.05
  --object-pose-loss-trace
)

print "===== SCANNET EXTERNAL OBJECT POSE ONLINE LOOP 100F ====="
print "scene=$SCENE_ID frames=$FRAME_COUNT frame_start=$FRAME_START frame_stride=$FRAME_STRIDE"
print "horizon_device=$HORIZON_DEVICE sam_device=$SAM_DEVICE loop_device=cpu"
print "output=$OUTPUT_DIR"

PYTHONPATH="$HORIZON_REPO:$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$HORIZON_PYTHON" -m streaming_couping.scripts.generate_horizonstream_geometry_cache \
  "${HORIZON_ARGS[@]}" >"$GEOMETRY_LOG" 2>&1

[[ -f "$GEOMETRY_CACHE" ]] || { print -u2 "missing geometry cache: $GEOMETRY_CACHE"; exit 1; }

PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$SAM_PYTHON" -m streaming_couping.scripts.run_semantic_map \
  "${ARGS[@]}" >"$RUN_LOG" 2>&1

SUMMARY_PATH="$OUTPUT_DIR/object_pose_refinement/pose_refinement_summary.json"
[[ -f "$SUMMARY_PATH" ]] || {
  print -u2 "missing online-loop summary: $SUMMARY_PATH"
  tail -n 80 "$RUN_LOG"
  exit 1
}

set +e
VGGT_SAM_ROOT="$ROOT" \
VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT" \
SEMANTIC_MAP_MANIFEST="$MANIFEST_PATH" \
SEMANTIC_MAP_SCENE_ID="$SCENE_ID" \
HORIZONSTREAM_GEOMETRY_CACHE="$GEOMETRY_CACHE" \
OBJECT_POSE_LOSS_OUTPUT_DIR="$OUTPUT_DIR" \
OBJECT_POSE_LOSS_EVAL_DIR="$EVAL_DIR" \
STREAMING_COUPING_PYTHON="$SAM_PYTHON" \
  zsh "$ROOT/streaming_couping/commands_evaluate_scannet_object_pose_loss.txt" \
  >"$EVAL_LOG" 2>&1
eval_exit_code=$?
set -e
if (( eval_exit_code != 0 )); then
  print -u2 "GT evaluation failed status=$eval_exit_code log=$EVAL_LOG"
  tail -n 80 "$EVAL_LOG"
  exit "$eval_exit_code"
fi

EVAL_SUMMARY="$EVAL_DIR/summary.json"
[[ -f "$EVAL_SUMMARY" ]] || { print -u2 "missing evaluation summary: $EVAL_SUMMARY"; exit 1; }

PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$SAM_PYTHON" "$ROOT/streaming_couping/scripts/print_scannet_object_pose_loss_compact.py" \
  "$SUMMARY_PATH" "$EVAL_SUMMARY"
print "run_log=$RUN_LOG"
print "evaluation_log=$EVAL_LOG"
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$SAM_PYTHON" "$ROOT/streaming_couping/scripts/print_object_pose_loss_trace.py" \
  "$SUMMARY_PATH" "$EVAL_SUMMARY"
