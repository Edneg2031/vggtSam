#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
PROTOCOL_CONFIG="${SEMANTIC_MAPPING_MULTICLIP_CONFIG:-$STORAGE_ROOT/outputs/streaming_couping_multiclip/protocol.yaml}"
CONFIG_DIR="${SEMANTIC_MAPPING_MULTICLIP_RUN_ROOT:-$STORAGE_ROOT/outputs/streaming_couping_multiclip}"
STAGE="${SEMANTIC_MAPPING_MULTICLIP_STAGE:-all}"
STREAM_GPU0="${SEMANTIC_MAPPING_STREAM_GPU0:-1}"
STREAM_GPU1="${SEMANTIC_MAPPING_STREAM_GPU1:-2}"
SAM_GPU="${SEMANTIC_MAPPING_SAM_GPU:-3}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$PROTOCOL_CONFIG" ]] || {
  echo "missing protocol config: $PROTOCOL_CONFIG" >&2
  echo "run zsh streaming_couping/commands_plan_semantic_mapping_multiclip.txt first" >&2
  exit 1
}
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"

configs=("$CONFIG_DIR"/*_uniform*.yaml(N))
(( ${#configs} > 0 )) || { echo "no per-clip configs under $CONFIG_DIR" >&2; exit 1; }

run_cache() {
  echo "multi-scene stage=cache configs=$PROTOCOL_CONFIG"
  CUDA_VISIBLE_DEVICES="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU" \
    PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
    --config "$PROTOCOL_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0
}

run_baselines() {
  for config in $configs; do
    echo "multi-scene stage=baseline config=$config"
    CUDA_VISIBLE_DEVICES="$STREAM_GPU0,$STREAM_GPU1" \
      PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
      "$PYTHON_BIN" -m streaming_couping.scripts.run_v0_qk_pose_retrieval \
      --config "$config" \
      --streamvggt-devices cuda:0,cuda:1
    CUDA_VISIBLE_DEVICES="$STREAM_GPU0,$STREAM_GPU1" \
      PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
      "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
      --config "$config" \
      --stage audit \
      --audit-device cuda:0
  done
}

run_v21() {
  for config in $configs; do
    echo "multi-scene stage=v21 config=$config"
    CUDA_VISIBLE_DEVICES="$SAM_GPU" \
      PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
      "$PYTHON_BIN" -m streaming_couping.scripts.run_v21_semantic_map \
      --config "$config" \
      --sam3-device cuda:0
  done
}

run_v22() {
  for config in $configs; do
    echo "multi-scene stage=v22 config=$config"
    CUDA_VISIBLE_DEVICES="$SAM_GPU" \
      PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
      "$PYTHON_BIN" -m streaming_couping.scripts.run_v22_semantic_map \
      --config "$config" \
      --sam3-device cuda:0
  done
}

run_v23() {
  for config in $configs; do
    echo "multi-scene stage=v23 config=$config"
    PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT" \
      "$PYTHON_BIN" -m streaming_couping.scripts.run_v23_semantic_map \
      --config "$config"
  done
}

case "$STAGE" in
  cache) run_cache ;;
  baseline) run_baselines ;;
  v21) run_v21 ;;
  v22) run_v22 ;;
  v23) run_v23 ;;
  all)
    run_cache
    run_baselines
    run_v21
    run_v22
    run_v23
    ;;
  *)
    echo "invalid SEMANTIC_MAPPING_MULTICLIP_STAGE=$STAGE; use cache|baseline|v21|v22|v23|all" >&2
    exit 1
    ;;
esac

echo "multi-scene stage=$STAGE completed"
