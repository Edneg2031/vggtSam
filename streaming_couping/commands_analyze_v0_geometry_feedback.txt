#!/usr/bin/env zsh
set -euo pipefail

# Frozen V0 joint diagnostic.  It does not rerun StreamVGGT or SAM and never
# promotes a Veto or affine correction into the deployed semantic map.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG="streaming_couping/configs/v0_baseline.yaml"
OUTPUT_DIR="${V0_GEOMETRY_FEEDBACK_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_v0_geometry_feedback}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}

export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1
export PYTHONPATH="$ROOT"

"$PYTHON_BIN" -m streaming_couping.scripts.smoke_v0_geometry_feedback

echo "V0 joint historical-depth / scale-shift diagnostic"
echo "models are not rerun; GT is opened only after causal candidates are frozen"
"$PYTHON_BIN" -m streaming_couping.scripts.analyze_v0_geometry_feedback \
  --config "$CONFIG" \
  --output-dir "$OUTPUT_DIR" \
  --pose-source both \
  --confidence-threshold 0.30 \
  --track-score-threshold 0.50 \
  --min-history-points 16 \
  --max-history-points 4096 \
  --max-points-per-history-frame 512 \
  --mad-multiplier 3.0 \
  --absolute-padding-m 0.05 \
  --lower-quantile 0.05 \
  --upper-quantile 0.95 \
  --max-pair-points-per-query 2048 \
  --min-affine-samples 64 \
  --max-affine-samples 200000 \
  --affine-trim-quantile 0.90 \
  --affine-max-iterations 4 \
  --calibration-fraction 0.70

echo "===== COPYABLE V0 JOINT GEOMETRY FEEDBACK RESULT ====="
sed -n '1,240p' "$OUTPUT_DIR/copyable_result.txt"
echo "V0 joint geometry-feedback output=$OUTPUT_DIR"
