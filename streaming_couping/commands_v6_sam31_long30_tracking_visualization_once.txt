#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
CONFIG=streaming_couping/configs/v6_sam31_long30_base.yaml
CLIP=00a231a370_90_525_step15_37_68_54
OUTPUT=outputs/streaming_couping_v6_sam31_long30/tracking_visualization_once

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || {
  echo "missing Python: $PYTHON_BIN" >&2
  exit 1
}

# CPU-only post-processing of the existing feature cache. This command never
# loads SAM3.1, StreamVGGT, or a trained camera model. The normal V6 command
# does not render tracking visualizations.
CUDA_VISIBLE_DEVICES="" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v6_tracking_visualization \
  --config "$CONFIG" \
  --clip "$CLIP" \
  --output-dir "$OUTPUT" \
  --success-iou 0.50

for required in \
  "$OUTPUT/tracking_success_summary.csv" \
  "$OUTPUT/tracking_frames.csv" \
  "$OUTPUT/$CLIP/rgb_gt_v6_tracking/sequence_overview.png"
do
  [[ -s "$required" ]] || {
    echo "missing tracking visualization result: $required" >&2
    exit 1
  }
done

echo "tracking overview=$OUTPUT/$CLIP/rgb_gt_v6_tracking/sequence_overview.png"
echo "per-frame counts=$OUTPUT/tracking_frames.csv"
