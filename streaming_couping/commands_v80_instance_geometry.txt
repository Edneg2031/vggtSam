#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${V80_O27_STREAM_GPU0:-1}"
STREAM_GPU1="${V80_O27_STREAM_GPU1:-2}"
SAM_GPU="${V80_O27_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v80_instance_geometry.yaml
DATA_CONFIG=streaming_couping/configs/v74_temporal_data.yaml
OUTPUT="${V80_O27_OUTPUT:-outputs/streaming_couping_v80_instance_geometry}"
CACHE=outputs/streaming_couping_v74_temporal_scaling/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

NEED_REBUILD=0
if [[ "${V80_O27_REBUILD_CACHE:-0}" == "1" || ! -s "$CACHE" ]]; then
  NEED_REBUILD=1
else
  echo "V8 O2.7 validating existing cache: $CACHE"
  if ! PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.audit_v7_cache \
    --config "$DATA_CONFIG" \
    --output-dir "$OUTPUT/cache_audit_preflight" \
    --require-local \
    --strict; then
    echo "V8 O2.7 existing cache is stale/incomplete; rebuilding it"
    NEED_REBUILD=1
  fi
fi

if (( NEED_REBUILD )); then
  echo "V8 O2.7 rebuilding only the 30-frame observation cache"
  echo "physical GPUs=$VISIBLE_GPUS"
  echo "logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
  PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.run_instance_token_pose \
    --config "$DATA_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0 \
    --training-device cuda:0 \
    --rebuild-cache
else
  echo "V8 O2.7 reusing cache; both backbones stay unloaded"
fi

[[ -s "$CACHE" ]] || {
  echo "V8 O2.7 cache was not created: $CACHE" >&2
  exit 1
}

PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

echo "V8 O2.7 smoke: causal K and instance affine depth"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.smoke_v80_instance_geometry

echo "V8 O2.7: fixed-camera K + SAM-instance affine diagnosis"
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v80_instance_geometry \
  --config "$CONFIG" \
  --output-dir "$OUTPUT"

PRIMARY="$OUTPUT/v80_o27_decision.csv"
MEDIUM="$OUTPUT/v80_o27_medium_diagnosis.csv"
SUMMARY="$OUTPUT/v80_o27_fold_summary.csv"
FRAMES="$OUTPUT/v80_o27_frame_diagnostics.csv"
INSTANCE="$OUTPUT/v80_o27_instance_affine.csv"
DECISION="$OUTPUT/v80_o27_decision.md"
[[ -s "$PRIMARY" ]] || { echo "missing decision CSV: $PRIMARY" >&2; exit 1; }
[[ -s "$MEDIUM" ]] || { echo "missing medium CSV: $MEDIUM" >&2; exit 1; }
[[ -s "$SUMMARY" ]] || { echo "missing fold summary: $SUMMARY" >&2; exit 1; }
[[ -s "$FRAMES" ]] || { echo "missing frame diagnostics: $FRAMES" >&2; exit 1; }
[[ -s "$INSTANCE" ]] || { echo "missing instance affine diagnostics: $INSTANCE" >&2; exit 1; }
[[ -s "$DECISION" ]] || { echo "missing decision markdown: $DECISION" >&2; exit 1; }

echo "V8 O2.7 compact decision: $PRIMARY"
echo "V8 O2.7 medium diagnosis: $MEDIUM"
echo "V8 O2.7 decision: $DECISION"
