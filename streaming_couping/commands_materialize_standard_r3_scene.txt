#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
SCENE_ID="${R3_SCENE_ID:-0a184cf634}"
R3_OUTPUT="${R3_GEOMETRY_OUTPUT:-$STORAGE_ROOT/data/processed/residual_cross_scene_r3}"
STANDARD_ROOT="${SCANNETPP_PROCESSED_ROOT:-$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$R3_OUTPUT/manifest.json" ]] || {
  echo "missing R3 manifest: $R3_OUTPUT/manifest.json" >&2
  echo "Run first: zsh streaming_couping/commands_r3_prepare_cross_scene_geometry.txt" >&2
  exit 1
}

ARGS=()
if [[ "${R3_GEOMETRY_OVERWRITE:-0}" == "1" ]]; then
  ARGS+=(--overwrite)
fi

echo "Materialize the held-out R3 clip in the standard ScanNet++ layout"
echo "scene=$SCENE_ID"
echo "r3_manifest=$R3_OUTPUT/manifest.json"
echo "processed_root=$STANDARD_ROOT"
echo "RGB and pointmaps use symlinks when supported; no model/GPU/SAM is used"

PYTHONPATH="$ROOT:$ROOT/src" "$PYTHON_BIN" \
  -m streaming_couping.scripts.materialize_standard_scannetpp_scene \
  --r3-manifest "$R3_OUTPUT/manifest.json" \
  --processed-root "$STANDARD_ROOT" \
  --scene-id "$SCENE_ID" \
  "${ARGS[@]}"

export STANDARD_ROOT SCENE_ID
PYTHONPATH="$ROOT:$ROOT/src" "$PYTHON_BIN" - <<'PY'
import json
import os
from pathlib import Path

root = Path(os.environ["STANDARD_ROOT"])
scene = root / os.environ["SCENE_ID"]
manifest_path = root / "manifest.json"
scene_manifest = scene / "scene_manifest.json"
assert manifest_path.is_file(), manifest_path
assert scene_manifest.is_file(), scene_manifest
payload = json.loads(scene_manifest.read_text(encoding="utf8"))
frames = payload.get("frames", [])
assert frames, "standardized scene has no frames"
for frame in frames:
    assert Path(frame["image_path"]).is_file(), frame["image_path"]
    assert Path(frame["pointmap"]).is_file(), frame["pointmap"]
print(
    f"standard layout accepted scene={os.environ['SCENE_ID']} "
    f"frames={len(frames)} scene_dir={scene}"
)
PY

echo "done"
echo "scene_dir=$STANDARD_ROOT/$SCENE_ID"
echo "global_manifest=$STANDARD_ROOT/manifest.json"
