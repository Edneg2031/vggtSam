#!/usr/bin/env zsh

# One-shot sweep over the proposal-side branches of the object-consensus pose
# feedback experiment.
#
#   zsh streaming_couping/commands_run_scannet_object_pose_feedback_branches.txt
#
# No arguments and no environment variables are needed; edit the block below to
# change the branch list or the freshness thresholds.
#
# DEFAULT IS ONE BRANCH AND ONE REPEAT.  That is the run that produces the
# reported result: a stage-1 cache hit, one segmentation pass, the oracle
# control, the replay gate and the two comparison tables.  The four ablation
# branches and the two extra repeats are opt-in -- see BRANCHES and SAM_REPEATS
# below for what each costs and what it buys.
#
# GENERATIONS
#
# This sweep writes a new generation of run directories (_v2) rather than
# overwriting the previous one (_v1).  v1 is the five-prompt run that passed
# every criterion, and it is kept intact: nothing here writes to it.  The one
# thing read from it is the stage-1 geometry cache, and that is copied, not
# moved or linked, so v1 cannot be damaged even if stage 1 rebuilds.
#
# The prompt set is what changed between the two, so the sweep ends with a
# generation-against-generation table -- v1/baseline next to v2/baseline,
# v1/fresh next to v2/fresh -- which is the reading this round is for.  It also
# prints the prompt reachability inventory before spending any GPU time, since
# the added prompts only help if they name the objects the way the manifest
# does.
#
# BRANCHES
#
#   baseline          joint 6DoF, permanent anchors   (the previous behavior)
#   translation_only  joint budget spent on translation only
#   fresh             joint 6DoF, bounded reference age
#   factorized        rotation-then-translation, permanent anchors
#   fresh_factor      both
#
# HOW THIS DIFFERS FROM RUNNING THE FOUR PIPELINES IN SEQUENCE
#
# The refiner knobs live inside stage 2a, next to the segmentation model, so a
# naive sweep runs the segmentation model once per branch.  Replaying from the
# baseline's cached observations costs no GPU and makes the branch difference
# purely the knob under test.
#
# (Three identical baseline runs measured a spread of 2e-05 on the direct-ATE
# gain and 0 on the proposal count, so the pipeline itself turned out to be
# reproducible; the replay is kept because it is free, not because the
# segmentation model is noisy.  An earlier claim here that the run-to-run
# spread was 0.026 came from comparing runs built from different code
# revisions, not from repeating one.)
#
# The replay is gated: before any branch is compared, a replay with the
# baseline's own settings must reproduce the baseline proposals.  If that
# assertion fails the sweep aborts, because every later comparison would rest
# on a replay that is not faithful.
#
# Stage 3 (the object-map evaluation for the per-instance branch) is only run
# for the baseline: it needs the per-instance maps that only a full stage 2a
# exports, and the branch comparison does not read its output.
set -uo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON="${STREAMING_COUPING_PYTHON:-$(command -v python)}"

# ---- Generations ----------------------------------------------------------
# The prompt set is the independent variable, so each one gets its own
# generation of run directories and nothing here writes to an earlier one.
# v1 is the five-prompt run that passed every criterion; it is the result being
# kept, so it is never written to.  The only thing read from a previous
# generation is the stage-1 geometry cache -- and that is COPIED, never moved
# or linked, so the earlier generation cannot be damaged even if stage 1
# decides the cache needs rebuilding.
#
# To try another prompt set, change GENERATION below.  The prompt list is looked
# up from the generation, so the tag and the words cannot disagree; nothing else
# needs editing, because the run directories, the cache seed, the comparison
# table and the oracle's ground-truth set all derive from it.
# The frame window and the run-directory name come from the shared env file.
# A different window is a different prefix, so runs of two windows never
# collide -- and the stage-1 cache, which is window-specific, can never be
# seeded across them.
source "$ROOT/streaming_couping/object_pose_feedback_env.zsh"

#: Which generation this round runs, and -- through the table below -- which
#: prompt set that generation means.  One line to switch; the tag and the word
#: list cannot disagree because the list is looked up from the tag.
GENERATION="${OBJECT_POSE_FEEDBACK_GENERATION:-v4}"

# What each generation's prompt set is.  A generation IS a prompt set, so the
# two are recorded together rather than in two places that have to be kept in
# step by hand.
#
#   v1  the five words the reported result was produced with
#   v2  v1 plus seven rigid objects -- adding words lost a category that v1 had
#   v3  v1 minus dustbin, the best-scoring category, asking whether the set
#       still works without it.  Not "v1's objects minus dustbin": the prompt
#       list competes for a fixed number of object slots (see adapters.py), so
#       the object set that comes out is not predictable from the words in.
#   v4  v1 plus ONE word, chosen as the most-kept rigid object the inventory
#       lists as unreached.  ONE, because adding is not free: v2 added seven and
#       lost a category it already had.
GENERATION_PROMPTS=(
  "v1:bed wardrobe chair rug dustbin"
  "v2:bed wardrobe chair rug dustbin table mat nightstand cabinet picture window door"
  "v3:bed wardrobe chair rug"
  "v4:bed wardrobe chair rug dustbin table"
)

PROMPTS=()
if [[ -n "${OBJECT_POSE_FEEDBACK_PROMPTS:-}" ]]; then
  # An explicit list wins, so an ad-hoc set can be tried without inventing a
  # generation tag for it.
  PROMPTS=("${(@s:,:)OBJECT_POSE_FEEDBACK_PROMPTS}")
else
  for entry in "${GENERATION_PROMPTS[@]}"; do
    [[ "${entry%%:*}" == "$GENERATION" ]] || continue
    words="${entry#*:}"
    PROMPTS=("${(@s: :)words}")
  done
  if (( ${#PROMPTS} == 0 )); then
    print -u2 "no prompt set recorded for generation $GENERATION"
    known=("${GENERATION_PROMPTS[@]%%:*}")
    print -u2 "known generations: ${(j:, :)known}"
    print -u2 "add it to GENERATION_PROMPTS, or set OBJECT_POSE_FEEDBACK_PROMPTS"
    exit 1
  fi
fi

# Generations the closing table compares, oldest first.  A generation with no
# readable run is skipped, so listing one that was never run is harmless.
# These are generations WITHIN the current window; the 100-frame window's v1 is
# a different set of directories and is not listed here.
#
# The list lives in the shared env file because the readers use it too, and
# they have no access to GENERATION_PROMPTS.  So a generation added to one and
# not the other is silently missing from the comparison table -- which is
# exactly what happened to v4 on its first round.  This checks instead.
COMPARE_GENERATIONS=("${GENERATIONS[@]}")
for known in "${GENERATION_PROMPTS[@]}"; do
  tag="${known%%:*}"
  if (( ${GENERATIONS[(Ie)$tag]} == 0 )); then
    print -u2 "generation $tag has a prompt set but is not in the shared"
    print -u2 "  GENERATIONS list in object_pose_feedback_env.zsh, so the"
    print -u2 "  comparison table would omit it.  Add it there."
    exit 1
  fi
done
if (( ${GENERATIONS[(Ie)$GENERATION]} == 0 )); then
  print -u2 "generation $GENERATION is not in the shared GENERATIONS list"
  print -u2 "  in object_pose_feedback_env.zsh; the readers would not find its runs."
  exit 1
fi
#: The window the earlier generations were run at, kept so the reader can say
#: which window a listed generation belongs to rather than implying it is this
#: one.  See opf_run_dir in the shared env file.


PRIOR_RUN_DIR=""
for candidate in ${(Oa)COMPARE_GENERATIONS}; do
  [[ "$candidate" == "$GENERATION" ]] && continue
  if [[ -f "$(opf_run_dir "$candidate")/horizonstream_geometry.pt" ]]; then
    PRIOR_RUN_DIR="$(opf_run_dir "$candidate")"
    break
  fi
done
BASE_RUN_DIR="${OBJECT_POSE_FEEDBACK_RUN_DIR:-$(opf_run_dir "$GENERATION")}"
# Which proposal-side branches to run.  One is the default: the other four are
# ablations, they cost one segmentation pass each only when they need a replay
# (they do not -- see below), and they are read from the tables rather than
# re-derived every round.
#
#   baseline          the method; what every result is reported from
#   translation_only  falsified (worse everywhere it was tried)
#   fresh             bounded reference age; worse than baseline at both windows
#   factorized        rotation-then-translation; passes or fails depending on the
#                     prompt set, so it is not a configuration to report
#   fresh_factor      both of the above knobs
#
# The full sweep is one edit away:
#   OBJECT_POSE_FEEDBACK_BRANCHES=baseline,translation_only,fresh,factorized,fresh_factor
if [[ -n "${OBJECT_POSE_FEEDBACK_BRANCHES:-}" ]]; then
  BRANCHES=("${(@s:,:)OBJECT_POSE_FEEDBACK_BRANCHES}")
else
  BRANCHES=(baseline)
fi
#: The consensus variant the tables headline and the figures draw.  Matches
#: MAIN_VARIANT_NAME in object_pose_feedback.py; named here too because the
#: figures are drawn from this script.
MAIN_VARIANT="${OBJECT_POSE_FEEDBACK_MAIN_VARIANT:-robust_semantic}"
FRESH_MAX_AGE="${OBJECT_POSE_FEEDBACK_MAX_REFERENCE_AGE_FRAMES:-15}"
FRESH_REFRESH="${OBJECT_POSE_FEEDBACK_ANCHOR_REFRESH_INTERVAL_FRAMES:-20}"

# How many times to run the FULL baseline pipeline, to measure the spread of
# one unchanged configuration.  Three runs came out at 2e-05 on the direct-ATE
# gain and 0 on the proposal count, which is the evidence that a branch
# difference of a few percent is readable.
#
# 3 means two extra stage-2a passes; stage 1 is reused from the cache by
# --reuse-if-valid, so the geometry model runs once.  Set it to 1 to skip the
# measurement and read the table without a noise floor.
# Every repeat is a full segmentation pass, which is the expensive half of a
# round.  One is the default because the numbers are already known to be stable
# within a shared geometry cache (spread ~1e-05); three is what measures that
# spread, and is worth paying for when a noise floor is being quoted rather
# than read.
SAM_REPEATS="${OBJECT_POSE_FEEDBACK_SAM_REPEATS:-1}"

MANIFEST_PATH="${SEMANTIC_MAP_MANIFEST:-$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d/manifest.json}"
SCENE_ID="${SEMANTIC_MAP_SCENE_ID:-00a231a370}"
GEOMETRY_CACHE="$BASE_RUN_DIR/horizonstream_geometry.pt"
BASELINE_RUN_DIR="$BASE_RUN_DIR.baseline"
BASELINE_DIAGNOSTICS="$BASELINE_RUN_DIR/object_pose_refinement/feedback_diagnostics.pt"
RUN_SCRIPT="$ROOT/streaming_couping/commands_run_scannet_object_pose_feedback_100f.txt"

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1
# Pin the shared cache path explicitly.  The inner script derives it from its
# own RUN_DIR, so naming a run dir per branch (as this sweep must) would
# otherwise put a full geometry cache inside every branch directory -- stage 1
# would run once per branch and the oracle would find nothing at the shared
# path.  Exporting it is what makes "one geometry run for all branches" true.
export OBJECT_POSE_FEEDBACK_GEOMETRY_CACHE="$GEOMETRY_CACHE"
# One source of truth for the prompt list.  The inner script used to carry its
# own copy with a comment saying it "must match" this one -- which is a matter
# of remembering, and getting it wrong makes the oracle score a different set of
# objects than the SAM branches tracked.  Exporting removes the chance.
export OBJECT_POSE_FEEDBACK_PROMPTS="${(j:,:)PROMPTS}"
# The inner script is a separate zsh process and resolves its own interpreter;
# exporting the one this sweep settled on keeps the two the same, so the window
# check it runs uses the interpreter that is actually known to work here.
export STREAMING_COUPING_PYTHON="$PYTHON"

for required in "$RUN_SCRIPT" "$MANIFEST_PATH"; do
  [[ -f "$required" ]] || { print -u2 "missing: $required"; exit 1; }
done
# The geometry cache is stage 1's OUTPUT, not an input: requiring it here would
# abort exactly when the outputs were cleaned out, which is when the sweep
# should be building them.
if [[ -f "$GEOMETRY_CACHE" ]]; then
  print "geometry_cache=$GEOMETRY_CACHE (will be reused)"
else
  print "geometry_cache=$GEOMETRY_CACHE (absent; stage 1 will build it)"
fi

print "===== OBJECT POSE FEEDBACK BRANCH SWEEP ====="
print "base_run_dir=$BASE_RUN_DIR"
print "branches=${(j:, :)BRANCHES}"
print "freshness=$FRESH_MAX_AGE/$FRESH_REFRESH"

# ---- Seed this generation's cache from the previous one -------------------
# Stage 1 is the only GPU stage, and it does not depend on the prompt set: the
# prompts enter at stage 2a, where they decide which masks are requested.  So a
# generation that changes only the prompt set can reuse the previous one's
# geometry outright.  Copying rather than pointing at the old path keeps the
# two generations independent -- if the cache ever fails validation and gets
# rebuilt, it is rebuilt here, and v1 stays as it was.
if [[ ! -f "$GEOMETRY_CACHE" && -f "$PRIOR_RUN_DIR/horizonstream_geometry.pt" ]]; then
  print "seeding the geometry cache from the previous generation"
  print "  from: $PRIOR_RUN_DIR/horizonstream_geometry.pt"
  print "  to:   $GEOMETRY_CACHE"
  mkdir -p "$GEOMETRY_CACHE:h"
  cp "$PRIOR_RUN_DIR/horizonstream_geometry.pt" "$GEOMETRY_CACHE"
  # The chunk camera maps are a separate artifact written alongside; the inner
  # script hard-requires them, so a cache copied without them would fail after
  # stage 1 had already been skipped.
  cp "$PRIOR_RUN_DIR/horizonstream_geometry_chunk_cam_maps.pt" \
     "$GEOMETRY_CACHE:h/horizonstream_geometry_chunk_cam_maps.pt" 2>/dev/null \
    || print -u2 "note: no chunk camera maps to copy; stage 1 must supply them"
  print "  v1 is untouched (copy, not move)"
fi

# ---- Adopt a stranded geometry cache --------------------------------------
# A run aborted after stage 1 but before the cache was where it belongs leaves
# a full cache inside a branch directory.  Rescuing it saves the whole geometry
# pass, and --reuse-if-valid re-validates it, so adopting a stale one is safe.
if [[ ! -f "$GEOMETRY_CACHE" ]]; then
  # (N) is required: zsh makes an unmatched glob a hard error, so on the
  # ordinary case -- a new window with no leftovers -- the loop would abort the
  # sweep instead of finding nothing.
  for stranded in "$BASE_RUN_DIR".*/horizonstream_geometry.pt(N); do
    [[ -f "$stranded" ]] || continue
    # The base directory may itself be gone, so the cache cannot merely be
    # renamed into it.
    mkdir -p "$GEOMETRY_CACHE:h"
    print "adopting a stranded geometry cache from ${stranded:h:t}/"
    mv -n "$stranded" "$GEOMETRY_CACHE" || {
      print -u2 "could not move $stranded; leaving it in place"
      continue
    }
    mv -n "$stranded:h/horizonstream_geometry_chunk_cam_maps.pt" \
          "$GEOMETRY_CACHE:h/" 2>/dev/null
    break
  done
fi

# ---- Refuse to clobber anything this sweep would write --------------------
existing=("$BASE_RUN_DIR.oracle_mask")
for branch in "${BRANCHES[@]}"; do
  existing+=("$BASE_RUN_DIR.$branch")
done
if (( SAM_REPEATS > 1 )); then
  for repeat in $(seq 2 "$SAM_REPEATS"); do
    existing+=("$BASE_RUN_DIR.baseline_r$repeat")
  done
fi
stale=()
for expected in "${existing[@]}"; do
  [[ -e "$expected" ]] && stale+=("$expected")
done
if (( ${#stale} > 0 )); then
  print ""
  print "archiving run directories left over from an earlier attempt:"
  for entry in "${stale[@]}"; do
    archive="$entry.previous"
    rm -rf "$archive"
    mv "$entry" "$archive"
    print "  ${entry:t} -> ${archive:t}"
  done
  print "(one generation is kept per branch; delete the *.previous dirs when"
  print " you no longer want them)"
fi

# ---- Prompt reachability, before anything costs GPU time ------------------
# The whole round rests on the added prompts actually reaching the objects the
# inventory listed.  Matching is substring-based on the manifest label, so this
# is really a check that the names are spelled the way the manifest spells
# them -- cheap to get wrong, and expensive to discover after stage 2a.  It is
# a check and not a gate: a missing reach still leaves a run worth having.
if [[ -f "$GEOMETRY_CACHE" ]]; then
  print ""
  print "########## prompt reachability (CPU, seconds) ##########"
  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.list_scene_objects \
    --manifest "$MANIFEST_PATH" \
    --scene-id "$SCENE_ID" \
    --geometry-cache "$GEOMETRY_CACHE" \
    --prompts "${PROMPTS[@]}" \
    --report-out "$BASE_RUN_DIR.prompt_reachability.txt" \
    || print -u2 "inventory check failed; continuing (it is a check, not a gate)"
  print ""
  print "$GENERATION prompts (${#PROMPTS}): ${(j:, :)PROMPTS}"
fi

# ---- Stage 2a once, for the baseline --------------------------------------
print ""
print "########## branch=baseline (full pipeline; the only GPU stage) ##########"
# The log's directory is the run directory, which the inner script creates --
# and a redirect is opened before the command runs, so it has to exist first.
mkdir -p "$BASELINE_RUN_DIR"
OBJECT_POSE_FEEDBACK_PROPOSAL_BRANCH=baseline \
OBJECT_POSE_FEEDBACK_RUN_DIR="$BASELINE_RUN_DIR" \
  zsh "$RUN_SCRIPT" >"$BASELINE_RUN_DIR/sweep.log" 2>&1
if (( $? != 0 )); then
  print -u2 "baseline run FAILED; nothing else can be compared against it"
  print -u2 "  last lines of $BASELINE_RUN_DIR/sweep.log:"
  tail -n 20 "$BASELINE_RUN_DIR/sweep.log" >&2
  exit 1
fi
# The stage-2a metrics are the same shape for every branch, and the comparison
# below reads the numbers from the artifacts rather than from this text -- so
# only the run's own headline is echoed, and the rest stays in its log.
print "$(grep -m1 '^pose=raw_pose ' "$BASELINE_RUN_DIR/sweep.log" || print 'stage 2a finished')"
[[ -f "$BASELINE_DIAGNOSTICS" ]] || {
  print -u2 "missing baseline diagnostics: $BASELINE_DIAGNOSTICS"
  exit 1
}
# ---- Oracle control: GT masks instead of the segmenter (CPU only) ---------
# This deliberately feeds ground truth into candidate generation, which the
# project otherwise forbids, because it is the only way to split the proposal
# error between "the segmentation model got the mask/identity wrong" and
# "nearest-neighbour correspondence or degenerate geometry is the limit".
# Its output is a diagnostic upper bound and must never be reported as a branch
# of the system. Compare its prop_err against the baseline's; that single
# comparison is the result.
ORACLE_RUN_DIR="$BASE_RUN_DIR.oracle_mask"
run_dirs=("$ORACLE_RUN_DIR")
print ""
print "########## oracle control: ground-truth instance masks (CPU) ##########"
print "run_dir=$ORACLE_RUN_DIR"
print "NOTE: diagnostic upper bound only; not a method result."
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.run_object_pose_loss_oracle \
  --geometry-cache "$GEOMETRY_CACHE" \
  --manifest "$MANIFEST_PATH" \
  --scene-id "$SCENE_ID" \
  --prompts "${PROMPTS[@]}" \
  --output-dir "$ORACLE_RUN_DIR" \
  >"$ORACLE_RUN_DIR.log" 2>&1
if (( $? == 0 )); then
  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.run_object_pose_feedback \
    --run-dir "$ORACLE_RUN_DIR" \
    --geometry-cache "$GEOMETRY_CACHE" \
    --manifest "$MANIFEST_PATH" \
    --scene-id "$SCENE_ID" \
    --output-dir "$ORACLE_RUN_DIR/object_pose_feedback" \
    --horizon-repo "$ROOT/externals/horizonstream" \
    >>"$ORACLE_RUN_DIR.log" 2>&1 \
    || print -u2 "oracle: stage-2b analysis failed"
  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.analyze_object_pose_feedback_attribution \
    --feedback-dir "$ORACLE_RUN_DIR/object_pose_feedback" \
    >>"$ORACLE_RUN_DIR.log" 2>&1 \
    || print -u2 "oracle: attribution failed (prop_err is still reported)"
  # One line, because the point of the control is one comparison: its prop_err
  # against the baseline's.  Everything else it prints is per-category detail
  # that is only worth reading once something in that one number looks wrong.
  print "oracle prop_err=$(PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" "$PYTHON" -c "
import json,sys
with open('$ORACLE_RUN_DIR/object_pose_feedback/summary.json') as handle:
    s = json.load(handle)
print(s.get('answers',{}).get('q2_single_proposal_error',{}).get('translation_median_m'))
" 2>/dev/null) full_output=$ORACLE_RUN_DIR.log"
else
  print -u2 "oracle control FAILED; the SAM-vs-GT comparison will be missing"
  print -u2 "  see $ORACLE_RUN_DIR.log"
fi


# ---- Gate: a replay of the baseline must reproduce the whole chain --------
# Proposals matching is necessary but not sufficient: what the comparison
# reads is the stage-2b output.  So replay, re-run stage 2b, and compare the
# decisions end to end.  Any material difference means the branch differences
# would be sitting on top of an unstable chain.
print ""
print "########## replay determinism gate ##########"
GATE_DIR="$BASE_RUN_DIR.replay_gate"
rm -rf "$GATE_DIR"
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.run_object_pose_loss_replay \
  --diagnostics "$BASELINE_DIAGNOSTICS" \
  --output-dir "$GATE_DIR" \
  --check-equivalence
if (( $? != 0 )); then
  print -u2 "replay does not reproduce the baseline proposals; aborting before"
  print -u2 "any branch comparison, because every branch would rest on it."
  exit 1
fi
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.run_object_pose_feedback \
  --run-dir "$GATE_DIR" \
  --geometry-cache "$GEOMETRY_CACHE" \
  --manifest "$MANIFEST_PATH" \
  --scene-id "$SCENE_ID" \
  --output-dir "$GATE_DIR/object_pose_feedback" \
  --horizon-repo "$ROOT/externals/horizonstream"
if (( $? != 0 )); then
  print -u2 "stage 2b failed on the replay gate; aborting"
  exit 1
fi
print "decisions from a replay of the baseline, vs the baseline itself:"
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.compare_json \
  --prefix decisions \
  --structural-only \
  "$BASELINE_RUN_DIR/object_pose_feedback/summary.json" \
  "$GATE_DIR/object_pose_feedback/summary.json"
if (( $? != 0 )); then
  print -u2 "the replayed chain does not reproduce the baseline structurally"
  print -u2 "(a decision flipped, or a field appeared or vanished); aborting,"
  print -u2 "because no branch difference above could be read."
  exit 1
fi
print "the numeric drift above is the ANALYSIS-CHAIN noise floor: the same"
print "cached inputs replayed twice move the pose metrics by that much, so a"
print "branch difference has to clear it as well as the segmentation spread."
rm -rf "$GATE_DIR"

# ---- Optional: repeat the whole baseline to measure the segmenter's spread -
noise_floor_dirs=()
if (( SAM_REPEATS > 1 )); then
  for repeat in $(seq 2 "$SAM_REPEATS"); do
    repeat_dir="$BASE_RUN_DIR.baseline_r$repeat"
    print ""
    print "########## baseline repeat $repeat/$SAM_REPEATS (GPU) ##########"
    mkdir -p "$repeat_dir"
    OBJECT_POSE_FEEDBACK_PROPOSAL_BRANCH=baseline \
    OBJECT_POSE_FEEDBACK_RUN_DIR="$repeat_dir" \
      zsh "$RUN_SCRIPT" >"$repeat_dir/sweep.log" 2>&1
    if (( $? != 0 )); then
      print -u2 "baseline repeat $repeat FAILED; it will not join the noise floor"
      continue
    fi
    print "finished (log: $repeat_dir/sweep.log)"
    noise_floor_dirs+=(--noise-floor "$repeat_dir")
  done
  # the first baseline is part of the same population
  noise_floor_dirs+=(--noise-floor "$BASELINE_RUN_DIR")
fi

# ---- Replay each branch, then analyse it ----------------------------------
failed_branches=()
for branch in "${BRANCHES[@]}"; do
  run_dir="$BASE_RUN_DIR.$branch"
  run_dirs+=("$run_dir")
  if [[ "$branch" == "baseline" ]]; then
    print ""
    print "########## baseline already ran; analysing ##########"
  else
    replay_flags=()
    case "$branch" in
      fresh)
        replay_flags+=(--object-pose-loss-max-reference-age-frames "$FRESH_MAX_AGE")
        replay_flags+=(--object-pose-loss-anchor-refresh-interval-frames "$FRESH_REFRESH")
        ;;
      factorized)
        replay_flags+=(--object-pose-loss-proposal-mode rotation_then_translation)
        ;;
      translation_only)
        replay_flags+=(--object-pose-loss-proposal-mode translation_only)
        ;;
      fresh_factor)
        replay_flags+=(--object-pose-loss-max-reference-age-frames "$FRESH_MAX_AGE")
        replay_flags+=(--object-pose-loss-anchor-refresh-interval-frames "$FRESH_REFRESH")
        replay_flags+=(--object-pose-loss-proposal-mode rotation_then_translation)
        ;;
    esac
    print ""
    print "########## branch=$branch (replay, CPU) ##########"
    print "run_dir=$run_dir"
    PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
      "$PYTHON" -m streaming_couping.scripts.run_object_pose_loss_replay \
      --diagnostics "$BASELINE_DIAGNOSTICS" \
      --output-dir "$run_dir" \
      "${replay_flags[@]}"
    if (( $? != 0 )); then
      print -u2 "branch $branch FAILED at the replay stage"
      failed_branches+=("$branch")
      continue
    fi
  fi

  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.run_object_pose_feedback \
    --run-dir "$run_dir" \
    --geometry-cache "$GEOMETRY_CACHE" \
    --manifest "$MANIFEST_PATH" \
    --scene-id "$SCENE_ID" \
    --output-dir "$run_dir/object_pose_feedback" \
    --horizon-repo "$ROOT/externals/horizonstream" \
    >>"$run_dir/analysis.log" 2>&1
  if (( $? != 0 )); then
    print -u2 "branch $branch FAILED at the stage-2b analysis"
    failed_branches+=("$branch")
    continue
  fi

  # Sections (a)-(d) are per-category detail.  Nothing in the comparison below
  # reads them, and printing them once per branch buried the table that does
  # decide the next step -- so they go to a file next to the run they describe.
  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.analyze_object_pose_feedback_attribution \
    --feedback-dir "$run_dir/object_pose_feedback" \
    >"$run_dir/object_pose_feedback/attribution.txt" 2>&1 \
    || print -u2 "branch $branch: attribution failed (comparison table still usable)"
done

print ""
print "===== BRANCH COMPARISON ====="
summary_args=()
for run_dir in "${run_dirs[@]}"; do
  summary_args+=(--run-dir "$run_dir")
done
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.summarize_feedback_branches \
  "${summary_args[@]}" \
  "${noise_floor_dirs[@]}" \
  --legend "$GENERATION prompts (${#PROMPTS}): ${(j:, :)PROMPTS}" \
  --json-out "$BASE_RUN_DIR.branches.json"

# ---- The comparison this round is actually about --------------------------
# Widening the prompt set is the independent variable, so the reading is
# generation against generation, branch by branch: v1/baseline next to
# v2/baseline, v1/fresh next to v2/fresh.  Interleaved rather than blocked so
# each pair sits on adjacent rows and a difference is visible without scrolling.
#
# No noise floor is attached: the floor measured above belongs to v2's own
# repeats, and v1 was not re-run this round, so there is no paired spread to
# quote for the difference between generations.  A v1-vs-v2 gap is read against
# that floor, not as though it had one of its own.
print ""
print "===== GENERATION COMPARISON (the prompt set is the variable) ====="
generation_dirs=()
for branch in "${BRANCHES[@]}"; do
  for generation in "${COMPARE_GENERATIONS[@]}"; do
    run_dir="$RUNS_ROOT/${RUN_PREFIX}_$generation.$branch"
    [[ -f "$run_dir/object_pose_feedback/summary.json" ]] \
      && generation_dirs+=("$run_dir")
  done
done
if (( ${#generation_dirs} > 1 )); then
  generation_args=()
  for run_dir in "${generation_dirs[@]}"; do
    generation_args+=(--run-dir "$run_dir")
  done
  legend=()
  for entry in "${GENERATION_PROMPTS[@]}"; do
    legend+=(--legend "${entry%%:*} = ${${entry#*:}:gs: :, :} prompts")
  done
  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.summarize_feedback_branches \
    "${generation_args[@]}" \
    "${legend[@]}" \
    --legend "a generation is a prompt set, not a code change: stage 1 is a copy" \
    --legend "of an earlier generation's cache, so the raw trajectory is identical" \
    --json-out "$BASE_RUN_DIR.prompt_comparison.json"
else
  print "only one generation has a readable run; nothing to compare yet"
  print "(looked for summaries under $RUNS_ROOT/${RUN_PREFIX}_*.baseline etc.)"
fi

if (( ${#failed_branches} > 0 )); then

  print -u2 "FAILED branches: ${(j:, :)failed_branches}"
