#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${BASELINE_STREAM_GPU0:-1}"
STREAM_GPU1="${BASELINE_STREAM_GPU1:-2}"
SAM_GPU="${BASELINE_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v0_baseline.yaml

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V0 smoke: causal tracking, QK joint geometry and raw fallback"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_dynamic_instance_baseline

ARGS=()
[[ "${BASELINE_REBUILD_CACHE:-0}" == "1" ]] && ARGS+=(--rebuild-cache)

echo "V0 physical GPUs=$VISIBLE_GPUS"
echo "stage 1/3: cache StreamVGGT geometry and SAM3.1 semantic tracks"
echo "logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
  --config "$CONFIG" \
  --stage cache \
  --streamvggt-devices cuda:0,cuda:1 \
  --sam3-device cuda:2 \
  --geometry-device cuda:0 \
  "${ARGS[@]}"

QK_OUTPUT=outputs/streaming_couping_v0/qk_joint_geometry_retrieval
QK_POSE="$QK_OUTPUT/qk_joint_geometry_output.pt"
QK_SUMMARY="$QK_OUTPUT/qk_joint_geometry_summary.json"
if [[ "${BASELINE_REBUILD_QK:-0}" == "1" || ! -s "$QK_POSE" || ! -s "$QK_SUMMARY" ]]; then
  rm -rf "$QK_OUTPUT"
  echo "stage 2/3: training-free RGB/QK joint geometry replay"
  echo "No SAM or GT enters candidate generation; frozen camera/depth/point heads run"
  CUDA_VISIBLE_DEVICES="$STREAM_GPU0,$STREAM_GPU1" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_v0_qk_pose_retrieval \
    --config "$CONFIG" \
    --streamvggt-devices cuda:0,cuda:1
else
  echo "stage 2/3: reuse existing clean QK artifact (strictly validated in stage 3)"
fi

echo "stage 3/3: select QK pose, audit tracking and score the 29-frame sequence"
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
  --config "$CONFIG" \
  --stage audit \
  --audit-device cuda:0

[[ -s outputs/streaming_couping_v0/baseline_summary.json ]] || {
  echo "missing baseline_summary.json" >&2
  exit 1
}
[[ ! -e outputs/streaming_couping_v0/validation/v0_pose_control_validation.csv ]] || {
  echo "retired V0 r3 validation CSV still exists" >&2
  exit 1
}
PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path
summary = json.loads(Path("outputs/streaming_couping_v0/baseline_summary.json").read_text())
assert summary["baseline_version"] == "v0"
assert summary["schema"] == 5
assert summary["implementation_revision"] == "qk_joint_geometry_semantic_tracking_r6"
assert summary["selected_pose_branch"] == "retrieve_qk"
assert summary["selected_pose_exact_raw"] == 0
assert summary["pose_modification_applied"] is True
assert summary["pose_selection_fallback_used"] == 0
assert summary["pose_improvement_claim"] is True
assert summary["selected_center_gain_vs_raw_percent"] > 0
assert summary["selected_rotation_gain_vs_raw_percent"] > 0
assert len(summary["pose_evaluation_frames"]) == 29
assert summary["sam_pose_inputs"] == 0
assert summary["candidate_geometry_available"] is True
assert summary["candidate_pointmap_used"] is False
assert summary["candidate_pointmap_used_for_pose"] is False
assert summary["depth_source"] == "raw_streamvggt"
assert summary["intrinsics_source"] == "raw_streamvggt"
assert summary["tracking_baseline_acceptance_pass"] == 1
assert summary["configured_instance_prompts"] == [
    "bed", "wardrobe", "picture", "mat", "chair"
]
assert summary["configured_permanent_slot_capacity"] == 16
audit = summary["tracking_audit"]
tracks = audit["discovered_track_count"]
late_births = audit["late_birth_count"]
slots = audit["permanent_slot_capacity"]
center_gain = summary["selected_center_gain_vs_raw_percent"]
rotation_gain = summary["selected_rotation_gain_vs_raw_percent"]
print(
    "V0 accepted tracking + QK-pose baseline "
    f"tracks={tracks} "
    f"late_births={late_births} "
    f"slots={slots} "
    f"center_gain={center_gain:.4f}% "
    f"R_gain={rotation_gain:.4f}%"
)
'
echo "V0 selected-pose artifact: outputs/streaming_couping_v0/poses.pt"
echo "V0 SAM semantic tracking + QK joint-geometry baseline completed"
