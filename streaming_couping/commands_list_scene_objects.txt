#!/usr/bin/env zsh

# What objects does this scene have, and which ones is the run missing?
#
#   zsh streaming_couping/commands_list_scene_objects.txt
#
# Read-only and CPU only, seconds.  No model runs.
#
# The prompts are a hard ceiling: an object no prompt matches cannot be
# proposed, corrected, or scored, however well it is seen.  This prints
#
#   * every ground-truth object visible in the run's own frame window, with the
#     mask size it gets on the run's own grid and which prompt reaches it;
#   * the objects no prompt reaches at all;
#   * the objects visible but too small (or too large) for the pipeline's own
#     mask filters, so they yield no observation even when prompted;
#   * what the segmenter actually produced, against the ground-truth side.
#
# To test a wider prompt set, add the words to PROMPTS below and run again: an
# added prompt changes which masks are requested, so it needs a stage-2a re-run,
# but no model change and no new labels.
set -uo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON="${STREAMING_COUPING_PYTHON:-$(command -v python)}"

RUNS_ROOT="$STORAGE_ROOT/outputs"
PREFIX="semantic_map_100frames_horizonstream_object_pose_feedback_90_189"
# Newest last, so the search below takes the most recent generation that has
# each artifact.
GENERATIONS=(v1 v2 v3)

# The ground-truth side of the inventory only needs a run's frames and grid;
# the segmenter side needs a run's diagnostics, and only the NEWEST generation's
# diagnostics say anything about the current prompt set.  The two are resolved
# separately because an old generation can supply the first while the second is
# still missing -- and when the diagnostics do come from the old one, say so,
# because a v1 segmenter listing read while thinking about v2 prompts is exactly
# the confusion this check exists to prevent.
newest="${GENERATIONS[-1]}"
BASE_RUN_DIR="${OBJECT_POSE_FEEDBACK_RUN_DIR:-$RUNS_ROOT/${PREFIX}_$newest}"
GEOMETRY_CACHE=""
DIAGNOSTICS=""
for generation in ${(Oa)GENERATIONS}; do
  [[ -n "$GEOMETRY_CACHE" ]] || {
    candidate="$RUNS_ROOT/${PREFIX}_$generation/horizonstream_geometry.pt"
    [[ -f "$candidate" ]] && GEOMETRY_CACHE="$candidate"
  }
  [[ -n "$DIAGNOSTICS" ]] || {
    candidate="$RUNS_ROOT/${PREFIX}_$generation.baseline/object_pose_refinement/feedback_diagnostics.pt"
    [[ -f "$candidate" ]] && DIAGNOSTICS="$candidate"
  }
done

# The ground-truth side only needs a run's frames and grid, which any
# generation supplies; the segmenter side needs a run's diagnostics, and only a
# CURRENT generation's say anything about the current prompt set.  So say which
# generation each came from -- a listing read while thinking about the current
# prompts, but showing an older generation's tracks, is exactly the confusion
# this check exists to prevent.
for required in "$GEOMETRY_CACHE" "$DIAGNOSTICS"; do
  [[ -n "$required" ]] || continue
  case "$required" in
    *"${PREFIX}_$newest"*) ;;
    *) print "note: reading an older generation for $required:t" ;;
  esac
done
MANIFEST_PATH="${SEMANTIC_MAP_MANIFEST:-$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d/manifest.json}"
SCENE_ID="${SEMANTIC_MAP_SCENE_ID:-00a231a370}"

# Edit this list to test a different prompt set.
PROMPTS=(bed wardrobe chair rug dustbin table mat nightstand cabinet picture window door)

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1

[[ -f "$MANIFEST_PATH" ]] || { print -u2 "missing: $MANIFEST_PATH"; exit 1; }
[[ -f "$GEOMETRY_CACHE" ]] || { print -u2 "missing: $GEOMETRY_CACHE"; exit 1; }

# The segmenter side is optional: without a run, only the ground-truth side is
# reported, which is still the answer to "what is in this scene".
ARGS=(
  --manifest "$MANIFEST_PATH"
  --scene-id "$SCENE_ID"
  --geometry-cache "$GEOMETRY_CACHE"
  --prompts "${PROMPTS[@]}"
)
if [[ -f "$DIAGNOSTICS" ]]; then
  ARGS+=(--feedback-diagnostics "$DIAGNOSTICS")
else
  print -u2 "note: no run diagnostics at $DIAGNOSTICS"
  print -u2 "      reporting the ground-truth side only"
fi

print "===== SCENE OBJECT INVENTORY ====="
print "scene=$SCENE_ID"
print "prompts=${(j:, :)PROMPTS}"
print ""

PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.list_scene_objects "${ARGS[@]}"
