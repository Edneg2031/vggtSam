#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
PHYSICAL_GPU="${INSTANCE_GEOMETRY_GPU:-1}"
CONFIG=streaming_couping/configs/v1_instance_geometry.yaml
OUTPUT=outputs/streaming_couping_v1/instance_geometry

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V1 smoke: causal ownership, erosion, equal-area shift and bounded surfel residual"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v1_instance_geometry

echo "V1 SAM-indexed instance geometry feasibility + candidate audit"
echo "physical GPU=$PHYSICAL_GPU logical device=cuda:0"
echo "Uses the rebuilt V0 cache only; no model/checkpoint is loaded or trained"
echo "GT is unavailable until every sparse candidate has been frozen"
CUDA_VISIBLE_DEVICES="$PHYSICAL_GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_v1_instance_geometry \
  --config "$CONFIG" \
  --device cuda:0

[[ -s "$OUTPUT/instance_geometry_summary.json" ]] || {
  echo "missing V1 instance geometry summary" >&2
  exit 1
}
PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path
summary = json.loads(Path(
    "outputs/streaming_couping_v1/instance_geometry/instance_geometry_summary.json"
).read_text())
assert summary["schema"] == 1
assert summary["revision"] == "v1_causal_instance_surfel_audit_r1"
assert summary["baseline_revision"] == "v0_frozen_semantic_mapping_pipeline_r1"
assert summary["baseline_pose_branch"] == "retrieve_qk"
assert summary["baseline_pointmap"] == "raw_full_history_world_pointmap"
assert summary["model_trained"] == 0
assert summary["sam_hidden_features_used"] == 0
assert summary["pose_modified"] == 0
assert summary["history_candidate_writeback"] == 0
assert summary["candidate_generation_gt_fields"] == 0
assert "bed" not in summary["prompts"]
assert set(row["branch"] for row in summary["i0_branches"]) == {
    "correct_id", "shuffled_id", "shifted_mask"
}
decision = summary["decision"]
assert decision["selected_pointmap_modified"] == 0
if decision["i0_pass"]:
    assert decision["i1_run"] == 1
    assert set(row["branch"] for row in summary["i1_branches"]) == {
        "raw", "global_surfel", "foreground_union", "correct_id",
        "shuffled_id", "shifted_mask",
    }
    assert Path(summary["outputs"]["candidates"]).is_file()
else:
    assert decision["i1_run"] == 0
    assert summary["i1_branches"] == []
print(
    "V1 schema accepted "
    f"eligible={decision['"'"'eligible_track_count'"'"']} "
    f"I0={decision['"'"'i0_pass'"'"']} "
    f"I1_run={decision['"'"'i1_run'"'"']} "
    f"I1_pass={decision['"'"'i1_pass'"'"']}"
)
'

echo ""
sed -n '1,220p' "$OUTPUT/copyable_result.txt"
