#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
OUTPUT=outputs/streaming_couping_v73_correspondence

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$OUTPUT/v73_correspondence_ablation.csv" ]] || {
  echo "missing formal V7.3 CSV" >&2
  exit 1
}

PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.analyze_v73_results \
  --input "$OUTPUT/v73_correspondence_ablation.csv" \
  --output-dir "$OUTPUT/report"

echo "V7.3 compact decision table"
cat "$OUTPUT/report/v73_decision_table.csv"
