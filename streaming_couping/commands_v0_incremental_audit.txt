#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${BASELINE_STREAM_GPU0:-1}"
STREAM_GPU1="${BASELINE_STREAM_GPU1:-2}"
CONFIG=streaming_couping/configs/v0_baseline.yaml
OUTPUT=outputs/streaming_couping_v0/incremental_audit
CACHE=outputs/streaming_couping_v0/cache/00a231a370_90_525_step15_37_68_54.pt
QK_OUTPUT=outputs/streaming_couping_v0/qk_joint_geometry_retrieval/qk_joint_geometry_output.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V0 incremental audit: pose-pointmap consistency + context controls"
echo "One command runs both gates; no model is trained and formal V0 is unchanged"
echo "Policies: full history, QK Top-4, recent Top-4 and random Top-4 with 3 seeds"

if [[ "${V0_AUDIT_REBUILD_BASELINE:-0}" == "1" || ! -s "$CACHE" || ! -s "$QK_OUTPUT" ]]; then
  echo "required V0 cache/QK artifact missing or rebuild requested; running baseline first"
  zsh streaming_couping/commands_v0_baseline.txt
else
  echo "reuse existing V0 cache and QK artifact"
fi

echo "V0 incremental-audit CPU smoke"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v0_incremental_audit

rm -rf "$OUTPUT"
echo "physical GPUs=$STREAM_GPU0,$STREAM_GPU1 logical StreamVGGT=cuda:0,cuda:1"
CUDA_VISIBLE_DEVICES="$STREAM_GPU0,$STREAM_GPU1" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_v0_incremental_audit \
  --config "$CONFIG" \
  --streamvggt-devices cuda:0,cuda:1

SUMMARY="$OUTPUT/incremental_audit_summary.json"
REPORT="$OUTPUT/copyable_result.txt"
[[ -s "$SUMMARY" ]] || { echo "missing $SUMMARY" >&2; exit 1; }
[[ -s "$REPORT" ]] || { echo "missing $REPORT" >&2; exit 1; }
[[ -s "$OUTPUT/pose_pointmap_consistency.csv" ]] || {
  echo "missing pose-pointmap consistency CSV" >&2
  exit 1
}
[[ -s "$OUTPUT/context_policy_summary.csv" ]] || {
  echo "missing context-policy summary CSV" >&2
  exit 1
}
[[ -s "$OUTPUT/context_retrieval_frames.csv" ]] || {
  echo "missing context-policy retrieval CSV" >&2
  exit 1
}

PYTHONPATH=. "$PYTHON_BIN" -c '
import json
import math
from pathlib import Path

path = Path("outputs/streaming_couping_v0/incremental_audit/incremental_audit_summary.json")
summary = json.loads(path.read_text())
assert summary["schema"] == 1
assert summary["revision"] == "v0_pose_pointmap_and_context_policy_audit_r1"
assert summary["baseline_version"] == "v0"
assert summary["stream_frames"] == 30
assert summary["evaluated_pose_frames"] == 29
assert summary["model_trained"] == 0
assert summary["sam_used_by_context_candidates"] == 0
assert summary["candidate_generation_fields"] == ["stream_images", "frame_indices"]
assert summary["total_history_budget"] == 5
assert summary["anchor_frames"] == 1
assert summary["random_seeds"] == [0, 1, 2]
assert {row["branch"] for row in summary["pose_pointmap_consistency"]} == {
    "raw_pose", "qk_pose"
}
for row in summary["pose_pointmap_consistency"]:
    assert row["valid_frames"] == 30
    for key in (
        "mean_frame_reprojection_median_px",
        "mean_frame_reprojection_p90_px",
        "mean_frame_relative_depth_median",
        "mean_frame_relative_depth_p90",
        "mean_positive_z_rate",
        "mean_in_bounds_rate",
    ):
        assert math.isfinite(row[key]), (row["branch"], key, row[key])
methods = {row["method"] for row in summary["context_policy_results"]}
assert methods == {
    "full_history",
    "qk_topk",
    "recent_topk",
    "random_topk_seed_0",
    "random_topk_seed_1",
    "random_topk_seed_2",
}
for row in summary["context_policy_results"]:
    for key in (
        "center_error_native",
        "rotation_degrees",
        "pointmap_paired_weighted_rmse",
        "depth_abs_rel",
    ):
        assert math.isfinite(row[key]), (row["method"], key, row[key])
assert summary["random_control_summary"]["runs"] == 3
decision = summary["decisions"]
for key in (
    "qk_pose_improves_over_full",
    "qk_pose_beats_recent",
    "qk_pose_beats_random_mean",
    "qk_beats_recent_and_random_pose_controls",
    "headwise_context_conflict_observed",
):
    assert decision[key] in (0, 1)
assert decision["pose_pointmap_compatibility_claim"] == 0
assert decision["formal_v0_pose_modified"] == 0
assert decision["formal_v0_pointmap_modified"] == 0
assert decision["sam_pose_probe_run"] == 0
print(
    "V0 incremental audit schema passed "
    f"context_control={decision['"'"'qk_beats_recent_and_random_pose_controls'"'"']} "
    f"headwise_conflict={decision['"'"'headwise_context_conflict_observed'"'"']}"
)
'

echo ""
echo "===== V0 INCREMENTAL AUDIT COPYABLE RESULT ====="
cat "$REPORT"
