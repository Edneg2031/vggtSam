#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V73_GPU:-1}"
CONFIG=streaming_couping/configs/v73_correspondence_ablation.yaml
OUTPUT=outputs/streaming_couping_v73_correspondence
V71_OUTPUT=outputs/streaming_couping_v71_instance_causality
V72_OUTPUT=outputs/streaming_couping_v72_local_token_ablation

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$V71_OUTPUT/frozen_l0.pt" && -s "$V71_OUTPUT/v71_instance_causality.csv" ]] || {
  echo "V7.3 requires the completed V7.1 frozen L0 and CSV." >&2
  exit 1
}
for checkpoint in \
  camera_extra_all.pt \
  camera_extra_common_gate.pt \
  geometry_local_match_k08.pt \
  geometry_local_match_k16.pt; do
  [[ -s "$V72_OUTPUT/$checkpoint" ]] || {
    echo "V7.3 requires fixed V7.2 checkpoint: $V72_OUTPUT/$checkpoint" >&2
    echo "Run: zsh streaming_couping/commands_v72_local_token_ablation.txt" >&2
    exit 1
  }
done

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

RESUME_ARGS=(--resume)
if [[ "${V73_FRESH:-0}" == "1" ]]; then
  RESUME_ARGS=()
  echo "V7.3 full retraining requested"
fi

echo "V7.3 physical GPU=$GPU logical device=cuda:0"
echo "V7.3 uses frozen caches; SAM3.1 and StreamVGGT backbones are not loaded"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v73_correspondence_ablation \
  --config "$CONFIG" \
  --device cuda:0 \
  "${RESUME_ARGS[@]}"

[[ -s "$OUTPUT/v73_correspondence_ablation.csv" ]] || {
  echo "missing V7.3 result CSV" >&2
  exit 1
}
echo "V7.3 COPY THIS ONE CSV"
cat "$OUTPUT/v73_correspondence_ablation.csv"
