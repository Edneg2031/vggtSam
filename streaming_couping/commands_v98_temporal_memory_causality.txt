#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V98_GPU:-3}"
CONFIG=streaming_couping/configs/v98_temporal_memory_causality.yaml
DATA_CONFIG=streaming_couping/configs/v92_support_data.yaml
V97_OUTPUT="${V97_OUTPUT:-outputs/streaming_couping_v97_dense_descriptor_causality}"
OUTPUT="${V98_OUTPUT:-outputs/streaming_couping_v98_temporal_memory_causality}"
SOURCE_CACHE=outputs/streaming_couping_v92_support_factorization/cache/00a231a370_90_525_step15_37_68_54.pt
V97_DENSE="$V97_OUTPUT/cache/00a231a370_90_525_step15_dense.pt"
TEMPORAL_CACHE="$OUTPUT/cache/00a231a370_90_525_step15_temporal.pt"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$SOURCE_CACHE" ]] || {
  echo "V9.8 requires the retained V9.2 observation cache: $SOURCE_CACHE" >&2
  exit 1
}
[[ -s "$V97_DENSE" ]] || {
  echo "V9.8 requires the completed V9.7 dense cache: $V97_DENSE" >&2
  exit 1
}
[[ -s "$V97_OUTPUT/v97_dense_descriptor_summary.csv" ]] || {
  echo "V9.8 requires the completed V9.7 summary in $V97_OUTPUT" >&2
  exit 1
}
for fold in short medium long; do
  for architecture in sam_dense sam_train_off; do
    [[ -s "$V97_OUTPUT/checkpoints/${fold}_${architecture}.pt" ]] || {
      echo "V9.8 requires V9.7 checkpoint: $V97_OUTPUT/checkpoints/${fold}_${architecture}.pt" >&2
      exit 1
    }
  done
done

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V9.8 smoke: tracker memory boundary, prompt merge, gradients and decision schema"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.smoke_v98_temporal_memory_causality

echo "V9.8 validating the retained observation cache"
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

echo "V9.8 caching genuine SAM3.1 forward-only history-read maps"
echo "physical GPU=$GPU; logical cache/training device=cuda:0"
REBUILD_ARGS=()
if [[ "${V98_REBUILD_TEMPORAL:-0}" == "1" ]]; then
  REBUILD_ARGS=(--rebuild)
fi
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.build_v98_temporal_memory_cache \
  --config "$CONFIG" \
  --device cuda:0 \
  --temporal-cache-path "$TEMPORAL_CACHE" \
  --v97-dense-cache-path "$V97_DENSE" \
  "${REBUILD_ARGS[@]}"

[[ -s "$TEMPORAL_CACHE" ]] || {
  echo "missing V9.8 temporal cache: $TEMPORAL_CACHE" >&2
  exit 1
}

echo "V9.8 V9.7 train/future audit + correspondence-only temporal matcher training"
echo "The completed V9.7 zero-input control is reused exactly and is not retrained"
echo "No pose model is trained; controls=memory-off, detector-FPN, StreamVGGT, train-off, channel/time perturbations"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v98_temporal_memory_causality \
  --config "$CONFIG" \
  --output-dir "$OUTPUT" \
  --temporal-cache-path "$TEMPORAL_CACHE" \
  --v97-output-dir "$V97_OUTPUT" \
  --v97-dense-cache-path "$V97_DENSE"

SUMMARY="$OUTPUT/v98_temporal_memory_summary.csv"
FRAMES="$OUTPUT/v98_temporal_memory_frames.csv"
TRAINING="$OUTPUT/v98_temporal_memory_training.csv"
AUDIT="$OUTPUT/v98_v97_train_future_audit.csv"
DECISION="$OUTPUT/v98_temporal_memory_decision.md"
METADATA="$OUTPUT/v98_temporal_memory_metadata.json"
TRACKS="$OUTPUT/v98_temporal_track_diagnostics.csv"
for file in "$SUMMARY" "$FRAMES" "$TRAINING" "$AUDIT" "$DECISION" "$METADATA" "$TRACKS"; do
  [[ -s "$file" ]] || { echo "missing V9.8 output: $file" >&2; exit 1; }
done

echo "V9.8 COPY THIS DECISION"
cat "$DECISION"
echo "V9.8 DYNAMIC OBJECTS"
cat "$TRACKS"
echo "V9.8 compact 18-row CSV: $SUMMARY"
echo "V9.8 V9.7 train/future audit: $AUDIT"
echo "V9.8 frame diagnostics: $FRAMES"
echo "V9.8 provenance: $METADATA"
