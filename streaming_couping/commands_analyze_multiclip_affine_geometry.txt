#!/usr/bin/env zsh
set -euo pipefail

# Frozen multi-scene diagnostic for causal depth scale-shift calibration and
# depth-head ray reconstruction.  It reuses existing caches only.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
PROTOCOL="${SEMANTIC_MAPPING_MULTICLIP_CONFIG:-$STORAGE_ROOT/outputs/streaming_couping_multiclip/protocol.yaml}"
POSE_ROOT="${SEMANTIC_MAPPING_MULTICLIP_RUN_ROOT:-$STORAGE_ROOT/outputs/streaming_couping_multiclip}"
OUTPUT_DIR="${MULTICLIP_AFFINE_GEOMETRY_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_multiclip_affine_geometry}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$PROTOCOL" ]] || {
  echo "missing multi-clip protocol: $PROTOCOL" >&2
  echo "run zsh streaming_couping/commands_plan_semantic_mapping_multiclip.txt first" >&2
  exit 1
}

export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1
export PYTHONPATH="$ROOT"

echo "multi-clip causal affine / depth-head ray diagnostic"
echo "models are not rerun; frozen caches are reused"
"$PYTHON_BIN" -m streaming_couping.scripts.smoke_multiclip_affine_geometry

"$PYTHON_BIN" -m streaming_couping.scripts.analyze_multiclip_affine_geometry \
  --config "$PROTOCOL" \
  --output-dir "$OUTPUT_DIR" \
  --pose-source both \
  --pose-root "$POSE_ROOT" \
  --confidence-threshold 0.30 \
  --track-score-threshold 0.50 \
  --min-history-points 16 \
  --max-history-points 4096 \
  --max-points-per-history-frame 512 \
  --max-pair-points-per-query 2048 \
  --min-affine-samples 64 \
  --max-affine-samples 200000 \
  --affine-trim-quantile 0.90 \
  --affine-max-iterations 4 \
  --calibration-fraction 0.70

echo "===== COPYABLE MULTI-CLIP AFFINE GEOMETRY RESULT ====="
sed -n '1,260p' "$OUTPUT_DIR/copyable_result.txt"
echo "multi-clip affine geometry output=$OUTPUT_DIR"
