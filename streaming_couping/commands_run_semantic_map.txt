#!/usr/bin/env zsh
set -euo pipefail

# Edit this section, then run:
#   zsh streaming_couping/commands_run_semantic_map.txt
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
HORIZON_PYTHON="${HORIZONSTREAM_PYTHON:-/home/huawei/miniconda3/envs/horizonstream/bin/python}"
# SAM3.1 is importable from the HorizonStream environment on the target
# server, so both stages use one interpreter by default.  Set
# STREAMING_COUPING_PYTHON only when an older split environment is needed.
SAM_PYTHON="${STREAMING_COUPING_PYTHON:-$HORIZON_PYTHON}"

INPUT_MODE="rgb"               # rgb | cache
GEOMETRY_BACKEND="horizonstream"  # horizonstream | streamvggt
MANIFEST_PATH="$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d/manifest.json"
SCENE_ID="00a231a370"
DATASET_FRAME_INDICES=()
FRAME_PATHS=()
FRAME_START=90
FRAME_STRIDE=1
FRAME_COUNT=50
PROMPTS=("bed" "wardrobe" "chair" "rug" "dustbin")

OUTPUT_DIR="$STORAGE_ROOT/outputs/semantic_map_50frames_horizonstream_consecutive"
HORIZON_CACHE_PATH="$OUTPUT_DIR/horizonstream_geometry.pt"
V0_CACHE_PATH="$STORAGE_ROOT/outputs/streaming_couping_v0/cache/00a231a370_90_525_step15_37_68_54.pt"
RECOVERY_CONFIG="streaming_couping/configs/recovery_dynamic_instance.yaml"

# Run both map policies from one shared HorizonStream + SAM inference.  The
# raw branch remains the baseline; temporal_consensus is an isolated candidate
# under OUTPUT_DIR/temporal_consensus.
FUSION_POLICY="both"       # raw | temporal_consensus | both

# HorizonStream runs first and exits before SAM is loaded.
HORIZON_REPO="externals/horizonstream"
HORIZON_CHECKPOINT="/home/bod/86Nas/95_data_bak/FoundationModels/HorizonStream.pt"
HORIZON_DEVICE="cuda:0"
HORIZON_WINDOW_SIZE=10
HORIZON_SLIDING_SIZE=1        # Stable streaming mode for consecutive-frame input.
HORIZON_IMAGE_SIZE=518
HORIZON_PATCH_SIZE=14
HORIZON_CROP=1
HORIZON_PRECISION="float16"  # float16 | bfloat16 | float32
HORIZON_REUSE_CACHE=1

# Leave empty to use sam3.device from RECOVERY_CONFIG (currently cuda:2).
SAM_DEVICE=""
SAM_GROUNDING_BATCH_SIZE=1    # Raise to 2 or 4 only after a successful run.
SAM_OFFLOAD_VIDEO_TO_CPU=1
MAX_SCENE_VOXELS=1000000

# Optional experimental layers.  Both are disabled for the reproducible raw
# SAM + HorizonStream baseline.  Enable them explicitly for an ablation and
# use a separate OUTPUT_DIR so baseline artifacts are not overwritten.
GEOMETRY_GUIDANCE=0
GEOMETRY_GUIDANCE_PROMPT_MODE="box_points"  # box_points | box_only | points_only
GEOMETRY_GUIDANCE_MIN_HISTORY_POINTS=16
GEOMETRY_GUIDANCE_MAX_HISTORY_POINTS=4096
GEOMETRY_GUIDANCE_MAX_HISTORY_FRAMES=8
GEOMETRY_GUIDANCE_MAX_HISTORY_GAP=30
GEOMETRY_GUIDANCE_MAX_POSITIVE_POINTS=6
GEOMETRY_GUIDANCE_MIN_CANDIDATE_SUPPORT=0.25
GEOMETRY_GUIDANCE_RELIABLE_SUPPORT_MARGIN=0.10
GEOMETRY_GUIDANCE_RELIABLE_SCORE_MARGIN=0.10
GEOMETRY_GUIDANCE_MAX_AREA_RATIO=4.0

MAP_WRITE_GATE=0
MAP_WRITE_MIN_OBSERVATIONS=2
MAP_WRITE_REENTRY_CONFIRMATION=2
MAP_WRITE_REENTRY_GAP=3
MAP_WRITE_MIN_MASK_PIXELS=32
MAP_WRITE_MIN_OBSERVATION_POINTS=16
MAP_WRITE_MIN_RELIABILITY=0.20
MAP_WRITE_MIN_REPROJECTION_CONSISTENCY=0.20
MAP_WRITE_MIN_AREA_RATIO=0.35
MAP_WRITE_MAX_AREA_RATIO=2.85
MAP_WRITE_SOFT_WEIGHT_FLOOR=0.25

# Causal temporal-consensus map branch.  These settings affect only semantic
# voxel writes; raw object_tracks are always retained unchanged.
TEMPORAL_CONSENSUS_HISTORY_FRAMES=8
TEMPORAL_CONSENSUS_MAX_HISTORY_POINTS=4096
TEMPORAL_CONSENSUS_MIN_HISTORY_POINTS=16
TEMPORAL_CONSENSUS_SUPPORT_RADIUS_M=0.08
TEMPORAL_CONSENSUS_MIN_SUPPORT_POINTS=16
TEMPORAL_CONSENSUS_MIN_SUPPORT_RATIO=0.20
TEMPORAL_CONSENSUS_MAX_NOVEL_POINTS=256
TEMPORAL_CONSENSUS_NOVEL_WEIGHT=0.20

# Optional SAM persistent-instance pose refinement.  This is a post-geometry,
# training-free ablation and is deliberately disabled for the baseline.
# When enabled, OUTPUT_DIR receives raw_pose/ and object_pose_refined/ maps,
# plus object_pose_refinement/ diagnostics.  The HorizonStream cache is never
# overwritten by this stage.
OBJECT_POSE_REFINEMENT=0
OBJECT_POSE_MIN_GAP=10
OBJECT_POSE_MIN_TRACK_SCORE=0.50
OBJECT_POSE_MIN_MASK_PIXELS=32
OBJECT_POSE_MAX_MASK_AREA_RATIO=0.85
OBJECT_POSE_MIN_GEOMETRY_POINTS=24
OBJECT_POSE_MIN_GEOMETRY_CONFIDENCE=0.30
OBJECT_POSE_MAX_PAIRS_PER_INSTANCE=8
OBJECT_POSE_MAX_TOTAL_CANDIDATE_PAIRS=256
OBJECT_POSE_FEATURE_BACKEND="rgb_patch"  # rgb_patch | dinov3
OBJECT_POSE_FEATURE_PATCH_SIZE=8
OBJECT_POSE_FEATURE_MAX_POINTS=1024
OBJECT_POSE_FEATURE_COSINE_THRESHOLD=0.65
OBJECT_POSE_FEATURE_RATIO_MARGIN=0.02
OBJECT_POSE_MAX_FEATURE_MATCHES=256
OBJECT_POSE_RANSAC_ITERATIONS=256
OBJECT_POSE_RANSAC_INLIER_THRESHOLD=0.10
OBJECT_POSE_MIN_MATCHES=12
OBJECT_POSE_MIN_INLIERS=6
OBJECT_POSE_MIN_INLIER_RATIO=0.30
OBJECT_POSE_MAX_REGISTRATION_RMSE=0.12
OBJECT_POSE_MAX_ROTATION_DISAGREEMENT=60.0
OBJECT_POSE_MAX_TRANSLATION_DISAGREEMENT=1.0
OBJECT_POSE_REJECT_EXTREME_MULTIPLIER=2.0
OBJECT_POSE_LOW_CONSISTENCY_WEIGHT=0.25
OBJECT_POSE_WEIGHT=1.0
OBJECT_POSE_SEQUENTIAL_WEIGHT=10.0
OBJECT_POSE_HUBER_DELTA=0.10
OBJECT_POSE_ROTATION_RESIDUAL_SCALE=1.0
OBJECT_POSE_MAX_CORRECTION_ROTATION=45.0
OBJECT_POSE_MAX_CORRECTION_TRANSLATION=1.0
OBJECT_POSE_OPTIMIZER_MAX_NFEV=100
OBJECT_POSE_DINOV3_CHECKPOINT=""
OBJECT_POSE_DINOV3_ROOT="/home/bod/86Nas/95_data_bak/FoundationModels/dinov3"
OBJECT_POSE_DINOV3_VARIANT="dinov3-vitl16"
OBJECT_POSE_DINOV3_DEVICE="cuda:0"
OBJECT_POSE_DINOV3_DTYPE="bfloat16"

# Optional per-process GPU visibility. Device indices are remapped after this.
HORIZON_CUDA_VISIBLE_DEVICES=""
SAM_CUDA_VISIBLE_DEVICES=""

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTORCH_CUDA_ALLOC_CONF="${PYTORCH_CUDA_ALLOC_CONF:-expandable_segments:True}"

[[ -x "$SAM_PYTHON" ]] || {
  print -u2 "missing SAM Python: $SAM_PYTHON"
  exit 1
}

# Explicit CLI arguments are an advanced, map-stage-only escape hatch. They
# must include --geometry-cache when HorizonStream geometry is desired.
if (( $# > 0 )); then
  PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$SAM_PYTHON" -m streaming_couping.scripts.run_semantic_map "$@"
  exit $?
fi

SOURCE_ARGS=()
if [[ "$INPUT_MODE" == "rgb" ]]; then
  if [[ -n "$MANIFEST_PATH" ]]; then
    [[ -n "$SCENE_ID" ]] || {
      print -u2 "SCENE_ID must be set when MANIFEST_PATH is used."
      exit 2
    }
    SOURCE_ARGS+=(--manifest "$MANIFEST_PATH" --scene-id "$SCENE_ID")
    if (( ${#DATASET_FRAME_INDICES[@]} > 0 )); then
      SOURCE_ARGS+=(--dataset-frame-indices "${DATASET_FRAME_INDICES[@]}")
    fi
  elif (( ${#FRAME_PATHS[@]} > 0 )); then
    SOURCE_ARGS+=(--frames "${FRAME_PATHS[@]}")
  else
    print -u2 "Set MANIFEST_PATH or FRAME_PATHS near the top of this file."
    exit 2
  fi
fi

SELECTION_ARGS=(
  --frame-start "$FRAME_START"
  --frame-stride "$FRAME_STRIDE"
  --frame-count "$FRAME_COUNT"
)
MAP_ARGS=()

case "$INPUT_MODE" in
  rgb)
    if (( ${#PROMPTS[@]} == 0 )); then
      print -u2 "PROMPTS must contain at least one text prompt."
      exit 2
    fi
    case "$GEOMETRY_BACKEND" in
      horizonstream)
        [[ -x "$HORIZON_PYTHON" ]] || {
          print -u2 "missing HorizonStream Python: $HORIZON_PYTHON"
          exit 1
        }
        [[ -d "$HORIZON_REPO/horizonstream" ]] || {
          print -u2 "missing HorizonStream source: $ROOT/$HORIZON_REPO"
          print -u2 "run: git submodule update --init externals/horizonstream"
          exit 1
        }
        [[ -f "$HORIZON_CHECKPOINT" ]] || {
          print -u2 "missing HorizonStream checkpoint: $HORIZON_CHECKPOINT"
          exit 1
        }
        HORIZON_ARGS=(
          "${SOURCE_ARGS[@]}"
          "${SELECTION_ARGS[@]}"
          --repo-path "$HORIZON_REPO"
          --checkpoint "$HORIZON_CHECKPOINT"
          --output-cache "$HORIZON_CACHE_PATH"
          --device "$HORIZON_DEVICE"
          --window-size "$HORIZON_WINDOW_SIZE"
          --sliding-size "$HORIZON_SLIDING_SIZE"
          --image-size "$HORIZON_IMAGE_SIZE"
          --patch-size "$HORIZON_PATCH_SIZE"
          --precision "$HORIZON_PRECISION"
        )
        if (( HORIZON_CROP )); then
          HORIZON_ARGS+=(--crop)
        else
          HORIZON_ARGS+=(--no-crop)
        fi
        if (( HORIZON_REUSE_CACHE )); then
          HORIZON_ARGS+=(--reuse-if-valid)
        else
          HORIZON_ARGS+=(--no-reuse-if-valid)
        fi
        print "[1/2] HorizonStream geometry cache"
        if [[ -n "$HORIZON_CUDA_VISIBLE_DEVICES" ]]; then
          CUDA_VISIBLE_DEVICES="$HORIZON_CUDA_VISIBLE_DEVICES" \
            PYTHONUNBUFFERED=1 \
            PYTHONPATH="$ROOT/$HORIZON_REPO:$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
            "$HORIZON_PYTHON" \
            -m streaming_couping.scripts.generate_horizonstream_geometry_cache \
            "${HORIZON_ARGS[@]}"
        else
          PYTHONUNBUFFERED=1 \
            PYTHONPATH="$ROOT/$HORIZON_REPO:$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
            "$HORIZON_PYTHON" \
            -m streaming_couping.scripts.generate_horizonstream_geometry_cache \
            "${HORIZON_ARGS[@]}"
        fi
        MAP_ARGS+=(--geometry-cache "$HORIZON_CACHE_PATH")
        ;;
      streamvggt)
        print "[1/2] StreamVGGT geometry will run inside the map process"
        ;;
      *)
        print -u2 "GEOMETRY_BACKEND must be horizonstream or streamvggt."
        exit 2
        ;;
    esac
    MAP_ARGS+=("${SOURCE_ARGS[@]}" "${SELECTION_ARGS[@]}")
    MAP_ARGS+=(--prompts "${PROMPTS[@]}")
    ;;
  cache)
    [[ -f "$V0_CACHE_PATH" ]] || {
      print -u2 "missing frozen V0 cache: $V0_CACHE_PATH"
      exit 1
    }
    MAP_ARGS+=(--cache "$V0_CACHE_PATH" "${SELECTION_ARGS[@]}")
    ;;
  *)
    print -u2 "INPUT_MODE must be rgb or cache."
    exit 2
    ;;
esac

MAP_ARGS+=(--output-dir "$OUTPUT_DIR")
MAP_ARGS+=(--fusion-policy "$FUSION_POLICY")
MAP_ARGS+=(--max-scene-voxels "$MAX_SCENE_VOXELS")
MAP_ARGS+=(--sam-grounding-batch-size "$SAM_GROUNDING_BATCH_SIZE")
MAP_ARGS+=(
  --temporal-consensus-history-frames "$TEMPORAL_CONSENSUS_HISTORY_FRAMES"
  --temporal-consensus-max-history-points "$TEMPORAL_CONSENSUS_MAX_HISTORY_POINTS"
  --temporal-consensus-min-history-points "$TEMPORAL_CONSENSUS_MIN_HISTORY_POINTS"
  --temporal-consensus-support-radius-m "$TEMPORAL_CONSENSUS_SUPPORT_RADIUS_M"
  --temporal-consensus-min-support-points "$TEMPORAL_CONSENSUS_MIN_SUPPORT_POINTS"
  --temporal-consensus-min-support-ratio "$TEMPORAL_CONSENSUS_MIN_SUPPORT_RATIO"
  --temporal-consensus-max-novel-points "$TEMPORAL_CONSENSUS_MAX_NOVEL_POINTS"
  --temporal-consensus-novel-weight "$TEMPORAL_CONSENSUS_NOVEL_WEIGHT"
)
if (( SAM_OFFLOAD_VIDEO_TO_CPU )); then
  MAP_ARGS+=(--sam-offload-video-to-cpu)
else
  MAP_ARGS+=(--no-sam-offload-video-to-cpu)
fi
if (( GEOMETRY_GUIDANCE )); then
  MAP_ARGS+=(
    --geometry-guidance
    --geometry-guidance-prompt-mode "$GEOMETRY_GUIDANCE_PROMPT_MODE"
    --geometry-guidance-min-history-points "$GEOMETRY_GUIDANCE_MIN_HISTORY_POINTS"
    --geometry-guidance-max-history-points "$GEOMETRY_GUIDANCE_MAX_HISTORY_POINTS"
    --geometry-guidance-max-history-frames "$GEOMETRY_GUIDANCE_MAX_HISTORY_FRAMES"
    --geometry-guidance-max-history-gap "$GEOMETRY_GUIDANCE_MAX_HISTORY_GAP"
    --geometry-guidance-max-positive-points "$GEOMETRY_GUIDANCE_MAX_POSITIVE_POINTS"
    --geometry-guidance-min-candidate-support "$GEOMETRY_GUIDANCE_MIN_CANDIDATE_SUPPORT"
    --geometry-guidance-reliable-support-margin "$GEOMETRY_GUIDANCE_RELIABLE_SUPPORT_MARGIN"
    --geometry-guidance-reliable-score-margin "$GEOMETRY_GUIDANCE_RELIABLE_SCORE_MARGIN"
    --geometry-guidance-max-area-ratio "$GEOMETRY_GUIDANCE_MAX_AREA_RATIO"
  )
fi
if (( MAP_WRITE_GATE )); then
  MAP_ARGS+=(
    --map-write-gate
    --map-write-min-observations "$MAP_WRITE_MIN_OBSERVATIONS"
    --map-write-reentry-confirmation "$MAP_WRITE_REENTRY_CONFIRMATION"
    --map-write-reentry-gap "$MAP_WRITE_REENTRY_GAP"
    --map-write-min-mask-pixels "$MAP_WRITE_MIN_MASK_PIXELS"
    --map-write-min-observation-points "$MAP_WRITE_MIN_OBSERVATION_POINTS"
    --map-write-min-reliability "$MAP_WRITE_MIN_RELIABILITY"
    --map-write-min-reprojection-consistency "$MAP_WRITE_MIN_REPROJECTION_CONSISTENCY"
    --map-write-min-area-ratio "$MAP_WRITE_MIN_AREA_RATIO"
    --map-write-max-area-ratio "$MAP_WRITE_MAX_AREA_RATIO"
    --map-write-soft-weight-floor "$MAP_WRITE_SOFT_WEIGHT_FLOOR"
  )
fi
if (( OBJECT_POSE_REFINEMENT )); then
  MAP_ARGS+=(
    --object-pose-refinement
    --object-pose-min-gap "$OBJECT_POSE_MIN_GAP"
    --object-pose-min-track-score "$OBJECT_POSE_MIN_TRACK_SCORE"
    --object-pose-min-mask-pixels "$OBJECT_POSE_MIN_MASK_PIXELS"
    --object-pose-max-mask-area-ratio "$OBJECT_POSE_MAX_MASK_AREA_RATIO"
    --object-pose-min-geometry-points "$OBJECT_POSE_MIN_GEOMETRY_POINTS"
    --object-pose-min-geometry-confidence "$OBJECT_POSE_MIN_GEOMETRY_CONFIDENCE"
    --object-pose-max-pairs-per-instance "$OBJECT_POSE_MAX_PAIRS_PER_INSTANCE"
    --object-pose-max-total-candidate-pairs "$OBJECT_POSE_MAX_TOTAL_CANDIDATE_PAIRS"
    --object-pose-feature-backend "$OBJECT_POSE_FEATURE_BACKEND"
    --object-pose-feature-patch-size "$OBJECT_POSE_FEATURE_PATCH_SIZE"
    --object-pose-feature-max-points "$OBJECT_POSE_FEATURE_MAX_POINTS"
    --object-pose-feature-cosine-threshold "$OBJECT_POSE_FEATURE_COSINE_THRESHOLD"
    --object-pose-feature-ratio-margin "$OBJECT_POSE_FEATURE_RATIO_MARGIN"
    --object-pose-max-feature-matches "$OBJECT_POSE_MAX_FEATURE_MATCHES"
    --object-pose-ransac-iterations "$OBJECT_POSE_RANSAC_ITERATIONS"
    --object-pose-ransac-inlier-threshold "$OBJECT_POSE_RANSAC_INLIER_THRESHOLD"
    --object-pose-min-matches "$OBJECT_POSE_MIN_MATCHES"
    --object-pose-min-inliers "$OBJECT_POSE_MIN_INLIERS"
    --object-pose-min-inlier-ratio "$OBJECT_POSE_MIN_INLIER_RATIO"
    --object-pose-max-registration-rmse "$OBJECT_POSE_MAX_REGISTRATION_RMSE"
    --object-pose-max-rotation-disagreement "$OBJECT_POSE_MAX_ROTATION_DISAGREEMENT"
    --object-pose-max-translation-disagreement "$OBJECT_POSE_MAX_TRANSLATION_DISAGREEMENT"
    --object-pose-reject-extreme-disagreement-multiplier "$OBJECT_POSE_REJECT_EXTREME_MULTIPLIER"
    --object-pose-low-consistency-weight "$OBJECT_POSE_LOW_CONSISTENCY_WEIGHT"
    --object-pose-weight "$OBJECT_POSE_WEIGHT"
    --object-pose-sequential-weight "$OBJECT_POSE_SEQUENTIAL_WEIGHT"
    --object-pose-huber-delta "$OBJECT_POSE_HUBER_DELTA"
    --object-pose-rotation-residual-scale "$OBJECT_POSE_ROTATION_RESIDUAL_SCALE"
    --object-pose-max-pose-correction-rotation "$OBJECT_POSE_MAX_CORRECTION_ROTATION"
    --object-pose-max-pose-correction-translation "$OBJECT_POSE_MAX_CORRECTION_TRANSLATION"
    --object-pose-optimizer-max-nfev "$OBJECT_POSE_OPTIMIZER_MAX_NFEV"
    --object-pose-dinov3-root "$OBJECT_POSE_DINOV3_ROOT"
    --object-pose-dinov3-variant "$OBJECT_POSE_DINOV3_VARIANT"
    --object-pose-dinov3-device "$OBJECT_POSE_DINOV3_DEVICE"
    --object-pose-dinov3-dtype "$OBJECT_POSE_DINOV3_DTYPE"
  )
  if [[ -n "$OBJECT_POSE_DINOV3_CHECKPOINT" ]]; then
    MAP_ARGS+=(--object-pose-dinov3-checkpoint "$OBJECT_POSE_DINOV3_CHECKPOINT")
  fi
fi
[[ -n "$RECOVERY_CONFIG" ]] && MAP_ARGS+=(--recovery-config "$RECOVERY_CONFIG")
if [[ -n "$SAM_DEVICE" ]]; then
  MAP_ARGS+=(--sam-device "$SAM_DEVICE")
elif [[ -n "$SAM_CUDA_VISIBLE_DEVICES" ]]; then
  MAP_ARGS+=(--sam-device "cuda:0")
fi

print "[2/2] SAM3.1 semantic-map fusion"
if [[ -n "$SAM_CUDA_VISIBLE_DEVICES" ]]; then
  CUDA_VISIBLE_DEVICES="$SAM_CUDA_VISIBLE_DEVICES" \
    PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$SAM_PYTHON" -m streaming_couping.scripts.run_semantic_map "${MAP_ARGS[@]}"
else
  PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$SAM_PYTHON" -m streaming_couping.scripts.run_semantic_map "${MAP_ARGS[@]}"
fi
