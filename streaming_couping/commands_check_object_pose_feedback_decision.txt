#!/usr/bin/env zsh

# Re-read the object-consensus pose-feedback verdict from an existing run.
#
#   zsh streaming_couping/commands_check_object_pose_feedback_decision.txt
#
# CPU only, seconds.  Stage 2b is re-run on the run directory that already
# exists, and the per-variant verdicts are printed.
#
# This is enough whenever only the DECISION changed -- which variant is
# headlined, which criteria are applied -- because the proposals and the
# replayed trajectories are untouched.  Stage 2a (the segmentation model) and
# stage 1 (the geometry model) do not have to run again.
#
# Which run it reads is picked, not fixed: the newest generation that is
# actually present.  When both are present the table carries both, because the
# prompt set is what separates them and the comparison is the point.
set -uo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON="${STREAMING_COUPING_PYTHON:-$(command -v python)}"

source "$ROOT/streaming_couping/object_pose_feedback_env.zsh"

# Newest first: the verdict is re-read for the current generation, and the
# comparison table carries the older ones beside it.
GENERATIONS=(${(Oa)GENERATIONS})

# The prompt list has to match the generation being read: it decides which
# ground-truth categories the object-map reference is built from, and scoring a
# 5-prompt run against a 12-prompt reference would pool categories the run
# never produced.
GENERATION_PROMPTS=(
  "v1:bed wardrobe chair rug dustbin"
  "v2:bed wardrobe chair rug dustbin table mat nightstand cabinet picture window door"
  "v3:bed wardrobe chair rug"
)

if [[ -n "${OBJECT_POSE_FEEDBACK_RUN_DIR:-}" ]]; then
  BASE_RUN_DIR="$OBJECT_POSE_FEEDBACK_RUN_DIR"
else
  BASE_RUN_DIR=""
  for generation in "${GENERATIONS[@]}"; do
    candidate="$(opf_run_dir "$generation")"
    [[ -d "$candidate.baseline" ]] && { BASE_RUN_DIR="$candidate"; break }
  done
  [[ -n "$BASE_RUN_DIR" ]] || {
    print -u2 "no run of any known generation is present"
    print -u2 "run commands_run_scannet_object_pose_feedback_branches.txt first"
    exit 1
  }
fi
GENERATION="${${BASE_RUN_DIR:t}##*_}"
RUN_DIR="$BASE_RUN_DIR.baseline"
GEOMETRY_CACHE="$BASE_RUN_DIR/horizonstream_geometry.pt"
MANIFEST_PATH="${SEMANTIC_MAP_MANIFEST:-$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d/manifest.json}"
SCENE_ID="${SEMANTIC_MAP_SCENE_ID:-00a231a370}"

PROMPTS=()
for entry in "${GENERATION_PROMPTS[@]}"; do
  [[ "${entry%%:*}" == "$GENERATION" ]] || continue
  words="${entry#*:}"
  PROMPTS=("${(@s: :)words}")
done
if (( ${#PROMPTS} == 0 )); then
  # An unlisted generation still has to be scorable: fall back to the newest
  # set on record rather than to an empty prompt list, which would silently
  # build the object-map reference from no categories at all.
  entry="${GENERATION_PROMPTS[-1]}"
  words="${entry#*:}"
  PROMPTS=("${(@s: :)words}")
  print -u2 "note: generation $GENERATION is not in GENERATION_PROMPTS;"
  print -u2 "      scoring against the newest recorded set instead"
fi

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1

for required in "$RUN_DIR/object_pose_refinement/feedback_diagnostics.pt" \
                "$GEOMETRY_CACHE" "$MANIFEST_PATH"; do
  [[ -f "$required" ]] || {
    print -u2 "missing: $required"
    print -u2 "run commands_run_scannet_object_pose_feedback_branches.txt first"
    exit 1
  }
done

print "===== OBJECT POSE FEEDBACK VERDICT ====="
print "run_dir=$RUN_DIR"
print "prompts=${#PROMPTS}"
print "cpu_only=1  segmentation_rerun=0"

PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.run_object_pose_feedback \
  --run-dir "$RUN_DIR" \
  --geometry-cache "$GEOMETRY_CACHE" \
  --manifest "$MANIFEST_PATH" \
  --scene-id "$SCENE_ID" \
  --output-dir "$RUN_DIR/object_pose_feedback" \
  --horizon-repo "$ROOT/externals/horizonstream"
if (( $? != 0 )); then
  print -u2 "stage-2b re-run failed"
  exit 1
fi

# ---- Per-variant verdict, both generations --------------------------------
# The baselines of the two generations are adjacent on purpose: they differ
# only in the prompt set, so that pair is the reading.  The other branches are
# a v2 question and are read in the sweep's own table.
print ""
print "===== PER-VARIANT VERDICT ====="
legend=()
for entry in "${GENERATION_PROMPTS[@]}"; do
  legend+=(--legend "${entry%%:*} = ${${entry#*:}:gs: :, :} prompts")
done
verdict_args=()
verdict_seen=()
for candidate in "$(opf_run_dir "${GENERATIONS[-1]}" baseline)" "$RUN_DIR"; do
  [[ -f "$candidate/object_pose_feedback/summary.json" ]] || continue
  # Reading v1 means RUN_DIR *is* the v1 baseline; the same row must not be
  # printed twice, which reads as two runs that happen to agree.
  (( ${verdict_seen[(Ie)$candidate]} )) && continue
  verdict_seen+=("$candidate")
  verdict_args+=(--run-dir "$candidate")
done
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.summarize_feedback_branches \
  "${verdict_args[@]}" \
  "${legend[@]}"

print ""
print "the 'decision' column is the headlined variant's verdict; 'go_variants'"
print "lists every consensus variant that passed ALL criteria, and 'main_failed'"
print "the criteria the headlined one did not."

# ---- Does the pose improvement reach the object map? ----------------------
# Same observations, placed by each trajectory in turn, against a reference
# built from the GT masks through the GROUND-TRUTH poses.  CPU only: no model
# runs, because the camera-space points are already in the diagnostics.
print ""
print "===== OBJECT MAP PER TRAJECTORY ====="
PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.evaluate_pose_feedback_object_map \
  --run-dir "$RUN_DIR" \
  --geometry-cache "$GEOMETRY_CACHE" \
  --manifest "$MANIFEST_PATH" \
  --scene-id "$SCENE_ID" \
  --prompts "${PROMPTS[@]}" \
  --json-out "$RUN_DIR/object_pose_feedback/object_map_metrics.json" \
  || print -u2 "object-map evaluation failed (the verdict above still stands)"
