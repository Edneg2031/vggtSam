#!/usr/bin/env zsh
set -euo pipefail

# Legacy filename retained so the server command does not change. TrackHead-BA
# is intentionally not rerun: its one-frame positive control already failed.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${BASELINE_STREAM_GPU0:-1}"
STREAM_GPU1="${BASELINE_STREAM_GPU1:-2}"
SAM_GPU="${BASELINE_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v0_baseline.yaml
CLIP=00a231a370_90_525_step15_37_68_54
PNP_OUTPUT=outputs/streaming_couping_v0/feature_pnp_factor_candidate
PNP_SUMMARY="$PNP_OUTPUT/candidate_summary.json"
PNP_FOLD_CSV="$PNP_OUTPUT/fold_decision.csv"
PNP_FRAME_CSV="$PNP_OUTPUT/frame_diagnostics.csv"
PNP_DECISION_MD="$PNP_OUTPUT/decision.md"
PNP_REVISION=causal_sift_landmark_history_factor_r3
LG_OUTPUT=outputs/streaming_couping_v0/lightglue_pnp_candidate
LG_SUMMARY="$LG_OUTPUT/candidate_summary.json"
LG_FOLD_CSV="$LG_OUTPUT/fold_decision.csv"
LG_FRAME_CSV="$LG_OUTPUT/frame_diagnostics.csv"
LG_PAIR_CSV="$LG_OUTPUT/pair_diagnostics.csv"
LG_DECISION_MD="$LG_OUTPUT/decision.md"
LG_REVISION=frozen_aliked_lightglue_native_all_causal_r7
LG_REPO=externals/LightGlue
LG_COMMIT=eb42fee2d71449efb0aa5c10549752b5d75384d8
LG_MODEL_ROOT="${LIGHTGLUE_MODEL_ROOT:-/home/bod/86Nas/95_data_bak/FoundationModels/LightGlue}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V0 TrackHead-BA skipped: retained one-frame validity control failed"
echo "V0 preparing a stable dynamic-tracking/geometry cache"
if PYTHONPATH=. "$PYTHON_BIN" -m \
  streaming_couping.scripts.prepare_v0_candidate_cache \
  --config "$CONFIG" \
  --clip "$CLIP"; then
  :
else
  PREPARE_STATUS=$?
  if [[ "$PREPARE_STATUS" != "2" ]]; then
    echo "V0 cache discovery failed with status=$PREPARE_STATUS" >&2
    exit "$PREPARE_STATUS"
  fi
  echo "No compatible cache remains; rebuilding the V0 cache once"
  echo "physical GPUs=$VISIBLE_GPUS"
  echo "logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
    --config "$CONFIG" \
    --stage cache \
    --rebuild-cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0
  PYTHONPATH=. "$PYTHON_BIN" -m \
    streaming_couping.scripts.prepare_v0_candidate_cache \
    --config "$CONFIG" \
    --clip "$CLIP"
fi

PNP_OUTPUT_REVISION=""
if [[ -s "$PNP_SUMMARY" ]]; then
  PNP_OUTPUT_REVISION=$(PYTHONPATH=. "$PYTHON_BIN" - "$PNP_SUMMARY" <<'PY'
import json
import sys
from pathlib import Path

try:
    print(json.loads(Path(sys.argv[1]).read_text()).get("implementation_revision", ""))
except (OSError, ValueError):
    print("")
PY
)
fi

if [[ "${V0_FEATURE_PNP_FORCE_RERUN:-0}" == "1" ]] || \
   [[ "$PNP_OUTPUT_REVISION" != "$PNP_REVISION" ]] || \
   [[ ! -s "$PNP_SUMMARY" ]] || [[ ! -s "$PNP_FOLD_CSV" ]] || \
   [[ ! -s "$PNP_FRAME_CSV" ]] || [[ ! -s "$PNP_DECISION_MD" ]]; then
  echo "V0 Feature-PnP smoke: matching, equal-count and locked fallback"
  PYTHONPATH=. "$PYTHON_BIN" -m \
    streaming_couping.scripts.smoke_v0_feature_pnp_candidate
  echo "V0 Feature-PnP landmark-source x causal-history factorization"
  echo "No matcher/pose model is trained; GT is scoring-only"
  PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" -m \
    streaming_couping.scripts.run_v0_feature_pnp_candidate \
    --config "$CONFIG"
else
  echo "V0 Feature-PnP r3 outputs already exist; skipping recomputation"
  echo "Set V0_FEATURE_PNP_FORCE_RERUN=1 only for an intentional rerun"
fi

[[ -s "$PNP_SUMMARY" ]] || { echo "missing Feature-PnP summary" >&2; exit 1; }
[[ -s "$PNP_FOLD_CSV" ]] || { echo "missing Feature-PnP fold CSV" >&2; exit 1; }
[[ -s "$PNP_FRAME_CSV" ]] || { echo "missing Feature-PnP frame CSV" >&2; exit 1; }
[[ -s "$PNP_DECISION_MD" ]] || { echo "missing Feature-PnP decision markdown" >&2; exit 1; }

PYTHONPATH=. "$PYTHON_BIN" - "$PNP_SUMMARY" "$PNP_FOLD_CSV" "$PNP_FRAME_CSV" <<'PY'
import csv
import json
import sys
from collections import Counter, defaultdict
from pathlib import Path

summary = json.loads(Path(sys.argv[1]).read_text())
assert summary["implementation_revision"] == "causal_sift_landmark_history_factor_r3"
assert summary["gt_used_for_candidate_generation"] is False
assert summary["pose_model_trained"] is False
assert summary["matcher_trained"] is False
decision = summary["decision"]
print(
    "V0 Feature-PnP completed "
    f"primary={summary['primary_branch_locked_before_scoring']} "
    f"all_fold_pass={decision['primary_all_fold_pass']} "
    "selected_pose_unchanged=1"
)
with Path(sys.argv[2]).open(encoding="utf8", newline="") as handle:
    folds = list(csv.DictReader(handle))
print("V0 Feature-PnP fold decision")
for row in folds:
    print(
        f"  fold={row['fold']} landmark={row['landmark_source']} "
        f"history={row['history_scope']} support={row['support_method']} "
        f"active={row['active_frames']}/4 "
        f"equal_count={row['equal_count_feasible_frames']}/4 "
        f"center_gain={float(row['center_gain_percent']):.6g}% "
        f"center_worse={row['center_worse_frames']} "
        f"rotation_gain={float(row['rotation_gain_percent']):.6g}% "
        f"pass={row['fold_pass']}"
    )
with Path(sys.argv[3]).open(encoding="utf8", newline="") as handle:
    frames = list(csv.DictReader(handle))
grouped = defaultdict(list)
for row in frames:
    grouped[row["method"]].append(row)
print("V0 Feature-PnP optimization audit")
primary = summary["primary_branch_locked_before_scoring"]
for method, rows in grouped.items():
    reasons = Counter(row["reason"] for row in rows)
    active = sum(int(row["optimized"]) for row in rows)
    feasible = sum(int(row["equal_count_feasible"]) for row in rows)
    print(
        f"  method={method} active={active}/{len(rows)} "
        f"equal_count={feasible}/{len(rows)} reasons={dict(reasons)}"
    )
    if method != primary:
        continue
    print("    locked-primary per-frame diagnostics")
    for row in rows:
        print(
            "    "
            f"frame={row['frame_index']} "
            f"pool={row['pooled_unique_matches']} "
            f"locked={row['locked_equal_count']} "
            f"anchor_epe={row['landmark_anchor_reprojection_mean_epe_pixels']} "
            f"inliers={row['inliers']} "
            f"rmse={row['reprojection_rmse_pixels']} "
            f"rotation_update_deg={row['rotation_update_degrees']} "
            f"center_update={row['center_update_native']} "
            f"reason={row['reason']}"
        )
PY

echo "V0 Feature-PnP r3 retained as completed negative evidence"

echo "V0 r7 preparing pinned ALIKED-LightGlue inference dependency"
if [[ ! -d "$LG_REPO/.git" ]]; then
  git clone https://github.com/cvg/LightGlue.git "$LG_REPO"
  git -C "$LG_REPO" checkout --detach "$LG_COMMIT"
fi
LG_ACTUAL_COMMIT=$(git -C "$LG_REPO" rev-parse HEAD)
[[ "$LG_ACTUAL_COMMIT" == "$LG_COMMIT" ]] || {
  echo "LightGlue checkout mismatch: expected=$LG_COMMIT actual=$LG_ACTUAL_COMMIT" >&2
  echo "Move the existing external checkout or set it to the pinned commit." >&2
  exit 1
}
if ! PYTHONPATH=. "$PYTHON_BIN" -c 'import kornia' >/dev/null 2>&1; then
  echo "Installing the missing LightGlue runtime dependency kornia"
  "$PYTHON_BIN" -m pip install 'kornia>=0.6.11,<1'
fi

mkdir -p "$LG_MODEL_ROOT/torch/hub/checkpoints"
ALIKED_WEIGHT="$LG_MODEL_ROOT/torch/hub/checkpoints/aliked-n16.pth"
LG_WEIGHT="$LG_MODEL_ROOT/torch/hub/checkpoints/aliked_lightglue_v0-1_arxiv.pth"
if [[ ! -s "$ALIKED_WEIGHT" ]]; then
  echo "Downloading public frozen ALIKED weight to mounted model storage"
  curl -L --fail --show-error -o "$ALIKED_WEIGHT.part" \
    https://github.com/Shiaoming/ALIKED/raw/main/models/aliked-n16.pth
  mv "$ALIKED_WEIGHT.part" "$ALIKED_WEIGHT"
fi
if [[ ! -s "$LG_WEIGHT" ]]; then
  echo "Downloading public frozen ALIKED-LightGlue weight to mounted model storage"
  curl -L --fail --show-error -o "$LG_WEIGHT.part" \
    https://github.com/cvg/LightGlue/releases/download/v0.1_arxiv/aliked_lightglue.pth
  mv "$LG_WEIGHT.part" "$LG_WEIGHT"
fi
PYTHONPATH=. "$PYTHON_BIN" - "$ALIKED_WEIGHT" "$LG_WEIGHT" <<'PY'
import hashlib
import sys
from pathlib import Path

expected = {
    "aliked-n16.pth": "5be8704840ed662d9d8c561bf7279c222092674e7eb05fd0feab94899e9d82f2",
    "aliked_lightglue_v0-1_arxiv.pth": "d975e965b105311a6143194852297dff4f02aea5cc2e10cecfed966ca0e22503",
}
for value in sys.argv[1:]:
    path = Path(value)
    digest = hashlib.sha256(path.read_bytes()).hexdigest()
    if digest != expected[path.name]:
        raise SystemExit(
            f"LightGlue weight checksum mismatch path={path} sha256={digest}"
        )
    print(f"LightGlue weight verified path={path} sha256={digest}")
PY
export TORCH_HOME="$LG_MODEL_ROOT/torch"

LG_OUTPUT_REVISION=""
if [[ -s "$LG_SUMMARY" ]]; then
  LG_OUTPUT_REVISION=$(PYTHONPATH=. "$PYTHON_BIN" - "$LG_SUMMARY" <<'PY'
import json
import sys
from pathlib import Path

try:
    print(json.loads(Path(sys.argv[1]).read_text()).get("implementation_revision", ""))
except (OSError, ValueError):
    print("")
PY
)
fi

if [[ "${V0_LIGHTGLUE_PNP_FORCE_RERUN:-0}" == "1" ]] || \
   [[ "$LG_OUTPUT_REVISION" != "$LG_REVISION" ]] || \
   [[ ! -s "$LG_SUMMARY" ]] || [[ ! -s "$LG_FOLD_CSV" ]] || \
   [[ ! -s "$LG_FRAME_CSV" ]] || [[ ! -s "$LG_PAIR_CSV" ]] || \
   [[ ! -s "$LG_DECISION_MD" ]]; then
  echo "V0 LightGlue-PnP smoke: frozen matcher adapter and spatial audit"
  PYTHONPATH=.:"$LG_REPO" "$PYTHON_BIN" -m \
    streaming_couping.scripts.smoke_v0_lightglue_pnp_candidate
  echo "V0 LightGlue-PnP frozen correspondence causality audit"
  echo "Fixed native pointmap + all causal history + PnP; GT is scoring-only"
  CUDA_VISIBLE_DEVICES="$STREAM_GPU0" PYTHONUNBUFFERED=1 \
    PYTHONPATH=.:"$LG_REPO" "$PYTHON_BIN" -m \
    streaming_couping.scripts.run_v0_lightglue_pnp_candidate \
    --config "$CONFIG"
else
  echo "V0 LightGlue-PnP r7 outputs already exist; skipping recomputation"
  echo "Set V0_LIGHTGLUE_PNP_FORCE_RERUN=1 only for an intentional rerun"
fi

[[ -s "$LG_SUMMARY" ]] || { echo "missing LightGlue-PnP summary" >&2; exit 1; }
[[ -s "$LG_FOLD_CSV" ]] || { echo "missing LightGlue-PnP fold CSV" >&2; exit 1; }
[[ -s "$LG_FRAME_CSV" ]] || { echo "missing LightGlue-PnP frame CSV" >&2; exit 1; }
[[ -s "$LG_PAIR_CSV" ]] || { echo "missing LightGlue-PnP pair CSV" >&2; exit 1; }
[[ -s "$LG_DECISION_MD" ]] || { echo "missing LightGlue-PnP decision markdown" >&2; exit 1; }

PYTHONPATH=. "$PYTHON_BIN" - "$LG_SUMMARY" "$LG_FOLD_CSV" "$LG_FRAME_CSV" <<'PY'
import csv
import json
import sys
from collections import Counter
from pathlib import Path

summary = json.loads(Path(sys.argv[1]).read_text())
assert summary["implementation_revision"] == "frozen_aliked_lightglue_native_all_causal_r7"
assert summary["gt_used_for_candidate_generation"] is False
assert summary["pose_model_trained"] is False
assert summary["matcher_trained"] is False
print(
    "V0 LightGlue-PnP completed "
    f"primary={summary['primary_method_locked_before_scoring']} "
    f"all_fold_pass={summary['decision']['primary_all_fold_pass']} "
    "selected_pose_unchanged=1"
)
with Path(sys.argv[2]).open(encoding="utf8", newline="") as handle:
    folds = list(csv.DictReader(handle))
print("V0 LightGlue-PnP fold decision")
for row in folds:
    print(
        f"  fold={row['fold']} method={row['method']} "
        f"active={row['active_frames']}/4 "
        f"equal_count={row['equal_count_feasible_frames']}/4 "
        f"center_gain={float(row['center_gain_percent']):.6g}% "
        f"center_worse={row['center_worse_frames']} "
        f"rotation_gain={float(row['rotation_gain_percent']):.6g}% "
        f"pass={row['fold_pass']}"
    )
with Path(sys.argv[3]).open(encoding="utf8", newline="") as handle:
    frames = list(csv.DictReader(handle))
print("V0 LightGlue-PnP correspondence/optimization audit")
for method in summary["methods"]:
    rows = [row for row in frames if row["method"] == method]
    print(
        f"  method={method} "
        f"active={sum(int(row['optimized']) for row in rows)}/{len(rows)} "
        f"equal_count={sum(int(row['equal_count_feasible']) for row in rows)}/{len(rows)} "
        f"reasons={dict(Counter(row['reason'] for row in rows))}"
    )
    for row in rows:
        print(
            "    "
            f"frame={row['frame_index']} "
            f"pooled={row['pooled_unique_current_matches']} "
            f"locked={row['locked_equal_count']} "
            f"hull={float(row['current_selected_hull_coverage_fraction']):.6g} "
            f"inliers={row['inliers']} "
            f"rmse={row['reprojection_rmse_pixels']} "
            f"reason={row['reason']}"
        )
PY

echo "Send the V0 LightGlue-PnP fold decision and correspondence/optimization audit for review"
