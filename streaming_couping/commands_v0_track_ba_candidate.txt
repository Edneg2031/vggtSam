#!/usr/bin/env zsh
set -euo pipefail

# Legacy filename retained so the server command does not change. TrackHead-BA
# is intentionally not rerun: its one-frame positive control already failed.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${BASELINE_STREAM_GPU0:-1}"
STREAM_GPU1="${BASELINE_STREAM_GPU1:-2}"
SAM_GPU="${BASELINE_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v0_baseline.yaml
CLIP=00a231a370_90_525_step15_37_68_54
PNP_OUTPUT=outputs/streaming_couping_v0/feature_pnp_factor_candidate
PNP_SUMMARY="$PNP_OUTPUT/candidate_summary.json"
PNP_FOLD_CSV="$PNP_OUTPUT/fold_decision.csv"
PNP_FRAME_CSV="$PNP_OUTPUT/frame_diagnostics.csv"
PNP_DECISION_MD="$PNP_OUTPUT/decision.md"
PNP_REVISION=causal_sift_landmark_history_factor_r3

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V0 TrackHead-BA skipped: retained one-frame validity control failed"
echo "V0 preparing a stable dynamic-tracking/geometry cache"
if PYTHONPATH=. "$PYTHON_BIN" -m \
  streaming_couping.scripts.prepare_v0_candidate_cache \
  --config "$CONFIG" \
  --clip "$CLIP"; then
  :
else
  PREPARE_STATUS=$?
  if [[ "$PREPARE_STATUS" != "2" ]]; then
    echo "V0 cache discovery failed with status=$PREPARE_STATUS" >&2
    exit "$PREPARE_STATUS"
  fi
  echo "No compatible cache remains; rebuilding the V0 cache once"
  echo "physical GPUs=$VISIBLE_GPUS"
  echo "logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
    --config "$CONFIG" \
    --stage cache \
    --rebuild-cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0
  PYTHONPATH=. "$PYTHON_BIN" -m \
    streaming_couping.scripts.prepare_v0_candidate_cache \
    --config "$CONFIG" \
    --clip "$CLIP"
fi

PNP_OUTPUT_REVISION=""
if [[ -s "$PNP_SUMMARY" ]]; then
  PNP_OUTPUT_REVISION=$(PYTHONPATH=. "$PYTHON_BIN" - "$PNP_SUMMARY" <<'PY'
import json
import sys
from pathlib import Path

try:
    print(json.loads(Path(sys.argv[1]).read_text()).get("implementation_revision", ""))
except (OSError, ValueError):
    print("")
PY
)
fi

if [[ "${V0_FEATURE_PNP_FORCE_RERUN:-0}" == "1" ]] || \
   [[ "$PNP_OUTPUT_REVISION" != "$PNP_REVISION" ]] || \
   [[ ! -s "$PNP_SUMMARY" ]] || [[ ! -s "$PNP_FOLD_CSV" ]] || \
   [[ ! -s "$PNP_FRAME_CSV" ]] || [[ ! -s "$PNP_DECISION_MD" ]]; then
  echo "V0 Feature-PnP smoke: matching, equal-count and locked fallback"
  PYTHONPATH=. "$PYTHON_BIN" -m \
    streaming_couping.scripts.smoke_v0_feature_pnp_candidate
  echo "V0 Feature-PnP landmark-source x causal-history factorization"
  echo "No matcher/pose model is trained; GT is scoring-only"
  PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" -m \
    streaming_couping.scripts.run_v0_feature_pnp_candidate \
    --config "$CONFIG"
else
  echo "V0 Feature-PnP r3 outputs already exist; skipping recomputation"
  echo "Set V0_FEATURE_PNP_FORCE_RERUN=1 only for an intentional rerun"
fi

[[ -s "$PNP_SUMMARY" ]] || { echo "missing Feature-PnP summary" >&2; exit 1; }
[[ -s "$PNP_FOLD_CSV" ]] || { echo "missing Feature-PnP fold CSV" >&2; exit 1; }
[[ -s "$PNP_FRAME_CSV" ]] || { echo "missing Feature-PnP frame CSV" >&2; exit 1; }
[[ -s "$PNP_DECISION_MD" ]] || { echo "missing Feature-PnP decision markdown" >&2; exit 1; }

PYTHONPATH=. "$PYTHON_BIN" - "$PNP_SUMMARY" "$PNP_FOLD_CSV" "$PNP_FRAME_CSV" <<'PY'
import csv
import json
import sys
from collections import Counter, defaultdict
from pathlib import Path

summary = json.loads(Path(sys.argv[1]).read_text())
assert summary["implementation_revision"] == "causal_sift_landmark_history_factor_r3"
assert summary["gt_used_for_candidate_generation"] is False
assert summary["pose_model_trained"] is False
assert summary["matcher_trained"] is False
decision = summary["decision"]
print(
    "V0 Feature-PnP completed "
    f"primary={summary['primary_branch_locked_before_scoring']} "
    f"all_fold_pass={decision['primary_all_fold_pass']} "
    "selected_pose_unchanged=1"
)
with Path(sys.argv[2]).open(encoding="utf8", newline="") as handle:
    folds = list(csv.DictReader(handle))
print("V0 Feature-PnP fold decision")
for row in folds:
    print(
        f"  fold={row['fold']} landmark={row['landmark_source']} "
        f"history={row['history_scope']} support={row['support_method']} "
        f"active={row['active_frames']}/4 "
        f"equal_count={row['equal_count_feasible_frames']}/4 "
        f"center_gain={float(row['center_gain_percent']):.6g}% "
        f"center_worse={row['center_worse_frames']} "
        f"rotation_gain={float(row['rotation_gain_percent']):.6g}% "
        f"pass={row['fold_pass']}"
    )
with Path(sys.argv[3]).open(encoding="utf8", newline="") as handle:
    frames = list(csv.DictReader(handle))
grouped = defaultdict(list)
for row in frames:
    grouped[row["method"]].append(row)
print("V0 Feature-PnP optimization audit")
primary = summary["primary_branch_locked_before_scoring"]
for method, rows in grouped.items():
    reasons = Counter(row["reason"] for row in rows)
    active = sum(int(row["optimized"]) for row in rows)
    feasible = sum(int(row["equal_count_feasible"]) for row in rows)
    print(
        f"  method={method} active={active}/{len(rows)} "
        f"equal_count={feasible}/{len(rows)} reasons={dict(reasons)}"
    )
    if method != primary:
        continue
    print("    locked-primary per-frame diagnostics")
    for row in rows:
        print(
            "    "
            f"frame={row['frame_index']} "
            f"pool={row['pooled_unique_matches']} "
            f"locked={row['locked_equal_count']} "
            f"anchor_epe={row['landmark_anchor_reprojection_mean_epe_pixels']} "
            f"inliers={row['inliers']} "
            f"rmse={row['reprojection_rmse_pixels']} "
            f"rotation_update_deg={row['rotation_update_degrees']} "
            f"center_update={row['center_update_native']} "
            f"reason={row['reason']}"
        )
PY

echo "Send the V0 Feature-PnP fold decision and optimization audit for review"
