#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${BASELINE_STREAM_GPU0:-1}"
STREAM_GPU1="${BASELINE_STREAM_GPU1:-2}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1"
CONFIG=streaming_couping/configs/v0_baseline.yaml
CACHE=outputs/streaming_couping_v0/cache/00a231a370_90_525_step15_37_68_54.pt
OUTPUT=outputs/streaming_couping_v0/clean_qk_pose_retrieval

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
[[ -s "$CACHE" ]] || {
  echo "missing retained V0 cache: $CACHE" >&2
  echo "run streaming_couping/commands_v0_baseline.txt first" >&2
  exit 1
}

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

rm -rf "$OUTPUT"
echo "V0 clean training-free StreamVGGT QK pose retrieval"
echo "physical GPUs=$VISIBLE_GPUS logical StreamVGGT=cuda:0,cuda:1"
echo "Candidate reads RGB only: no SAM mask/ID/memory, no GT and no point head"
echo "History policy: first-frame anchor + native QK Top-4"
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_v0_qk_pose_retrieval \
  --config "$CONFIG" \
  --streamvggt-devices cuda:0,cuda:1

[[ -s "$OUTPUT/clean_qk_pose_summary.json" ]] || {
  echo "missing clean_qk_pose_summary.json" >&2
  exit 1
}
[[ -s "$OUTPUT/clean_qk_pose_output.pt" ]] || {
  echo "missing clean_qk_pose_output.pt" >&2
  exit 1
}
[[ -s "$OUTPUT/copyable_result.txt" ]] || {
  echo "missing clean QK copyable_result.txt" >&2
  exit 1
}

PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path
path = Path("outputs/streaming_couping_v0/clean_qk_pose_retrieval/clean_qk_pose_summary.json")
summary = json.loads(path.read_text())
assert summary["revision"] == "v0_clean_streamvggt_qk_pose_retrieval_r1"
assert summary["stream_frames"] == 30
assert summary["evaluated_pose_frames"] == 29
assert summary["candidate_generation_fields"] == ["stream_images", "frame_indices"]
assert summary["candidate_generation_gt_fields"] == 0
assert summary["sam_pose_inputs"] == 0
assert summary["model_trained"] == 0
assert summary["point_head_run"] == 0
print("V0 clean QK pose schema passed")
'

echo ""
echo "===== V0 CLEAN QK POSE COPYABLE RESULT ====="
sed -n '1,160p' "$OUTPUT/copyable_result.txt"
