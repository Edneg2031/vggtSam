#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPUS="${MULTISCENE_GEOMETRY_GPUS:-1,2}"
CONFIG="${MULTISCENE_GEOMETRY_CONFIG:-streaming_couping/configs/multiscene_geometry.yaml}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
mkdir -p "$STORAGE_ROOT/outputs"

echo "Multi-scene frozen geometry cache preparation"
echo "physical GPUs=$GPUS logical devices=cuda:0,cuda:1"
echo "SAM=0 training=0 GT=opened only after frozen candidate generation"
CUDA_VISIBLE_DEVICES="$GPUS" PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT:$ROOT/src" \
  "$PYTHON_BIN" -m streaming_couping.scripts.prepare_multiscene_geometry_cache \
  --config "$CONFIG" \
  "$@"
