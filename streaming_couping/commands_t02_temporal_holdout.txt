#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${T02_STREAM_GPU0:-1}"
STREAM_GPU1="${T02_STREAM_GPU1:-2}"
SAM_GPU="${T02_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
V0_CONFIG=streaming_couping/configs/v0_t02_temporal_holdout.yaml
T0_CONFIG=streaming_couping/configs/t0_t02_temporal_holdout.yaml
TEMP_ROOT=outputs/streaming_couping_t02_holdout_tmp
T0_OUTPUT=outputs/streaming_couping_t02_temporal_holdout_anchor_probe

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

cleanup_t02_temporary_geometry() {
  rm -rf "$TEMP_ROOT"
}
trap cleanup_t02_temporary_geometry EXIT

echo "T0.2 same-scene unseen-time confirmation: frames 533..591 step 2"
echo "Formal outputs/streaming_couping_v0 is read-only and never removed"
echo "Temporary holdout cache/QK will be deleted when this command exits"

zsh streaming_couping/commands_t02_sequence_planner.txt
PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path

summary = json.loads(Path(
    "outputs/streaming_couping_t02_sequence_planner/summary.json"
).read_text())
expected = list(range(533, 592, 2))
assert summary["planner_ready"] == 1, summary["diagnostics"]
assert summary["recommended_scene_id"] == "00a231a370"
assert summary["recommended_clip_name"] == "00a231a370_533_591_step2_t02_holdout"
assert summary["recommended_frame_indices"] == expected
print("T0.2 holdout preflight accepted: 30 later frames, all required files present")
'

SOURCE_READY=0
if [[ "${T02_REBUILD_ANCHORS:-0}" != "1" \
      && -s "$T0_OUTPUT/anchors_candidate.pt" \
      && -s "$T0_OUTPUT/anchor_scores.csv" \
      && -s "$T0_OUTPUT/summary.json" ]]; then
  if PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path
import torch

root = Path("outputs/streaming_couping_t02_temporal_holdout_anchor_probe")
summary = json.loads((root / "summary.json").read_text())
candidate = torch.load(
    root / "anchors_candidate.pt", map_location="cpu", weights_only=False
)
assert summary["revision"] == "t0_sam_indexed_independent_3d_anchor_probe_r1"
assert summary["scene_id"] == candidate["scene_id"] == "00a231a370"
assert summary["clip"] == candidate["clip_name"] == "00a231a370_533_591_step2_t02_holdout"
assert summary["frames"] == list(range(533, 592, 2))
assert list(candidate["frame_indices"]) == list(range(533, 592, 2))
assert summary["candidate_generation_gt_fields"] == 0
assert summary["candidate_generation_raw_pointmap_fields"] == 0
'; then
    SOURCE_READY=1
  fi
fi

if [[ "$SOURCE_READY" == "1" ]]; then
  echo "reuse frozen temporal-holdout T0 anchor artifacts"
else
  rm -rf "$TEMP_ROOT" "$T0_OUTPUT"
  echo "stage 1/3: build isolated holdout StreamVGGT/SAM cache"
  echo "physical GPUs=$VISIBLE_GPUS logical StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
  CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
    --config "$V0_CONFIG" \
    --stage cache \
    --streamvggt-devices cuda:0,cuda:1 \
    --sam3-device cuda:2 \
    --geometry-device cuda:0

  echo "stage 2/3: run the frozen first-anchor + QK Top-4 camera policy"
  echo "No SAM or GT enters QK candidate generation"
  CUDA_VISIBLE_DEVICES="$STREAM_GPU0,$STREAM_GPU1" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_v0_qk_pose_retrieval \
    --config "$V0_CONFIG" \
    --streamvggt-devices cuda:0,cuda:1

  echo "stage 3/3: freeze T0 anchors, then score them against raw/GT"
  PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_t0_sam_indexed_anchor_probe
  CUDA_VISIBLE_DEVICES="$STREAM_GPU0" PYTHONUNBUFFERED=1 PYTHONPATH=. \
    "$PYTHON_BIN" -m streaming_couping.scripts.run_t0_sam_indexed_anchor_probe \
    --config "$T0_CONFIG" \
    --device cuda:0

  [[ -s "$T0_OUTPUT/anchors_candidate.pt" ]] || { echo "missing holdout T0 candidates" >&2; exit 1; }
  [[ -s "$T0_OUTPUT/anchor_scores.csv" ]] || { echo "missing holdout T0 scores" >&2; exit 1; }
  [[ -s "$T0_OUTPUT/summary.json" ]] || { echo "missing holdout T0 summary" >&2; exit 1; }
  PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path
import torch

root = Path("outputs/streaming_couping_t02_temporal_holdout_anchor_probe")
summary = json.loads((root / "summary.json").read_text())
candidate = torch.load(root / "anchors_candidate.pt", map_location="cpu", weights_only=False)
expected = list(range(533, 592, 2))
assert summary["scene_id"] == candidate["scene_id"] == "00a231a370"
assert summary["clip"] == candidate["clip_name"] == "00a231a370_533_591_step2_t02_holdout"
assert summary["frames"] == expected
assert list(candidate["frame_indices"]) == expected
assert summary["candidate_generation_gt_fields"] == 0
assert summary["candidate_generation_raw_pointmap_fields"] == 0
print("T0.2 source anchors accepted; temporary dense cache can now be deleted")
'
  cleanup_t02_temporary_geometry
fi

echo "final stage: apply the frozen condition<100 and views>=4 gate"
zsh streaming_couping/commands_t02_multiview_confirmation.txt
echo "T0.2 completed; retained only planner, sparse-anchor and confirmation outputs"
