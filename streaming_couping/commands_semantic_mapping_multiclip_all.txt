#!/usr/bin/env zsh
set -euo pipefail

# One-command multi-scene V2.3 validation:
# plan -> frozen cache -> V0/QK audit -> V2.1 -> V2.2 -> V2.3 -> evaluation.
# Override SEMANTIC_MAPPING_SCENES to use a different scene set.

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
SCENES="${SEMANTIC_MAPPING_SCENES:-00a231a370,0a184cf634,1a8e0d78c0,1eacc65607}"
RUN_ROOT="${SEMANTIC_MAPPING_MULTICLIP_RUN_ROOT:-$STORAGE_ROOT/outputs/streaming_couping_multiclip}"
CONFIG="$RUN_ROOT/protocol.yaml"
EVAL_OUTPUT="${SEMANTIC_MAPPING_ABLATION_OUTPUT:-$STORAGE_ROOT/outputs/streaming_couping_semantic_map_multiclip_ablation}"
STREAM_GPU0="${SEMANTIC_MAPPING_STREAM_GPU0:-1}"
STREAM_GPU1="${SEMANTIC_MAPPING_STREAM_GPU1:-2}"
SAM_GPU="${SEMANTIC_MAPPING_SAM_GPU:-3}"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export SEMANTIC_MAPPING_MULTICLIP_CONFIG="$CONFIG"
export SEMANTIC_MAPPING_MULTICLIP_RUN_ROOT="$RUN_ROOT"
export SEMANTIC_MAPPING_STREAM_GPU0="$STREAM_GPU0"
export SEMANTIC_MAPPING_STREAM_GPU1="$STREAM_GPU1"
export SEMANTIC_MAPPING_SAM_GPU="$SAM_GPU"

echo "===== MULTI-SCENE V2.3 FULL PIPELINE ====="
echo "scenes=$SCENES"
echo "run_root=$RUN_ROOT"
echo "physical_gpus=$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"

echo "[1/3] plan deterministic multi-scene protocol"
SEMANTIC_MAPPING_SCENES="$SCENES" \
  zsh streaming_couping/commands_plan_semantic_mapping_multiclip.txt

echo "[2/3] build caches, run baselines and export V2.1/V2.2/V2.3"
SEMANTIC_MAPPING_MULTICLIP_STAGE=all \
  zsh streaming_couping/commands_run_semantic_mapping_multiclip.txt

echo "[3/3] evaluate all frozen branches"
SEMANTIC_MAPPING_MULTICLIP_CONFIG="$CONFIG" \
SEMANTIC_MAPPING_MULTICLIP_RUN_ROOT="$RUN_ROOT" \
SEMANTIC_MAPPING_ABLATION_OUTPUT="$EVAL_OUTPUT" \
  zsh streaming_couping/commands_semantic_mapping_multiclip_ablation.txt

SUMMARY="$EVAL_OUTPUT/semantic_map_multiclip_ablation_summary.json"
[[ -s "$SUMMARY" ]] || {
  echo "missing final multi-clip summary: $SUMMARY" >&2
  exit 1
}
echo "===== MULTI-SCENE V2.3 FULL PIPELINE COMPLETED ====="
echo "summary=$SUMMARY"
