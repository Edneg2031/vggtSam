#!/usr/bin/env zsh
set -euo pipefail

# Stage-2b attribution post-mortem for the object-consensus pose-feedback run.
#
#   zsh streaming_couping/commands_analyze_object_pose_feedback_attribution.txt
#
# Read-only unless the CSVs predate a column the analysis needs.  It reads the
# artifacts stage 2b already wrote and prints the attribution sections: no
# model runs, no GPU, seconds.  Answers where the next iteration should go --
# whether consensus collapsed, whether the reliability scores actually predict
# proposal correctness, which frames the gate rejects, and whether the proposal
# error tracks how stale the reference set is.
#
# Stage 2b is regenerated automatically (CPU only, deterministic replay) when
# object_proposals.csv is missing a column the analysis reads -- today that is
# reference_roles, needed to split the anchor reference from the local history
# one.  The existing output is backed up first and existing backups are never
# overwritten; the decisions in summary.json are diffed afterwards, because
# recording extra diagnostics must not move any number.
#
# RERUN_STAGE_2B=0|1 forces that decision off or on.  No other input is needed.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON="${STREAMING_COUPING_PYTHON:-$(command -v python)}"

RUNS_ROOT="$STORAGE_ROOT/outputs"
# The baseline branch of the newest generation that has one.  Resolved rather
# than fixed so this keeps working across generations -- and deterministically,
# because a sweep leaves several branch directories behind and "which one did
# it pick" must not depend on the order a glob happens to come back in.
DEFAULT_RUN_DIR=""
for generation in v2 v1; do
  candidate="$RUNS_ROOT/semantic_map_100frames_horizonstream_object_pose_feedback_90_189_$generation.baseline"
  if [[ -d "$candidate/object_pose_feedback" ]]; then
    DEFAULT_RUN_DIR="$candidate"
    break
  fi
done

RUN_DIR="${OBJECT_POSE_FEEDBACK_RUN_DIR:-$DEFAULT_RUN_DIR}"

# Fall back to a search, and on several candidates take them newest-name-first
# (`On`) rather than making the caller disambiguate -- this script is meant to
# be runnable on its own with no arguments, and the generation tag is at the
# end of the name, so descending name order is descending generation.
if [[ -z "$RUN_DIR" || ! -d "$RUN_DIR/object_pose_feedback" ]]; then
  candidates=("$STORAGE_ROOT"/outputs/*/object_pose_feedback(N/On))
  if (( ${#candidates} > 0 )); then
    RUN_DIR="${candidates[1]:h}"
    print "note: no baseline run for a known generation; using"
    print "      $RUN_DIR"
    print "      (${#candidates} candidate(s), newest name first)"
  else
    print -u2 "missing feedback dir: ${RUN_DIR:-<none>}/object_pose_feedback"
    print -u2 "searched: $STORAGE_ROOT/outputs/*/object_pose_feedback"
    print -u2 "run commands_run_scannet_object_pose_feedback_branches.txt first"
    exit 1
  fi
fi
print "attribution_run_dir=$RUN_DIR"

FEEDBACK_DIR="$RUN_DIR/object_pose_feedback"
# A branch run directory does not hold the geometry cache -- the sweep pins one
# cache per generation, in the un-suffixed base directory, so every branch
# shares a single geometry pass.  Look there before giving up on the run dir.
GEOMETRY_CACHE="$RUN_DIR/horizonstream_geometry.pt"
# Strip the branch suffix from the directory NAME only: the storage root itself
# may contain a dot, so trimming the first dot in the whole path would cut the
# wrong end off.
if [[ ! -f "$GEOMETRY_CACHE" ]]; then
  GEOMETRY_CACHE="${RUN_DIR:h}/${${RUN_DIR:t}%%.*}/horizonstream_geometry.pt"
fi
MANIFEST_PATH="${SEMANTIC_MAP_MANIFEST:-$STORAGE_ROOT/data/processed/scannetpp_pinhole_2d/manifest.json}"
SCENE_ID="${SEMANTIC_MAP_SCENE_ID:-00a231a370}"
HORIZON_REPO="$ROOT/externals/horizonstream"
RERUN_STAGE_2B="${RERUN_STAGE_2B:-auto}"

# These must mirror the stage-2a refiner thresholds; a mismatch is refused by
# the run itself.  See docs/object_pose_feedback.md.
MIN_CONSENSUS_OBJECTS="${OBJECT_POSE_FEEDBACK_MIN_CONSENSUS_OBJECTS:-2}"
MAX_CONSENSUS_TRANSLATION_M="${OBJECT_POSE_FEEDBACK_MAX_CONSENSUS_TRANSLATION_M:-0.20}"
MAX_CONSENSUS_ROTATION_DEG="${OBJECT_POSE_FEEDBACK_MAX_CONSENSUS_ROTATION_DEG:-8.0}"

cd "$ROOT"
export PYTHONUNBUFFERED=1

[[ -n "$PYTHON" ]] || { print -u2 "no python interpreter found"; exit 1; }
for name in object_proposals.csv consensus_metrics.csv frame_metrics.csv; do
  [[ -f "$FEEDBACK_DIR/$name" ]] || {
    print -u2 "missing stage-2b artifact: $FEEDBACK_DIR/$name"
    print -u2 "was stage 2b run with --object-pose-loss-export-feedback-diagnostics?"
    exit 1
  }
done

# Automatic: regenerate stage 2b only when a column the analysis reads is
# absent from the current CSVs.  Recording extra diagnostics never changes a
# computed value, so the replay is safe to repeat.
if [[ "$RERUN_STAGE_2B" == "auto" ]]; then
  missing_columns=()
  for column in reference_roles; do
    head -n 1 "$FEEDBACK_DIR/object_proposals.csv" | grep -qF "$column" \
      || missing_columns+=("$column")
  done
  if (( ${#missing_columns} > 0 )); then
    RERUN_STAGE_2B=1
    print "note: object_proposals.csv lacks ${(j:, :)missing_columns}; regenerating stage 2b to record it"
  else
    RERUN_STAGE_2B=0
  fi
fi

print "===== OBJECT POSE FEEDBACK ATTRIBUTION ====="
print "feedback_dir=$FEEDBACK_DIR"
print "rerun_stage_2b=$RERUN_STAGE_2B"

BACKUP="$FEEDBACK_DIR.bak"

if [[ "$RERUN_STAGE_2B" == "1" ]]; then
  for required in "$GEOMETRY_CACHE" "$MANIFEST_PATH"; do
    [[ -f "$required" ]] || { print -u2 "missing: $required"; exit 1; }
  done
  if [[ -e "$BACKUP" ]]; then
    print -u2 "refusing to overwrite an existing backup: $BACKUP"
    print -u2 "move it aside first if you really want a fresh backup."
    exit 1
  fi
  cp -r "$FEEDBACK_DIR" "$BACKUP"
  print "backup=$BACKUP"

  PYTHONPATH="$HORIZON_REPO:$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.run_object_pose_feedback \
    --run-dir "$RUN_DIR" \
    --geometry-cache "$GEOMETRY_CACHE" \
    --manifest "$MANIFEST_PATH" \
    --scene-id "$SCENE_ID" \
    --output-dir "$FEEDBACK_DIR" \
    --horizon-repo "$HORIZON_REPO" \
    --min-consensus-objects "$MIN_CONSENSUS_OBJECTS" \
    --max-consensus-translation-m "$MAX_CONSENSUS_TRANSLATION_M" \
    --max-consensus-rotation-deg "$MAX_CONSENSUS_ROTATION_DEG"
fi

# Report against the backup whenever one exists, not only on the run that made
# it: the comparison answers "did the number move?", which stays a fair
# question on every later run.  Recording extra diagnostics must not move any
# number; a newly recorded informational field is not the same finding as a
# flipped decision, so the differing paths are named rather than just counted.
# Warn, do not abort -- the attribution is still worth reading when it trips.
if [[ -f "$BACKUP/summary.json" && -f "$FEEDBACK_DIR/summary.json" ]]; then
  print "decisions vs the backup: $BACKUP"
  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.compare_json \
    --prefix decisions \
    "$BACKUP/summary.json" "$FEEDBACK_DIR/summary.json" \
    || print -u2 "WARNING: decisions moved; check the paths above before trusting the new (c2) section"
fi

PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
  "$PYTHON" -m streaming_couping.scripts.analyze_object_pose_feedback_attribution \
  --feedback-dir "$FEEDBACK_DIR"
