#!/usr/bin/env zsh

# Why did a prompt-set change move the result?  Which objects did it swap?
#
#   zsh streaming_couping/commands_compare_prompt_sets.txt
#
# CPU only, seconds, read-only: it reads what stage 2b already wrote.  No model
# runs, no GPU.
#
# The branch table answers "did it get better or worse".  For a prompt-set
# change that is not the useful question, because the prompt list is NOT
# additive -- asking for more objects changes which objects the segmenter
# tracks at all, including ones it tracked before.  STDOUT carries only the
# headline; the per-category detail goes to <prefix>.prompt_categories_<branch>.txt,
# because four wide sections per branch is more than anyone reads on every run.
#
# The report has, per category:
#
#   * proposals contributed and the GT error they carry, per generation;
#   * which categories only one generation has (the swap);
#   * how many frames each category actually entered the consensus in, and its
#     share of inlier votes -- an object that proposes but is never an inlier
#     changed nothing, and looks identical from the branch table;
#   * whether combining objects beat a single proposal, per run.
#
# Ground truth is read only to score what the run already produced; no decision
# in the pipeline sees it.
set -uo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON="${STREAMING_COUPING_PYTHON:-$(command -v python)}"

RUNS_ROOT="$STORAGE_ROOT/outputs"
PREFIX="semantic_map_100frames_horizonstream_object_pose_feedback_90_189"

# Every generation that has been run, oldest first.  Each is a prompt set; a
# generation with no readable run is skipped, so listing one is harmless.
GENERATIONS=(v1 v2 v3)

# The pairs to read.  The baselines first: they differ only in the prompt set,
# so that pair is the headline.  `fresh` follows because it is the only branch
# that recovered a positive gain under the wider prompt set.
PAIRS=(baseline fresh)

cd "$ROOT"
export VGGT_SAM_STORAGE_ROOT="$STORAGE_ROOT"
export PYTHONUNBUFFERED=1

found=0
for branch in "${PAIRS[@]}"; do
  run_dirs=()
  for generation in "${GENERATIONS[@]}"; do
    candidate="$RUNS_ROOT/${PREFIX}_$generation.$branch"
    [[ -f "$candidate/object_pose_feedback/object_proposals.csv" ]] \
      && run_dirs+=("$candidate")
  done
  if (( ${#run_dirs} < 2 )); then
    print -u2 "skipping branch=$branch: fewer than two generations have it"
    continue
  fi
  found=$(( found + 1 ))
  args=()
  for run_dir in "${run_dirs[@]}"; do
    args+=(--run-dir "$run_dir")
  done
  print ""
  print "########## branch=$branch ##########"
  PYTHONPATH="$ROOT${PYTHONPATH:+:$PYTHONPATH}" \
    "$PYTHON" -m streaming_couping.scripts.compare_feedback_categories \
    "${args[@]}" \
    --report-out "$RUNS_ROOT/${PREFIX}.prompt_categories_$branch.txt" \
    --json-out "$RUNS_ROOT/${PREFIX}.prompt_categories_$branch.json" \
    || print -u2 "comparison failed for branch=$branch"
done

if (( found == 0 )); then
  print -u2 ""
  print -u2 "nothing to compare: neither generation has a readable run."
  print -u2 "run commands_run_scannet_object_pose_feedback_branches.txt first."
  exit 1
fi
