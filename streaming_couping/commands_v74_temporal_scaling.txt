#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V74_GPU:-1}"
STEPS="${V74_STEPS:-3000}"
SEED="${V74_SEED:-0}"
OUTPUT="${V74_OUTPUT:-outputs/streaming_couping_v74_temporal_scaling}"
DEFAULT_BRANCHES=uniform_transport,geometry_transport,sam_transport
DEFAULT_BRANCHES+=,sam_geometry_transport,sam_geometry_train_sam_off
BRANCHES="${V74_BRANCHES:-$DEFAULT_BRANCHES}"
LONG_CACHE=outputs/streaming_couping_v7_fusion_ablation/cache/00a231a370_90_525_step15_37_68_54.pt

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }
FROZEN_ARGS=()
if [[ -n "${V74_FROZEN_L0:-}" ]]; then
  [[ -s "$V74_FROZEN_L0" ]] || {
    echo "missing explicit V74_FROZEN_L0: $V74_FROZEN_L0" >&2
    exit 1
  }
  FROZEN_ARGS=(--frozen-l0 "$V74_FROZEN_L0")
  echo "V7.4 explicit frozen L0=$V74_FROZEN_L0"
else
  FOUND_FROZEN_L0=0
  for candidate in \
    outputs/streaming_couping_v71_instance_causality/frozen_l0.pt \
    "$OUTPUT/frozen_l0.pt" \
    outputs/streaming_couping_v73_correspondence/frozen_l0.pt \
    outputs/streaming_couping_v72_local_token_ablation/frozen_l0.pt \
    outputs/streaming_couping_v73_long_capacity/geometry_transport_k08.pt \
    outputs/streaming_couping_v73_long_capacity/sam_geometry_transport_k08.pt
  do
    if [[ -s "$candidate" ]]; then
      FOUND_FROZEN_L0=1
    fi
  done
  [[ "$FOUND_FROZEN_L0" == "1" ]] || {
    echo "V7.4 could not find a clean L0 or signed V7.3 long checkpoint." >&2
    echo "Run commands_v71_instance_causality.txt only if it was deleted." >&2
    exit 1
  }
  echo "V7.4 will auto-select a provenance-compatible V7.1/V7.3/V7.2 L0"
fi
[[ -s "$LONG_CACHE" ]] || {
  echo "V7.4 requires the augmented V7.2 long cache: $LONG_CACHE" >&2
  exit 1
}

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V7.4 temporal scaling: protocol and tensor checks"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v74_protocol
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v73_runtime

RESUME_ARGS=(--resume)
if [[ "${V74_FRESH:-0}" == "1" ]]; then
  RESUME_ARGS=()
fi

echo "V7.4 GPU=$GPU K=8 steps=$STEPS seed=$SEED"
echo "Folds: 270-345->360-405; 270-405->420-465; 270-465->480-525"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v74_temporal_scaling \
  --config streaming_couping/configs/v73_correspondence_ablation.yaml \
  --output-dir "$OUTPUT" \
  --device cuda:0 \
  --token-count 8 \
  --steps "$STEPS" \
  --seed "$SEED" \
  --branches "$BRANCHES" \
  "${FROZEN_ARGS[@]}" \
  "${RESUME_ARGS[@]}"

[[ -s "$OUTPUT/v74_temporal_scaling.csv" ]] || {
  echo "missing V7.4 temporal-scaling CSV" >&2
  exit 1
}
echo "V7.4 COPY THIS CSV"
cat "$OUTPUT/v74_temporal_scaling.csv"
