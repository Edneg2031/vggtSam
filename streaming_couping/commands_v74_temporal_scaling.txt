#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V74_GPU:-1}"
STREAM_GPU0="${V74_STREAM_GPU0:-$GPU}"
STREAM_GPU1="${V74_STREAM_GPU1:-2}"
SAM_GPU="${V74_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
STEPS="${V74_STEPS:-3000}"
BASE_STEPS="${V74_BASE_STEPS:-1200}"
SEED="${V74_SEED:-0}"
OUTPUT="${V74_OUTPUT:-outputs/streaming_couping_v74_temporal_scaling}"
DATA_CONFIG=streaming_couping/configs/v74_temporal_data.yaml
CACHE_ROOT=outputs/streaming_couping_v74_temporal_scaling
DEFAULT_BRANCHES=uniform_transport,geometry_transport,sam_transport
DEFAULT_BRANCHES+=,sam_geometry_transport,sam_geometry_train_sam_off
BRANCHES="${V74_BRANCHES:-$DEFAULT_BRANCHES}"
LONG_CACHE="$CACHE_ROOT/cache/00a231a370_90_525_step15_37_68_54.pt"

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V7.4 dynamic-instance temporal scaling: protocol and tensor checks"
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v74_protocol
PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_v73_runtime

RESUME_ARGS=(--resume)
REBUILD_ARGS=()
if [[ "${V74_FRESH:-0}" == "1" ]]; then
  RESUME_ARGS=()
  REBUILD_ARGS=(--rebuild-cache)
fi

echo "V7.4 standalone cache physical GPUs=$VISIBLE_GPUS"
echo "V7.4 logical layout: StreamVGGT=cuda:0,cuda:1 SAM3.1=cuda:2"
echo "V7.4 instances: SAM3.1 forward discovery, 8 persistent slots, no fixed object reference"
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" \
PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_instance_token_pose \
  --config "$DATA_CONFIG" \
  --stage cache \
  --streamvggt-devices cuda:0,cuda:1 \
  --sam3-device cuda:2 \
  --geometry-device cuda:0 \
  --training-device cuda:0 \
  "${REBUILD_ARGS[@]}"

[[ -s "$LONG_CACHE" ]] || {
  echo "V7.4 standalone cache was not created: $LONG_CACHE" >&2
  exit 1
}
PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.audit_v7_cache \
  --config "$DATA_CONFIG" \
  --output-dir "$OUTPUT/cache_audit" \
  --require-local \
  --strict

echo "V7.4 training GPU=$GPU K=8 base_steps=$BASE_STEPS steps=$STEPS seed=$SEED"
echo "Folds: 270-345->360-405; 270-405->420-465; 270-465->480-525"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_v74_temporal_scaling \
  --config streaming_couping/configs/v73_correspondence_ablation.yaml \
  --data-config "$DATA_CONFIG" \
  --output-dir "$OUTPUT" \
  --device cuda:0 \
  --token-count 8 \
  --base-steps "$BASE_STEPS" \
  --steps "$STEPS" \
  --seed "$SEED" \
  --branches "$BRANCHES" \
  "${RESUME_ARGS[@]}"

[[ -s "$OUTPUT/v74_temporal_scaling.csv" ]] || {
  echo "missing V7.4 temporal-scaling CSV" >&2
  exit 1
}
[[ -s "$OUTPUT/v74_dynamic_instance_diagnostics.csv" ]] || {
  echo "missing V7.4 dynamic-instance diagnostics CSV" >&2
  exit 1
}
echo "V7.4 COPY THIS CSV"
cat "$OUTPUT/v74_temporal_scaling.csv"
echo "V7.4 COPY THIS INSTANCE DIAGNOSTIC IF TRACK COVERAGE LOOKS WRONG"
cat "$OUTPUT/v74_dynamic_instance_diagnostics.csv"
