set -euo pipefail

CONFIG=streaming_couping/configs/v4_coverage_first.yaml
OUTPUT=outputs/streaming_couping_v4_coverage_first
EXPORT="$OUTPUT/final_v4_coverage_first"
SOURCE_CHECKPOINT=outputs/streaming_couping_v5_ablation/variants/v4_match_additive_union/checkpoints/decoupled_dual_branch/checkpoint_best.pt
CHECKPOINT="$OUTPUT/checkpoints/decoupled_dual_branch/checkpoint_best.pt"

mkdir -p "$OUTPUT/checkpoints/decoupled_dual_branch"

# Reuse the common frozen observations.  If they are absent, build them once
# with the V4-compatible identity thresholds.
PYTHONUNBUFFERED=1 PYTHONPATH=src:. python -m streaming_couping.scripts.run_instance_token_pose \
  --config "$CONFIG" \
  --stage cache \
  --sam3-device cuda:3 \
  --geometry-device cuda:1 \
  --training-device cuda:1 \
  2>&1 | tee "$OUTPUT/cache.log"

# Freeze the exact previously evaluated V4 checkpoint when available.  Once
# copied, later runs never overwrite the release checkpoint.
if [[ ! -f "$CHECKPOINT" ]]; then
  if [[ -f "$SOURCE_CHECKPOINT" ]]; then
    cp -p "$SOURCE_CHECKPOINT" "$CHECKPOINT"
    echo "preserved existing V4 checkpoint: $SOURCE_CHECKPOINT"
  else
    echo "existing V4 checkpoint not found; training the standalone V4 release"
    PYTHONUNBUFFERED=1 PYTHONPATH=src:. python -m streaming_couping.scripts.run_instance_token_pose \
      --config "$CONFIG" \
      --stage train \
      --training-device cuda:1 \
      2>&1 | tee "$OUTPUT/train.log"
  fi
fi

rm -rf "$OUTPUT/evaluation" "$EXPORT"

PYTHONUNBUFFERED=1 PYTHONPATH=src:. python -m streaming_couping.scripts.run_instance_token_pose \
  --config "$CONFIG" \
  --stage eval \
  --geometry-device cuda:1 \
  --training-device cuda:1 \
  2>&1 | tee "$OUTPUT/eval.log"

PYTHONPATH=src:. python -m streaming_couping.scripts.run_instance_token_pose \
  --config "$CONFIG" \
  --stage export \
  --export-output-dir "$EXPORT"

PYTHONPATH=src:. python -m streaming_couping.scripts.plot_pose_comparison \
  --comparison-dir \
  "$EXPORT/00a231a370_90_240_37_68_54/comparison_gt_world" \
  --heldout-frame-indices 210 240

PYTHONPATH=src:. python -m streaming_couping.scripts.plot_pose_comparison \
  --comparison-dir \
  "$EXPORT/00a231a370_492_589_37_68_54/comparison_gt_world"

find "$EXPORT" -type f | sort > "$OUTPUT/v4_coverage_first_artifact_manifest.txt"
sha256sum \
  "$CONFIG" \
  "$CHECKPOINT" \
  streaming_couping/commands_v4_coverage_first.txt \
  streaming_couping/src/learned_pose/pipeline.py \
  streaming_couping/src/learned_pose/ray_pose.py \
  streaming_couping/src/learned_pose/export.py \
  > "$OUTPUT/v4_coverage_first_reproduction_inputs.sha256"

sed -n '1,12p' "$OUTPUT/evaluation/ray_pose_compact_summary.csv"
