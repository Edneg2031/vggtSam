#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${MULTISCENE_DIRECT_GPU:-1}"
CONFIG="${MULTISCENE_DIRECT_CONFIG:-streaming_couping/configs/multiscene_geometry_direct_train.yaml}"

echo "Stage 1 Fold1 multi-scene Direct Geometry Decoder"
echo "physical GPU=$GPU logical device=cuda:0"
echo "StreamVGGT aggregator/backbone/CameraHead frozen; original PointHead trainable; SAM=0"
CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH="$ROOT:$ROOT/src" \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_multiscene_geometry_direct_training \
  --config "$CONFIG" \
  "$@"
