#!/usr/bin/env zsh
set -euo pipefail

SCRIPT_PATH="${0:A}"
SCRIPT_DIR="${SCRIPT_PATH:h}"
ROOT="${VGGT_SAM_ROOT:-${SCRIPT_DIR:h}}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-}"
CONFIG="${E1_CONFIG:-streaming_couping/configs/e1_edge_directional_gt_audit.yaml}"
OUTPUT_DIR="${E1_OUTPUT_DIR:-outputs/streaming_couping_e1_edge_directional_gt_audit}"

cd "$ROOT"

if [[ -z "$PYTHON_BIN" ]]; then
  if command -v python3 >/dev/null 2>&1; then
    PYTHON_BIN="$(command -v python3)"
  elif command -v python >/dev/null 2>&1; then
    PYTHON_BIN="$(command -v python)"
  else
    echo "missing Python: set STREAMING_COUPING_PYTHON=/path/to/python" >&2
    exit 1
  fi
fi
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "E1 fixed-step edge directional GT-scoring audit"
echo "GT is unavailable to candidate generation and cannot select a branch or step"
echo "V0 selected pose remains raw StreamVGGT"

PYTHONPATH=. "$PYTHON_BIN" -m streaming_couping.scripts.smoke_e1_edge_directional_gt_audit

ARGS=()
[[ "${E1_REBUILD_CACHE:-0}" == "1" ]] && ARGS+=(--rebuild-cache)
[[ -n "${E1_DEVICE:-}" ]] && ARGS+=(--device "$E1_DEVICE")

PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.run_e1_edge_directional_gt_audit \
  --config "$CONFIG" \
  --stage all \
  --output-dir "$OUTPUT_DIR" \
  "${ARGS[@]}"

REPORT="$OUTPUT_DIR/copyable_result.txt"
[[ -s "$REPORT" ]] || { echo "missing E1 copyable report: $REPORT" >&2; exit 1; }
[[ -s "$OUTPUT_DIR/directional_frame_diagnostics.csv" ]] || { echo "missing E1 frame CSV" >&2; exit 1; }
[[ ! -e "$OUTPUT_DIR/poses.pt" ]] || { echo "E1 must not write poses.pt" >&2; exit 1; }

echo ""
echo "E1 result saved for copying:"
echo "$REPORT"
echo ""
echo "Copy it with:"
echo "cat $REPORT"
