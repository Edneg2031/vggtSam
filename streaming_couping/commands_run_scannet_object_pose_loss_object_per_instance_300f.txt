#!/usr/bin/env zsh
set -euo pipefail

# Per-instance object-only loss experiment on the first 300 frames
# (frame positions 0..299). HorizonStream camera poses and background
# geometry remain raw; each accepted SAM instance receives its own 6DoF
# object-point correction.
ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
# Use a fresh artifact directory because the previous 300-frame attempt could
# have stopped after writing maps but before GT evaluation completed.
OUTPUT_DIR="${OBJECT_POSE_LOSS_OBJECT_PER_INSTANCE_300F_OUTPUT_DIR:-$STORAGE_ROOT/outputs/semantic_map_300frames_horizonstream_object_pose_loss_object_per_instance_v2}"

export OBJECT_POSE_LOSS_OBJECT_ONLY_OUTPUT_DIR="$OUTPUT_DIR"
export OBJECT_POSE_LOSS_OBJECT_ONLY_INDEPENDENT_INSTANCES=1
export OBJECT_POSE_ALIGNED_BRANCH=object_pose_object_only_per_instance
export OBJECT_POSE_LOSS_FRAME_START=0
export OBJECT_POSE_LOSS_FRAME_STRIDE=1
export OBJECT_POSE_LOSS_FRAME_COUNT=300
export HORIZONSTREAM_GEOMETRY_CACHE="${HORIZONSTREAM_GEOMETRY_CACHE:-$STORAGE_ROOT/outputs/semantic_map_300frames_horizonstream_consecutive/horizonstream_geometry.pt}"

exec zsh "$ROOT/streaming_couping/commands_run_scannet_object_pose_loss_object_only_100f.txt"
