#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
GPU="${V73_GPU:-1}"
SEEDS_TEXT="${V73_SEEDS:-0 1 2}"
CONFIG=streaming_couping/configs/v73_correspondence_ablation.yaml
OUTPUT_ROOT=outputs/streaming_couping_v73_multiseed

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

seed_values=(${=SEEDS_TEXT})
(( ${#seed_values[@]} > 0 )) || { echo "V73_SEEDS is empty" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

for seed in "${seed_values[@]}"; do
  output="$OUTPUT_ROOT/seed_$seed"
  echo "V7.3 multi-seed training seed=$seed"
  CUDA_VISIBLE_DEVICES="$GPU" PYTHONUNBUFFERED=1 PYTHONPATH=. "$PYTHON_BIN" \
    -m streaming_couping.scripts.run_v73_correspondence_ablation \
    --config "$CONFIG" \
    --device cuda:0 \
    --seed "$seed" \
    --output-dir "$output" \
    --resume
done

PYTHONPATH=. "$PYTHON_BIN" \
  -m streaming_couping.scripts.aggregate_v73_seeds \
  --root "$OUTPUT_ROOT" \
  --seeds "$SEEDS_TEXT" \
  --output "$OUTPUT_ROOT/v73_seed_aggregate.csv"

echo "V7.3 COPY THIS MULTI-SEED CSV"
cat "$OUTPUT_ROOT/v73_seed_aggregate.csv"
