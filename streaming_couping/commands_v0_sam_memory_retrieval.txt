#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
STREAM_GPU0="${BASELINE_STREAM_GPU0:-1}"
STREAM_GPU1="${BASELINE_STREAM_GPU1:-2}"
SAM_GPU="${BASELINE_SAM_GPU:-3}"
VISIBLE_GPUS="$STREAM_GPU0,$STREAM_GPU1,$SAM_GPU"
CONFIG=streaming_couping/configs/v0_baseline.yaml
CACHE=outputs/streaming_couping_v0/cache/00a231a370_90_525_step15_37_68_54.pt
OUTPUT=outputs/streaming_couping_v0/sam_memory_retrieval

cd "$ROOT"
[[ -x "$PYTHON_BIN" ]] || { echo "missing Python: $PYTHON_BIN" >&2; exit 1; }

export PYTHONHASHSEED=0
export CUBLAS_WORKSPACE_CONFIG=:4096:8
export PYTORCH_CUDA_ALLOC_CONF=expandable_segments:True

echo "V0 SAM-memory retrieval smoke: native QK, fixed budget and identity controls"
PYTHONPATH=. "$PYTHON_BIN" -m \
  streaming_couping.scripts.smoke_v0_sam_memory_retrieval

echo "V0 validating/building the retained causal SAM tracking cache"
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m streaming_couping.scripts.run_dynamic_instance_baseline \
  --config "$CONFIG" \
  --stage cache \
  --streamvggt-devices cuda:0,cuda:1 \
  --sam3-device cuda:2 \
  --geometry-device cuda:0

[[ -s "$CACHE" ]] || { echo "missing V0 cache: $CACHE" >&2; exit 1; }
rm -rf "$OUTPUT"

echo "V0 frozen RetrieveVGGT-style full-frame KV intervention"
echo "physical GPUs=$VISIBLE_GPUS logical StreamVGGT=cuda:0,cuda:1"
echo "SAM hidden features are not read; no model is trained"
echo "GT is introduced only after every frozen branch is generated"
CUDA_VISIBLE_DEVICES="$VISIBLE_GPUS" PYTHONUNBUFFERED=1 PYTHONPATH=. \
  "$PYTHON_BIN" -m \
  streaming_couping.scripts.run_v0_sam_memory_retrieval \
  --config "$CONFIG" \
  --streamvggt-devices cuda:0,cuda:1

[[ -s "$OUTPUT/retrieval_summary.json" ]] || {
  echo "missing retrieval_summary.json" >&2
  exit 1
}
[[ -s "$OUTPUT/copyable_result.txt" ]] || {
  echo "missing copyable_result.txt" >&2
  exit 1
}
PYTHONPATH=. "$PYTHON_BIN" -c '
import json
from pathlib import Path
path = Path("outputs/streaming_couping_v0/sam_memory_retrieval/retrieval_summary.json")
summary = json.loads(path.read_text())
assert summary["revision"] == "v0_sam_memory_retrievevggt_probe_r1"
assert summary["model_trained"] == 0
assert summary["sam_hidden_features_used"] == 0
assert summary["raw_replay_equivalence"]["pass"] == 1
assert summary["decision"]["candidate_generation_gt_fields"] == 0
assert summary["decision"]["selected_pose_modified"] == 0
print("V0 SAM-memory retrieval schema and raw equivalence passed")
'

echo ""
echo "===== V0 SAM-MEMORY RETRIEVAL COPYABLE RESULT ====="
sed -n '1,240p' "$OUTPUT/copyable_result.txt"
