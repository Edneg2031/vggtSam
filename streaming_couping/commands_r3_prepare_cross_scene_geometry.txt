#!/usr/bin/env zsh
set -euo pipefail

ROOT="${VGGT_SAM_ROOT:-/home/wlh50060092/vggtSam}"
STORAGE_ROOT="${VGGT_SAM_STORAGE_ROOT:-/data184/open_source/vggtSam}"
SOURCE_ROOT="${R3_SCANNETPP_SOURCE_ROOT:-$STORAGE_ROOT/source/scanet_pp_pinhole}"
SCENE_ID="${R3_SCENE_ID:-0a184cf634}"
PYTHON_BIN="${STREAMING_COUPING_PYTHON:-/home/huawei/miniconda3/envs/3am/bin/python}"
OUTPUT="${R3_GEOMETRY_OUTPUT:-$STORAGE_ROOT/data/processed/residual_cross_scene_r3}"
REPORT="$OUTPUT/copyable_result.txt"
LOG="$OUTPUT/run.log"

cd "$ROOT"
mkdir -p "$OUTPUT"

write_shell_failure() {
  local message="$1"
  {
    echo "===== COPYABLE_R3_GEOMETRY_PREPARE_BEGIN ====="
    echo "revision=r3_frozen_cross_scene_geometry_preprocess_r1"
    echo "status=FAILED"
    echo "error=$message"
    echo "run_log=$LOG"
    echo "copyable_report=$REPORT"
    echo "===== COPYABLE_R3_GEOMETRY_PREPARE_END ====="
  } > "$REPORT"
  sed -n '1,120p' "$REPORT"
}

if [[ ! -x "$PYTHON_BIN" ]]; then
  write_shell_failure "missing Python: $PYTHON_BIN"
  exit 1
fi

OVERWRITE_ARGS=()
if [[ "${R3_GEOMETRY_OVERWRITE:-0}" == "1" ]]; then
  OVERWRITE_ARGS+=(--overwrite)
fi

set +e
PYTHONPATH="$ROOT:$ROOT/src" "$PYTHON_BIN" \
  -m streaming_couping.scripts.prepare_r3_cross_scene_geometry \
  --source-root "$SOURCE_ROOT" \
  --scene-id "$SCENE_ID" \
  --output-dir "$OUTPUT" \
  --start 90 \
  --stride 15 \
  --frame-count 30 \
  "${OVERWRITE_ARGS[@]}" \
  > "$LOG" 2>&1
STATUS=$?
set -e

if (( STATUS != 0 )); then
  if [[ ! -s "$REPORT" ]]; then
    write_shell_failure "geometry preparation failed; inspect $LOG"
  else
    sed -n '1,120p' "$REPORT"
  fi
  exit "$STATUS"
fi

for artifact in manifest.json summary.json copyable_result.txt run.log; do
  [[ -s "$OUTPUT/$artifact" ]] || {
    write_shell_failure "missing output artifact: $OUTPUT/$artifact"
    exit 1
  }
done

export OUTPUT SCENE_ID
PYTHONPATH="$ROOT:$ROOT/src" "$PYTHON_BIN" - <<'PY' >> "$LOG" 2>&1
import json
import os
from pathlib import Path

output = Path(os.environ["OUTPUT"])
summary = json.loads((output / "summary.json").read_text(encoding="utf8"))
manifest = json.loads((output / "manifest.json").read_text(encoding="utf8"))
assert summary["revision"] == "r3_frozen_cross_scene_geometry_preprocess_r1"
assert summary["status"] == "READY"
assert summary["scene_id"] == os.environ["SCENE_ID"]
assert summary["development_scene_id"] == "00a231a370"
assert summary["heldout_scene_overlap"] == 0
assert summary["frame_count"] == 30
assert summary["source_frame_indices"] == list(range(90, 526, 15))
assert summary["manifest_frame_indices"] == list(range(30))
for field in (
    "annotations_read",
    "semantic_masks_generated",
    "instance_masks_generated",
    "sam_loaded_or_run",
    "streamvggt_loaded_or_run",
    "gt_pointmap_values_used_for_frame_selection",
    "model_parameters_updated",
):
    assert summary[field] == 0, (field, summary[field])
scene = manifest["scenes"][0]
assert scene["scene_id"] == os.environ["SCENE_ID"]
assert len(scene["frames"]) == 30
for index, frame in enumerate(scene["frames"]):
    assert frame["sequence_index"] == index
    assert frame["source_sequence_index"] == 90 + 15 * index
    assert Path(frame["image_path"]).is_file()
    assert Path(frame["pointmap"]).is_file()
    assert frame["valid_pointmap_pixels"] >= 128
    assert "semantic_mask" not in frame
    assert "instance_mask" not in frame
print("r3 geometry-only artifact validation passed")
PY

echo ""
sed -n '1,220p' "$REPORT"
